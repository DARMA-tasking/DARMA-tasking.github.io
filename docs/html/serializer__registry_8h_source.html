57template <typename T>
58struct SerializerEntry {
59
60 template <typename Serializer>
61 SerializerEntry(
62 TypeIdx in_this_idx,
63 TypeIdx in_base_idx,
64 Serializer&& in_serializer
65 ) : this_idx_(in_this_idx),
66 base_idx_(in_base_idx),
67 serializer_(in_serializer)
68 { }
69
70 TypeIdx this_idx_ = no_type_idx;

100template <typename ObjT, typename SerializerT>
101Registrar<ObjT, SerializerT>::Registrar() {
102 auto& reg = getRegistry<ObjT>();
103 index = static_cast<TypeIdx>(reg.size());
104
105 debug_checkpoint(
106 "serializer registrar: %zu, ObjT=%s SerializerT=%s\n",
107 reg.size(),
108 typeid(ObjT).name(),
109 typeid(SerializerT).name()
110 );
111
112 reg.emplace_back(
113 SerializerEntry<ObjT>{
114 index,
115 no_type_idx, // Set later when linkDerivedToBase is called
116 [=](void* s, ObjT& obj) {
117 auto& ser = *reinterpret_cast<SerializerT*>(s);
118 // Disable virtual serializer dispatch because we are already in a
119 // virtualSerialize and otherwise we will recurse indefinitely
120 ser.setVirtualDisabled(true);
121 ser | obj;
122 }
123 }
124 );
125}

127template <typename ObjT, typename SerializerT>
128TypeIdx const Type<ObjT, SerializerT>::idx =
129 Registrar<ObjT, SerializerT>().index;
130
131template <typename ObjT>
132inline auto& getObjIdxRef(TypeIdx han) {
133 checkpointAssert(
134 han < static_cast<TypeIdx>(getRegistry<ObjT>().size()),
135 "Missing type idx in registry. A derived type was not registered with "
136 "checkpoint for a given serializer"
137 );
138 return getRegistry<ObjT>().at(han);
139}
140
141template <typename ObjT>
142inline auto getObjIdx(TypeIdx han) {
143 auto elm = getObjIdxRef<ObjT>(han);
144 return elm.serializer_;
145}
146
147template <typename DerivedT>
148inline std::function<void(void*, DerivedT&)> getBaseIdx(TypeIdx base_idx) {
149 auto& reg = getRegistry<DerivedT>();
150 for (auto entry : reg) {
151 if (entry.base_idx_ == base_idx) {
152 return entry.serializer_;
153 }
154 }
155 checkpointAssert(
156 false, "Error, could not find corresponding entry in derived for base"
157 );
158 return nullptr;
159}
