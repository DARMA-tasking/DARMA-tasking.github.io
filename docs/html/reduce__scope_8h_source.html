63struct ReduceScope {

66 using ValueType = std::variant<

71

73 return in.l0_ == l0_;
74 }
75 bool operator!=(ReduceScope const& in) const {
76 return !(this->operator==(in));
77 }
78
79 ValueType& get() { return l0_; }

81
82 template <typename T, typename... Args>
83 friend ReduceScope makeScope(Args&&... args);
84
85 template <typename SerializerT>
86 void serialize(SerializerT& s) {
87 s | l0_;
88 }
89
90 std::string str() const {
91 if (std::holds_alternative<StrongObjGroup>(l0_)) {
92 return fmt::format("objgroup[{:x}]", std::get<StrongObjGroup>(l0_).get());
93 } else if (std::holds_alternative<StrongVrtProxy>(l0_)) {
94 return fmt::format("vrtproxy[{:x}]", std::get<StrongVrtProxy>(l0_).get());
95 } else if (std::holds_alternative<StrongGroup>(l0_)) {
96 return fmt::format("group[{:x}]", std::get<StrongGroup>(l0_).get());
97 } else if (std::holds_alternative<StrongCom>(l0_)) {
98 return fmt::format("component[{}]", std::get<StrongCom>(l0_).get());
99 } else if (std::holds_alternative<StrongUserID>(l0_)) {
100 return fmt::format("userID[{}]", std::get<StrongUserID>(l0_).get());
101 } else {
102 return "<unknown-type>";
103 }
104 }
105
106private:
107 ValueType l0_;
108};
