65template <typename T, HandleEnum E>
66void Holder<T,E>::allocateDataWindow(std::size_t const in_len) {
67 std::size_t len = in_len == 0 ? count_ : in_len;
68 vt_debug_print(
69 terse, rdma,
70 "allocate: len={}, in_len={}, count_={}\n", len, in_len, count_
71 );
72 // Allocate data window
73 MPI_Comm comm = theContext()->getComm();
74 MPI_Alloc_mem(len * sizeof(T), MPI_INFO_NULL, &data_base_);
75 MPI_Win_create(
76 data_base_, len * sizeof(T), sizeof(T), MPI_INFO_NULL, comm,
77 &data_window_
78 );
79 if (not uniform_size_) {
80 // Allocate control window
81 MPI_Alloc_mem(sizeof(uint64_t), MPI_INFO_NULL, &control_base_);
82 MPI_Win_create(
83 control_base_, sizeof(uint64_t), sizeof(uint64_t), MPI_INFO_NULL, comm,
84 &control_window_
85 );
86 {
87 auto this_node = theContext()->getNode();
88 LockMPI _scope_lock(Lock::Exclusive, this_node, control_window_, false);
89 auto mpi_type = TypeMPI<uint64_t>::getType();
90 MPI_Put(&count_, 1, mpi_type, this_node, 0, 1, mpi_type, control_window_);
91 vt_debug_print(
92 verbose, rdma,
93 "setting allocate size: size={}, window={}\n",
94 count_, print_ptr(&control_window_)
95 );
96 }
97 }
98 ready_ = true;
99}

101template <typename T, HandleEnum E>
102std::size_t Holder<T,E>::getCount(vt::NodeType node, Lock l) {
103 uint64_t result = 0;
104 auto mpi_type = TypeMPI<uint64_t>::getType();
105 {
106 LockMPI _scope_lock(l, node, control_window_);
107 VT_ALLOW_MPI_CALLS;
108 MPI_Get(&result, 1, mpi_type, node, 0, 1, mpi_type, control_window_);
109 }
110 vt_debug_print(
111 verbose, rdma,
112 "getCount: node={}, result={}, window={}\n",
113 node, result, print_ptr(&control_window_)
114 );
115 return static_cast<std::size_t>(result);
116}

118template <typename T, HandleEnum E>
119void Holder<T,E>::deallocate() {
120 VT_ALLOW_MPI_CALLS;
121 if (E == HandleEnum::StaticSize and ready_) {
122 MPI_Win_free(&data_window_);
123 MPI_Free_mem(data_base_);
124 if (not uniform_size_) {
125 MPI_Win_free(&control_window_);
126 MPI_Free_mem(control_base_);
127 }
128 }

145template <typename T, HandleEnum E>
146RequestHolder Holder<T,E>::rget(
147 vt::NodeType node, Lock l, T* ptr, std::size_t len, int offset
148) {
149 auto mpi_type = TypeMPI<T>::getType();
150 auto mpi_type_str = TypeMPI<T>::getTypeStr();
151 RequestHolder r;
152 if (mpi2_) {
153 r.add([this, l, node, ptr, offset, len, mpi_type, mpi_type_str]{
154 LockMPI _scope_lock(l, node, data_window_);
155 vt_debug_print(
156 verbose, rdma,
157 "MPI_Get({}, {}, {}, {}, {}, {}, {}, window);\n",
158 print_ptr(ptr), len, mpi_type_str, node, offset, len, mpi_type_str
159 );
160 VT_ALLOW_MPI_CALLS;
161 MPI_Get(ptr, len, mpi_type, node, offset, len, mpi_type, data_window_);
162 });
163 } else {
164 LockMPI _scope_lock(l, node, data_window_);
165 vt_debug_print(
166 verbose, rdma,
167 "MPI_Rget({}, {}, {}, {}, {}, {}, {}, window);\n",
168 print_ptr(ptr), len, mpi_type_str, node, offset, len, mpi_type_str
169 );
170 VT_ALLOW_MPI_CALLS;
171 MPI_Rget(ptr, len, mpi_type, node, offset, len, mpi_type, data_window_, r.add());
172 }
173 return r;
174}
