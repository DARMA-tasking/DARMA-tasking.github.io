56template <typename MsgT, typename FuncT, ActiveTypedRDMAGetFnType<MsgT>* f>
57RDMA_HandlerType State::setRDMAGetFn(
58 MsgT* msg, FuncT const& fn, bool const& any_tag, TagType const& tag
59) {
60 vt_debug_print(
61 normal, rdma_state,
62 "setRDMAGetFn: GET tag={}, handle={}, any_tag={}\n",
63 tag, handle, print_bool(any_tag)
64 );
65
66 RDMA_HandlerType const handler = makeRdmaHandler(RDMA_TypeType::Get);
67
68 auto const reg_han = auto_registry::makeAutoHandlerRDMAGet<MsgT,f>();
69
70 if (any_tag) {
71 vtAssert(
72 tag == no_tag, "If any tag, you must not have a tag set"
73 );
74 }
75
76 this_get_handler = reg_han;
77 this_rdma_get_handler = handler;
78 user_state_get_msg_ = msg;
79
80 auto const& gen_fn = reinterpret_cast<RDMA_GetFunctionType>(fn);
81
82 if (tag == no_tag) {
83 rdma_get_fn = gen_fn;
84 get_any_tag = any_tag;
85 } else {
86 get_tag_holder[tag] = RDMA_TagGetHolderType{gen_fn,handler,reg_han};
87 }
88
89 return handler;
90}
