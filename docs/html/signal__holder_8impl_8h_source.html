76void SignalHolder<SignalT>::removeListener(
77 PipeType const& pid, ListenerPtrType listener
78) {
79 auto iter = listeners_.find(pid);
80 if (iter != listeners_.end()) {
81 auto liter = iter->second.begin();
82 while (liter != iter->second.end() && liter.get() != listener) ++liter;
83 if (liter != iter->second.end()) {
84 ListenerType elm = std::move(*liter);
85 iter->second.erase(liter);
