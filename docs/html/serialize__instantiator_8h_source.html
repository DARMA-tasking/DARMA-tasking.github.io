52template <typename ObjT, typename SerializerT, typename=void>
53struct InstantiateIfPossible;
54
55template <typename ObjT, typename SerializerT>
56struct InstantiateIfPossible<
57 ObjT,
58 SerializerT,
59 typename std::enable_if_t<
60 SerializableTraits<ObjT, SerializerT>::has_serialize_function or
61 SerializableTraits<ObjT, SerializerT>::is_bytecopyable
62 >
63> {
64 static void registerIt() {
65 dispatch::vrt::serializer_registry::makeObjIdx<ObjT, SerializerT>();
66
67 // If the \c SerializerT static registrations order happen differently for a
68 // given \c BaseT and \c DerviedT , you will get the wrong serializer when
69 // using the base_idx in the derived class. Thus, link the correct base
70 // serialize type idx with a given derived type idx during
71 // sizing/packing. Then, when it unpacks, it can find the right one when
72 // running on the derived type using the base idx.
73 // using BaseType = typename ObjT::_CheckpointVirtualSerializerBaseType;
74 // linkDerivedToBase<SerializerT, ObjT, BaseType>();
75 using BaseType = ::checkpoint::dispatch::vrt::checkpoint_base_type_t<ObjT>;
76 linkDerivedToBase<SerializerT, ObjT, BaseType>();
77 }
78};
79
80template <typename ObjT, typename SerializerT>
81struct InstantiateIfPossible<
82 ObjT,
83 SerializerT,
84 typename std::enable_if_t<
85 not (SerializableTraits<ObjT, SerializerT>::has_serialize_function or
86 SerializableTraits<ObjT, SerializerT>::is_bytecopyable)
87 >
88> {
89 static void registerIt() {
90 // do nothing; this object is not compatible with this serializer!
91 }
92};
93
94template <typename ObjT, typename SerializerT>
95static inline void instantiateObjSerializerImpl() {
96 InstantiateIfPossible<ObjT, SerializerT>::registerIt();
97}

99template <typename ObjT, typename... Ts>
100struct InstantiateTupleHelper;
101
102template <typename ObjT, typename T, typename... Ts>
103struct InstantiateTupleHelper<ObjT, T, Ts...> {
104 static void _recur_register() {
105 instantiateObjSerializerImpl<ObjT, T>();
