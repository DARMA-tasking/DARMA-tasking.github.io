168template <typename ObjT, typename MsgT, ActiveObjType<MsgT, ObjT> fn>
169ObjGroupManager::PendingSendType ObjGroupManager::send(ProxyElmType<ObjT> proxy, MsgSharedPtr<MsgT> msg) {
170 auto const proxy_bits = proxy.getProxy();
171 auto const dest_node = proxy.getNode();
172 auto const ctrl = proxy::ObjGroupProxy::getID(proxy_bits);
173 auto const han = auto_registry::makeAutoHandlerObjGroup<ObjT,MsgT,fn>(ctrl);
174
175 // set bit so it isn't recorded as it routes through bare
176 // handlers
177 envelopeSetCommLBDataRecordedAboveBareHandler(msg->env, true);
178
179 if (theContext()->getTask() != nullptr) {
180 auto dest_elm_id = elm::ElmIDBits::createObjGroup(proxy_bits, dest_node);
181 auto const num_bytes = serialization::MsgSizer<MsgT>::get(msg.get());
182 theContext()->getTask()->send(dest_elm_id, num_bytes);
183 }
184
185 vt_debug_print(
186 terse, objgroup,
187 "ObjGroupManager::send: proxy={:x}, node={}, ctrl={:x}, han={:x}\n",
188 proxy_bits, dest_node, ctrl, han
189 );
190
191 return send<MsgT>(msg,han,dest_node);
192}

213template <typename ObjT, auto f, typename... Args>
214decltype(auto)
215ObjGroupManager::invoke(ProxyElmType<ObjT> proxy, Args&&... args) {
216 auto const dest_node = proxy.getNode();
217 auto const this_node = theContext()->getNode();
218
219 vtAssert(
220 dest_node == this_node,
221 fmt::format(
222 "Attempting to invoke handler on node:{} instead of node:{}!\n",
223 this_node, dest_node
224 )
225 );
226
227#if vt_check_enabled(trace_enabled)
228 auto const ctrl = proxy::ObjGroupProxy::getID(proxy.getProxy());
229 auto const han = auto_registry::makeAutoHandlerObjGroupParam<
230 ObjT, decltype(f), f, void
231 >(ctrl);
232 auto const trace_event = theMsg()->makeTraceCreationSend(han, 0, false);
233
234 return runnable::makeRunnableVoidTraced(false, han, this_node, trace_event, 0)
235 .withObjGroup(get(proxy))
236 .runLambda(f, get(proxy), std::forward<Args>(args)...);
237
238#else
239 return runnable::makeRunnableVoid(false, uninitialized_handler, this_node)
240 .withObjGroup(get(proxy))
241 .runLambda(f, get(proxy), std::forward<Args>(args)...);
242#endif
243}

309template <typename ObjT, typename... Args>
310void ObjGroupManager::update(ProxyElmType<ObjT> proxy, Args&&... args) {
311 auto const this_node = theContext()->getNode();
312 vtAssert(this_node == proxy.getNode(), "You can only update a local obj");
313 auto const proxy_bits = proxy.getProxy();
314 auto iter = objs_.find(proxy_bits);
315 vtAssert(iter != objs_.end(), "Obj must exist on this node");
316 HolderBaseType* holder = iter->second.get();
317 auto obj_holder = static_cast<holder::Holder<ObjT>*>(holder);
318
319 auto const cur_obj_ptr = reinterpret_cast<std::byte*>(obj_holder->get());
320 auto map_iter = obj_to_proxy_.find(cur_obj_ptr);
321 vtAssert(map_iter != obj_to_proxy_.end(), "Object pointer must exist in map");
322 obj_to_proxy_.erase(map_iter);
323
324 obj_holder->reset(std::forward<Args>(args)...);
325
326 auto const new_obj_ptr = reinterpret_cast<std::byte*>(obj_holder->get());
327 obj_to_proxy_[new_obj_ptr] = proxy_bits;
328}
