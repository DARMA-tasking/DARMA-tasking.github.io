89template <typename ColT, typename IndexT, typename BaseProxyT>
90template <auto f, typename... Args>
91decltype(auto)
92Invokable<ColT, IndexT, BaseProxyT>::invoke(Args&&... args) const {
93 auto const& proxy = VrtElmProxy<ColT, IndexT>(
94 this->getCollectionProxy(), this->getElementProxy());
95
96 using MsgT = typename ObjFuncTraits<decltype(f)>::MsgT;
97 if constexpr (std::is_same_v<MsgT, NoMsg>) {
98 return theCollection()->invoke<ColT, f, Args...>(
99 proxy, std::forward<Args>(args)...
100 );
101 } else {
102 auto msg = makeMessage<MsgT>(std::forward<Args>(args)...);
103 return theCollection()->invokeMsg<MsgT, f>(proxy, msg, true);
104 }
105}
