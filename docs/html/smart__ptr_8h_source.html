98template <typename T>
99struct MsgSharedPtr final {
100 using MsgType = T;
101 using BaseMsgType = BaseMessage;
102
103 MsgSharedPtr(std::nullptr_t) {}
104
105 MsgSharedPtr(T* in) {
106 init(in, statics::Holder::getImpl<T>());
107 }
108
109 // Overload to retain ORIGINAL type-erased implementation.
110 MsgSharedPtr(T* in, MsgPtrImplBase* impl) {
111 init(in, impl);
112 }
113
114 MsgSharedPtr(MsgSharedPtr<T> const& in) {
115 if (in != nullptr) {
116 init(in.get(), in.impl_);
117 }
118 }
119
120 MsgSharedPtr(MsgSharedPtr<T>&& in) {
121 moveFrom(std::move(in));
122 }
123
124 MsgSharedPtr<T>& operator=(std::nullptr_t) {
125 clear();
126 return *this;
127 }
128
129 MsgSharedPtr<T>& operator=(MsgSharedPtr<T> const& in) {
130 clear();
131 init(in.get(), in.impl_);
132 return *this;
133 }
134
135 MsgSharedPtr<T>& operator=(MsgSharedPtr<T>&& in) {
136 clear();
137 moveFrom(std::move(in));
138 return *this;
139 }
140
141 ~MsgSharedPtr() {
142 clear();
143 }

145 template <typename U>
146 explicit operator MsgSharedPtr<U>() const { return to<U>(); }
147
148 explicit operator T*() const { return get(); }
149
151 template <typename U>
152 MsgSharedPtr<U> to() const {
153 return MsgSharedPtr<U>(
154 reinterpret_cast<U*>(ptr_),
155 /*N.B. retain ORIGINAL-type implementation*/ impl_);
156 }
157
160 template <typename U>
161 MsgSharedPtr<U> toVirtual() const {
162 return to<U>();
163 }
164
165 bool operator==(MsgSharedPtr<T> const& n) const { return ptr_ == n.ptr_; }
166 bool operator!=(MsgSharedPtr<T> const& n) const { return ptr_ != n.ptr_; }
167 bool operator==(std::nullptr_t) const { return ptr_ == nullptr; }
168 bool operator!=(std::nullptr_t) const { return ptr_ != nullptr; }
169
177 inline T* operator->() const { return get(); }
178
186 inline T& operator*() const { return *ptr_; }
187
195 inline T* get() const {
196 vtAssert(
197 ptr_, "Access attempted of invalid MsgPtr."
198 );
199 return reinterpret_cast<T*>(ptr_);
200 }
201
207 inline ByteType size() const {
208 vtAssert(
209 ptr_, "Access attempted of invalid MsgPtr."
210 );
211
212 // Obtain the size of the message from the block allocated by the
213 // memory pool allocator
214 return thePool()->allocatedSize(reinterpret_cast<std::byte*>(ptr_));
215 }

223 bool ownsMessage() const {
224 return ptr_ not_eq nullptr;

228 auto nrefs = envelopeGetRef(m.get()->env);
229 return os << "MsgSharedPtr("

231 << "size=" << m.size() << ","
232 << "ref=" << nrefs

236 template <
237 typename SerializerT,
238 typename = std::enable_if_t<
239 checkpoint::is_footprinter_v<SerializerT>
240 >
241 >
242 void serialize(SerializerT& s) {
243 if (ownsMessage()) {
244 auto ptr = get();
245 auto const msg_size = vt::serialization::MsgSizer<MsgType>::get(ptr);
246 s.addBytes(msg_size);
247
248 // skip footprinting of members, we rely on message size estimate instead
249 s.skip(impl_);
250 s.skip(ptr_);
251 }
252 }
253
254private:
255
259 void init(T* msgPtr, MsgPtrImplBase* impl) {
260 vtAssert(
261 msgPtr,
262 "MsgPtr cannot wrap 'null' messages."
263 );
264 assert("not initialized" && not ptr_);
265 assert("given impl" && impl);
266
267 ptr_ = msgPtr;
268 impl_ = impl;
269
270 // Could be moved to type-erased impl..
271 messageRef(msgPtr);
272 }
273
276 void clear() {
277 if (not ptr_) {
278 return;
279 }
280
281 T* msgPtr = get();
282
283 impl_->messageDeref(reinterpret_cast<std::byte*>(msgPtr));
284
285 ptr_ = nullptr;
286 }
287
289 void moveFrom(MsgSharedPtr<T>&& in) {
290 ptr_ = in.ptr_;
291 impl_ = in.impl_;
292 // clean take - nullify/prevent other cleanup
293 in.ptr_ = nullptr;
294 }
295
296private:
297 // Underlying raw message - access as correct type via get()
298 BaseMsgType* ptr_ = nullptr;
299 // Type-erased implementation support.
300 // Object has a STATIC LIFETIME / is not owned / should not be deleted.
301 MsgPtrImplBase* impl_ = nullptr;
302};
