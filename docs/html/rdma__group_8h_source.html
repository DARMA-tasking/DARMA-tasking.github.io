70 template <typename Walker>
71 void walk_region(RDMA_RegionType const& region, Walker&& walk) {
72 RDMA_ElmType lo = region.lo;
73 RDMA_ElmType hi = region.hi;
74 RDMA_ElmType elm_hi = lo;
75 RDMA_BlockType blk_lo;
76 NodeType cur_node;
77
78 vtAssertExpr(map.elm_map != nullptr);
79 vtAssertExpr(map.block_map != nullptr);
80
81 while (elm_hi < hi) {
82 auto const& ret = map.elm_map(lo, num_total_elems, num_blocks);
83 blk_lo = std::get<0>(ret);
84 elm_hi = std::get<2>(ret);
85 cur_node = map.block_map(blk_lo, num_blocks);
86
87 walk(cur_node, ret, lo, std::min(elm_hi,hi));
88
89 lo = elm_hi;
90 }
91 }
