53/*******************************************************************************
54 *
55 * Active Function Naming Scheme
56 *
57 *******************************************************************************
58 *
59 * - All registerable active function handlers start with "Active" and end
60 * with "Type", since they are a type alias.
61 *
62 * - If the active function is a pointer, it should end with `FnPtr' rather
63 * than just `Fn'
64 *
65 * -- Non-function pointer
66 *
67 * using ActiveFnType = void(BaseMessage*);
68 *
69 * -- Function pointer
70 *
71 * using ActiveFnPtrType = void(*)(BaseMessage*);
72 *
73 * - If the active function is typed, it should have `Typed' before the
74 * Fn/FnPtr:
75 *
76 * -- Typed function pointer
77 *
78 * template <typename MessageT>
79 * using ActiveTypedFnPtrType = void(*)(MessageT*);
80 *
81 * - If the active function is a closure (can have state associated with
82 * it beyond just a pointer), it should be post-fixed with `Closure'
83 *
84 * using ActiveClosureFnType = std::function<void(BaseMessage*)>;
85 *
86 *******************************************************************************
87 */
88
