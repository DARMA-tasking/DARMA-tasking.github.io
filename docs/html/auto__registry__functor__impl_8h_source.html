58template <typename FunctorT, typename MsgT, bool is_msg_direct>
59inline HandlerType makeAutoHandlerFunctor() {
60 // Arg (overload) differentiaton in adapter.
61 using AdapterType = FunctorAdapterArgs<FunctorT, MsgT, is_msg_direct>;
62 using ContainerType = AutoActiveFunctorContainerType;
63 using RegInfoType = AutoRegInfoType<AutoActiveFunctorType>;
64 using FuncType = ActiveFnPtrType;
65 using RunType =
66 RunnableFunctor<AdapterType, ContainerType, RegInfoType, FuncType, MsgT>;
67
68 constexpr bool is_auto = true;
69 constexpr bool is_functor = true;
70 constexpr auto reg_type = RegistryTypeEnum::RegGeneral;
71 return HandlerManagerType::makeHandler(
72 is_auto, is_functor, RunType::idx, reg_type
73 );
74}

76template <typename RunnableT, typename RegT, typename InfoT, typename FnT>
77struct RegistrarFunctor<
78 RunnableT, RegT, InfoT, FnT,
79 std::enable_if_t<std::is_same<InfoT, AutoRegInfo<AutoActiveType>>::value>
80> {
81 AutoHandlerType index;
82
83 RegistrarFunctor() {
84 using AdapterType = typename RunnableT::AdapterType;
85 using MsgType = typename AdapterType::MsgType;
86 using ObjType = typename AdapterType::ObjType;
87
88 RegT& reg = getAutoRegistryGen<RegT>();
89 index = reg.size(); // capture current index
90
91 auto fn = AdapterType::getFunction();
92 BaseHandlersDispatcherPtr d =
93 std::make_unique<HandlersDispatcher<MsgType, decltype(fn), ObjType>>(fn);
94
95 NumArgsType num_args = AdapterType::getNumArgs();
96
97 #if vt_check_enabled(trace_enabled)
98 std::string event_type_name = AdapterType::traceGetEventType();
99 std::string event_name = AdapterType::traceGetEventName();
100 auto const trace_ep =
101 trace::TraceRegistry::registerEventHashed(event_type_name, event_name);
102 reg.emplace_back(InfoT(NumArgsTag, std::move(d), trace_ep, num_args));
103 #else
104 reg.emplace_back(InfoT(NumArgsTag, std::move(d), num_args));
105 #endif
106 }
107};

143inline NumArgsType getAutoHandlerFunctorArgs(HandlerType const han) {
144 auto const id = HandlerManagerType::getHandlerIdentifier(han);
145
146 using ContainerType = AutoActiveFunctorContainerType;

150inline AutoActiveFunctorType const&
151getAutoHandlerFunctor(HandlerType const han) {
152 auto const id = HandlerManagerType::getHandlerIdentifier(han);
153 bool const is_auto = HandlerManagerType::isHandlerAuto(han);
154 bool const is_functor = HandlerManagerType::isHandlerFunctor(han);
155
156 vt_debug_print(
157 verbose, handler,
158 "getAutoHandlerFunctor: handler={}, id={}, is_auto={}, is_functor={}\n",
159 han, id, print_bool(is_auto), print_bool(is_functor)
160 );
161
162 assert((is_functor && is_auto) && "Handler should be auto and functor type!");
163
164 using ContainerType = AutoActiveFunctorContainerType;
165 return getAutoRegistryGen<ContainerType>().at(id).getFun();
166}
