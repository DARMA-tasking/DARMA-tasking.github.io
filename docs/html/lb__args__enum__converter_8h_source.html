62template <typename T>
63struct LBArgsEnumConverter {
64 using EnumToStrMap = typename std::unordered_map<T, std::string>;
65 using StrToEnumMap = typename std::unordered_map<std::string, T>;
66
74 LBArgsEnumConverter(
75 const std::string &par_name,
76 const std::string &enum_type,
77 const EnumToStrMap &enum_to_str
78 ) : enum_to_str_(enum_to_str),
79 par_name_(par_name),
80 enum_type_(enum_type)
81 {
82 for (const auto &it : enum_to_str_) {
83 str_to_enum_[it.second] = it.first;
84 }
85 }
86
87 virtual ~LBArgsEnumConverter() = default;
88
95 std::string getString(T e) const {
96 auto it = enum_to_str_.find(e);
97 if (it == enum_to_str_.end()) {
98 // this error indicates that you need to update the constructor for
99 // this converter to include all options
100 auto err = fmt::format(
101 "LBArgsEnumConverter: enum '{}' value '{}' corresponding to LB "
102 "argument '{}' does not have a string associated with it",
103 enum_type_, static_cast<std::underlying_type_t<T>>(e), par_name_
104 );
105 vtAbort(err);
106 }
107 return it->second;
108 }
109
116 T getEnum(const std::string &s) const {
117 auto it = str_to_enum_.find(s);
118 if (it == str_to_enum_.end()) {
119 // either the user typed something in wrong or you need to update the
120 // constructor for this converter to include all options
121 auto err = fmt::format(
122 "LBArgsEnumConverter: LB argument '{}' string '{}' is not a recognized "
123 "option for enum '{}'",
124 par_name_, s, enum_type_
125 );
126 vtAbort(err);
127 }
128 return it->second;
129 }
130
137 T getFromConfig(balance::ConfigEntry* config, T def_value) const {
138 std::string config_value = config->getOrDefault<std::string>(
139 par_name_, getString(def_value)
140 );
141 return getEnum(config_value);
142 }
143
144private:
145 EnumToStrMap enum_to_str_; //< 1 to 1 mapping of enumerated values to strings
146 StrToEnumMap str_to_enum_; //< 1 to 1 mapping of strings to enumerated values
147 std::string par_name_; //< parameter to look for in the config file
148 std::string enum_type_; //< name of the enumerated type for error handling
149};
