77template <typename MsgT, typename HandlerT, typename ObjT>
78struct HandlersDispatcher final : BaseHandlersDispatcher {
79 using ColTypedFnType = vrt::collection::ActiveColTypedFnType<MsgT, ObjT>;
80 using ColMemberTypedFnType = vrt::collection::ActiveColMemberTypedFnType<MsgT, ObjT>;
81
82 explicit HandlersDispatcher(HandlerT in_fn_ptr) : fp(in_fn_ptr) { }
83
84 template <typename U>
85 using ColMsgTrait = typename U::IsCollectionMessage;
86 template <typename U>
87 using IsColMsgTrait =
88 detection::is_detected_convertible<std::true_type, ColMsgTrait, U>;
89
90 template <typename U>
91 using WrapMsgTrait = typename U::IsWrapMsg;
92 template <typename U>
93 using IsWrapMsgTrait =
94 detection::is_detected_convertible<std::true_type, WrapMsgTrait, U>;
95
96 template <typename U>
97 using ColTrait = typename U::IsCollectionType;
98 template <typename U>
99 using IsColTrait =
100 detection::is_detected_convertible<std::true_type, ColTrait, U>;
101
102public:
103 void dispatch(messaging::BaseMsg* base_msg, std::byte* object) const override {
104 using T = HandlerT;
105
106 [[maybe_unused]] auto msg = static_cast<MsgT*>(base_msg);
107 [[maybe_unused]] auto elm = reinterpret_cast<ObjT*>(object);
108
109 if constexpr (std::is_same_v<T, ActiveVoidFnType*>) {
110 fp();
111 } else if constexpr (std::is_same_v<T, ActiveTypedFnType<MsgT>*>) {
112 fp(msg);
113 } else if constexpr (std::is_same_v<T, ColTypedFnType*>) {
114 if constexpr (IsColTrait<ObjT>::value and not IsColMsgTrait<MsgT>::value) {
115 auto wrap_msg = static_cast<
116 vrt::collection::ColMsgWrap<ObjT, MsgT, vt::Message>*
117 >(base_msg);
118 fp(elm, &wrap_msg->getMsg());
119 } else {
120 fp(elm, msg);
121 }
122 } else if constexpr (std::is_same_v<T, ColMemberTypedFnType>) {
123 if constexpr (IsColTrait<ObjT>::value and not IsColMsgTrait<MsgT>::value) {
124 auto wrap_msg = static_cast<
125 vrt::collection::ColMsgWrap<ObjT, MsgT, vt::Message>*
126 >(base_msg);
127 (elm->*fp)(&wrap_msg->getMsg());
128 } else {
129 (elm->*fp)(msg);
130 }
131 } else if constexpr (std::is_same_v<ObjT, SentinelObject>) {
132 std::apply(fp, msg->getTuple());
133 } else if constexpr (std::is_same_v<MsgT, void>) {
134 // do nothing
135 } else {
136 if constexpr (IsColTrait<ObjT>::value and IsWrapMsgTrait<MsgT>::value) {
137 auto& m = msg->getMsg();
138 std::apply(fp, std::tuple_cat(std::make_tuple(elm), m.getTuple()));
139 } else {
140 std::apply(fp, std::tuple_cat(std::make_tuple(elm), msg->getTuple()));
141 }
142 }
143 }
144
145private:
146 HandlerT fp = nullptr;
147};

159struct MapsDispatcher final : BaseMapsDispatcher {

162public:

172 static_cast<IndexT*>(cur_idx_ptr),
173 static_cast<IndexT*>(range_ptr),
174 num_nodes
175 );
