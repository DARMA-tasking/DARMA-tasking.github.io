52struct Map {
53 RDMA_BlockMapType block_map = nullptr;
54 RDMA_ElmMapType elm_map = nullptr;
55
56 Map() = default;
57
58 Map(RDMA_BlockMapType in_block_map, RDMA_ElmMapType in_elm_map)
59 : block_map(in_block_map), elm_map(in_elm_map)
60 { }
61
62 static NodeType defaultBlockMap(
63 RDMA_BlockType block, RDMA_BlockType __attribute__((unused)) num_blocks
64 ) {
65 auto const& num_nodes = theContext()->getNumNodes();
66 return block % num_nodes;
67 };
68
69 static RDMA_BlockElmRangeType defaultElmMap(
70 RDMA_ElmType elm, RDMA_ElmType num_elms, RDMA_BlockType num_blocks
71 ) {
72 auto const& block_size = num_elms / num_blocks;
73 auto const& block = elm / block_size;
74 return RDMA_BlockElmRangeType{
75 block, block * block_size, (block + 1) * block_size
76 };
77 }
78
79 template <typename Serializer>
80 void serialize(Serializer& s) {
81 s | block_map
82 | elm_map;
83 }
84};
85
86static Map default_map = Map(Map::defaultBlockMap, Map::defaultElmMap);
