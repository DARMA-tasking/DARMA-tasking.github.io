59template <typename SerializerT, typename T>
60struct BasicDispatcher {
61 static void serializeIntrusive(SerializerT& s, T& t) {
62 t.serialize(s);
63 }

65 static void serializeNonIntrusive(SerializerT& s, T& t) {

152 template <typename U = T>
153 void applyStatic(
154 SerializerT& s, T* val, SerialSizeType num, justFootprint<U>* = nullptr
155 ) {
156 debug_checkpoint(
157 "SerializerDispatch: justFootprint: val=%p typeid=%s\n",
158 static_cast<void*>(&val),
159 typeid(val).name()
160 );
161 static bool firstCall = true;
162 if (firstCall) {
163 firstCall = false;
164 debug_checkpoint(
165 "SerializerDispatch: warning: simplified footprinting in use!\n"
166 "Objects of type: typeid=%s will not be traversed\n",
167 typeid(val).name()
168 );
169 }
170
171 s.contiguousBytes(val, sizeof(T), num);
172 }

174 template <typename U = T>

176 SerializerT& s, T* val, SerialSizeType num, hasInSerialize<U>* = nullptr
177 ) {
178 debug_checkpoint(
179 "SerializerDispatch: intrusive serialize: val=%p typeid=%s\n",
180 static_cast<void*>(&val),
181 typeid(val).name()
182 );
183 for (SerialSizeType i = 0; i < num; i++) {
184 Dispatcher::serializeIntrusive(s, val[i]);
185 applyElm(s, val+i);
186 }

195 static_cast<void*>(&val),

220 ) {
221 if (s.isVirtualDisabled()) {
222 s.setVirtualDisabled(false);
223 applyStatic(s, val, num);
224 } else {
225 dispatch::vrt::virtualSerialize(val, s);
226 }
227 }

234 }
