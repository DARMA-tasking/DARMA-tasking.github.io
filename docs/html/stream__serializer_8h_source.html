54template<typename StreamT = std::ostream>
55struct StreamPacker : BaseSerializer {
56 StreamPacker(SerialSizeType, StreamT& m_stream)
57 : BaseSerializer(ModeType::Packing), stream(m_stream) {
58 //Nothing to do with the size.
59 //Pre-allocating a buffer for the stream has more problems than solutions.
60 }
61
62 void contiguousBytes(void* ptr, SerialSizeType size, SerialSizeType num_elms) {
63 stream.write(static_cast<char*>(ptr), size*num_elms);
64 n_bytes += size*num_elms;
65 }
66
67 SerialSizeType usedBufferSize() {
68 return n_bytes;
69 }
70
71private:
72 StreamT& stream;
73 SerialSizeType n_bytes = 0;
74};

81 void contiguousBytes(void* ptr, SerialSizeType size, SerialSizeType num_elms) {
82 stream.read(static_cast<char*>(ptr), size*num_elms);
83 if(!stream)
84 throw std::runtime_error("Stream unable to read required number of bytes!");
85 n_bytes += size*num_elms;
86 }
87
88 SerialSizeType usedBufferSize() {
89 return n_bytes;

91
92private:
