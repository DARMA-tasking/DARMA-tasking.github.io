69struct PendingRequest {
70 VirtualRequestIDType req_id = no_request_id;
71
72 ActionProxyType action = nullptr;
73
74 PendingRequest(
75 VirtualRequestIDType const& in_req_id, ActionProxyType in_action
76 ) : req_id(in_req_id), action(in_action)
77 { }
78
79 template <typename Serializer>
80 void serialize(Serializer& s) {
81 s | req_id
82 | action;
83 }
84};
