90 bool poll() override {
91 auto ret = has_event_ ? cudaEventQuery(event_) : cudaStreamQuery(stream_);
92 if (cudaSuccess != ret && cudaErrorNotReady != ret) {
93 vtAbort(fmt::format("Failure on stream event: {}: {}", cudaGetErrorName(ret), cudaGetErrorString(ret)));
94 }
95 return ret == cudaSuccess;
96 }
97
101 void done() override {
102 if (has_event_) {
103 vtAbortIf(cudaSuccess != cudaEventDestroy(event_), "Failed to destroy event");
104 }
105 if (cont_) {
106 cont_();
107 }
108 }
109
110private:
