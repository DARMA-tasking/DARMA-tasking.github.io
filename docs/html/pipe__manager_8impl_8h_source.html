69template <typename MsgT>
70void PipeManager::triggerSendBack(
71 PipeType const& pipe, [[maybe_unused]] MsgT* data
72) {
73 auto const& this_node = theContext()->getNode();
74 auto const& node_back = PipeIDBuilder::getNode(pipe);
75 if (node_back != this_node) {
76 // Send the message back to the owner node
77 vtAssertExpr(0);
78 } else {
79 // Directly trigger the action because the pipe meta-data is located here
80 vtAssertExpr(0);
81 }
82}
