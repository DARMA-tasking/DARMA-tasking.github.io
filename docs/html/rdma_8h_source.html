787 template <
788 RDMAManager::RDMA_TypeType rdma_type,
789 typename MsgT, typename FunctionT, ActiveTypedRDMAGetFnType<MsgT>* f
790 >
791 RDMA_HandlerType associateRdmaGetFunction(
792 MsgT* msg, RDMA_HandleType const& han, FunctionT const& fn,
793 bool const& any_tag, TagType const& tag
794 ) {
795 auto const& this_node = theContext()->getNode();
796 auto const handler_node = RDMA_HandleManagerType::getRdmaNode(han);
797 auto const& is_collective = RDMA_HandleManagerType::isCollective(han);
798
799 vtAssert(
800 (is_collective or handler_node == this_node)
801 , "Handle must be local to this node"
802 );
803
804 auto holder_iter = holder_.find(han);
805 vtAssert(
806 holder_iter != holder_.end(), "Holder for handler must exist here"
807 );
808
809 auto& state = holder_iter->second;
810
811 if (rdma_type == RDMAManager::RDMA_TypeType::Get) {
812 return state.setRDMAGetFn<MsgT,FunctionT,f>(msg, fn, any_tag, tag);
813 } else {
814 vtAssert(0, "Should be unreachable");
815 return no_rdma_handle;
816 }
817 }
