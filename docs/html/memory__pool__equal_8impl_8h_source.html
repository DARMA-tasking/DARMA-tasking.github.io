83template <int64_t num_bytes_t, bool use_header>
84std::byte* MemoryPoolEqual<num_bytes_t, use_header>::alloc(
85 size_t const& sz, size_t const& oversize
86) {
87 if (static_cast<size_t>(cur_slot_ + 1) >= holder_.size()) {
88 resizePool();
89 }
90
91 vtAssert(
92 static_cast<size_t>(cur_slot_+1) < holder_.size(),
93 "Must be within pool size, add capability to grow"
94 );
95
96 auto const& slot = cur_slot_;
97 std::byte* const ptr = holder_[slot];
98 std::byte* ptr_ret = ptr;
99 if (use_header) {
100 ptr_ret = HeaderManagerType::setHeader(
101 sz, oversize, ptr
102 );
103 }
104
105 vt_debug_print(
106 normal, pool,
107 "alloc: ptr={}, ptr_ret={} cur_slot={}, sz={}, oversize={}\n",
108 print_ptr(ptr), print_ptr(ptr_ret), cur_slot_, sz, oversize
109 );
110
111 cur_slot_++;
112
113 return ptr_ret;
114}

116template <int64_t num_bytes_t, bool use_header>
117void MemoryPoolEqual<num_bytes_t, use_header>::dealloc(std::byte* const t) {
118 vt_debug_print(
119 normal, pool,
120 "dealloc t={}, cur_slot={}\n", print_ptr(t), cur_slot_
121 );
122
123 vtAssert(
124 cur_slot_ - 1 >= 0, "Must be greater than zero"
125 );
126
127 std::byte* ptr_actual = t;
128 if (use_header) {
129 ptr_actual = HeaderManagerType::getHeaderPtr(t);
130 }
131
132 holder_[--cur_slot_] = ptr_actual;
133}

135template <int64_t num_bytes_t, bool use_header>
136void MemoryPoolEqual<num_bytes_t, use_header>::resizePool() {
137 SlotType const cur_size = holder_.size();
138 SlotType const new_size = cur_size == 0 ? pool_size_ : cur_size * 2;
139
140 holder_.resize(new_size);
141
142 for (auto i = cur_size; i < new_size; i++) {
143 holder_[i] = reinterpret_cast<std::byte*>(malloc(use_header ? num_full_bytes_ : num_bytes_t));
144 }
145}
