180template <typename ColT, typename IndexT, typename BaseProxyT>
181template <auto f, typename... Params>
182messaging::PendingSend
183Broadcastable<ColT, IndexT, BaseProxyT>::broadcast(Params&&... params) const {
184 using MsgT = typename ObjFuncTraits<decltype(f)>::MsgT;
185 if constexpr (std::is_same_v<MsgT, NoMsg>) {
186 using Tuple = typename ObjFuncTraits<decltype(f)>::TupleType;
187 using SendMsgT = ParamColMsg<Tuple, ColT>;
188 auto msg = vt::makeMessage<SendMsgT>();
189 msg->setParams(std::forward<Params>(params)...);
190 auto han = auto_registry::makeAutoHandlerCollectionMemParam<
191 ColT, decltype(f), f, SendMsgT
192 >();
193 auto proxy = this->getProxy();
194 return theCollection()->broadcastMsgUntypedHandler<SendMsgT, ColT, IndexT>(
195 proxy, msg.get(), han, true
196 );
197 } else {
198 auto msg = makeMessage<MsgT>(std::forward<Params>(params)...);
199 return broadcastMsg<MsgT, f>(msg);
200 }
201
202 // Silence nvcc warning (no longer needed for CUDA 11.7 and up)
203 return messaging::PendingSend{std::nullptr_t{}};
204}
