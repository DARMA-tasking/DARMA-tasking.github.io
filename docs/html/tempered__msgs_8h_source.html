57struct ClusterInfo {
58 LoadType load = 0;
59 BytesType bytes = 0;
60 SharedIDType shared_id = -1;
61 double intra_send_vol = 0, intra_recv_vol = 0;
62 std::unordered_map<SharedIDType, double> inter_cluster_send_vol,
63 inter_cluster_recv_vol;
64 std::unordered_map<elm::ElementIDStruct, double> obj_send_vol, obj_recv_vol;
65 NodeType home_node = uninitialized_destination;
66 BytesType edge_weight = 0;
67 BytesType max_object_working_bytes = 0;
68 BytesType max_object_working_bytes_outside = 0;
69 BytesType max_object_serialized_bytes = 0;
70 BytesType max_object_serialized_bytes_outside = 0;
71 BytesType cluster_footprint = 0;
72
73 void addInterClusterEdge(bool is_send, SharedIDType id, double volume) {
74 if (is_send) {
75 inter_cluster_send_vol[id] += volume;
76 } else {
77 inter_cluster_recv_vol[id] += volume;
78 }
79 }
80
81 void addIntraVolume(bool is_send, double volume) {
82 if (is_send) {
83 intra_send_vol += volume;
84 } else {
85 intra_recv_vol += volume;
86 }
87 }
88
89 void addObjEdge(bool is_send, elm::ElementIDStruct obj, double volume) {
90 if (is_send) {
91 obj_send_vol[obj] += volume;
92 } else {
93 obj_recv_vol[obj] += volume;
94 }
95 }
96
97 template <typename SerializerT>
98 void serialize(SerializerT& s) {
99 s | load | shared_id | bytes | intra_send_vol | intra_recv_vol;
100 s | inter_cluster_send_vol | inter_cluster_recv_vol;
101 s | obj_send_vol | obj_recv_vol;
102 s | home_node | edge_weight;
103 s | max_object_working_bytes;
104 s | max_object_working_bytes_outside;
105 s | max_object_serialized_bytes;
106 s | max_object_serialized_bytes_outside;
107 s | cluster_footprint;
108 }
109};
