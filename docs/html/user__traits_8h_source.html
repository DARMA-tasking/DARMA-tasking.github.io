179 //Has any types within T
180 template<typename... T>
181 using has_any = std::disjunction<has_trait<T>...>;

183 //Type with each trait T removed once, if present.

185 // ::Without<TraitOne, TraitTwo>
186 // == TraitHolder<TraitOne, TraitThree>;
187 template<typename... T>
