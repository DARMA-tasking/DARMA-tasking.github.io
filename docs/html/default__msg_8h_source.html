82 ReduceMsg,

100
101 explicit ReduceDataMsg(DataType&& in_val)
102 : MessageParentType(), ReduceCombine<void>(),
103 val_(std::forward<DataType>(in_val))
104 { }
105 explicit ReduceDataMsg(DataType const& in_val)
106 : MessageParentType(), ReduceCombine<void>(), val_(in_val)
107 { }
108
109 DataType& getTuple() { return val_; }
110 DataType const& getConstVal() const { return val_; }
111 DataType& getVal() { return val_; }
112 DataType&& getMoveVal() { return std::move(val_); }
113
114 bool isMsgCallback() const { return cb_.index() == 0; }
115 bool isParamCallback() const { return cb_.index() == 1; }
116 CallbackMsgType getMsgCallback() { return std::get<0>(cb_); }
117 CallbackParamType getParamCallback() { return std::get<1>(cb_); }
118 bool hasValidCallback() {
119 if (isMsgCallback()) {
120 return getMsgCallback().valid();
121 } else {
122 return getParamCallback().valid();
123 }
124 }

126 template <typename CallbackT>
127 void setCallback(CallbackT cb) {
128 if constexpr (std::is_same_v<CallbackT, CallbackParamType>) {
129 cb_ = cb;
130 } else if (
131 std::is_convertible_v<
132 typename GetCallbackType<CallbackT>::MsgT*, ReduceDataMsg<DataType>*
133 >
134 ) {
135 auto cb_ptr = reinterpret_cast<Callback<ReduceDataMsg<DataType>>*>(&cb);
136 cb_ = *cb_ptr;
137 } else {
138 static_assert(
139 std::is_same_v<CallbackT, CallbackParamType> or
140 std::is_convertible_v<
141 typename GetCallbackType<CallbackT>::MsgT*, ReduceDataMsg<DataType>*
142 >,
143 "Must be a convertible message callback or parameterized callback"
144 );
145 }
146 }
147
148 template <typename SerializeT>
149 void serialize(SerializeT& s) {
150 MessageParentType::serialize(s);
151 s | val_;
152 int index = cb_.index();
153 s | index;
154 if (s.isUnpacking()) {
155 if (index == 0) {
156 CallbackMsgType cb;
157 s | cb;
158 cb_ = cb;
159 } else {
160 CallbackParamType cb;
161 s | cb;
162 cb_ = cb;
163 }
164 } else {
165 if (index == 0) {
166 s | std::get<0>(cb_);
167 } else {
168 s | std::get<1>(cb_);
169 }
170 }
171 }
172
173protected:
174 DataType val_ = {};
175 std::variant<CallbackMsgType, CallbackParamType> cb_;
176};
