131 void add(EpochType new_epoch, PendingSend&& link) {
132 checkInit();
133
134 // Produce an event here, to be consumed when the send gets
135 // released, so that new_epoch doesn't appear to have completed in
136 // the interim
137 theTerm()->addDependency(last_epoch_, new_epoch);
138 theTerm()->addActionUnique(last_epoch_, PendingClosure(std::move(link)));
139
140 last_epoch_ = new_epoch;
141 }
142
151 static void mergeChainStep(DependentSendChain &a, DependentSendChain &b,
152 EpochType new_epoch, PendingSend&& link) {
153 a.checkInit();
154 b.checkInit();
155
156 theTerm()->addDependency(a.last_epoch_, new_epoch);
157 theTerm()->addDependency(b.last_epoch_, new_epoch);
158
159 auto c1 = MergedClosure(std::make_shared<PendingSend>(std::move(link)));
160 auto c2 = c1;
161
162 // closure is intentionally copied here; basically the ref count will go down
163 // when all actions are completed and execute the PendingSend
164 theTerm()->addActionUnique(a.last_epoch_, std::move(c1));
165 theTerm()->addActionUnique(b.last_epoch_, std::move(c2));
166
167 a.last_epoch_ = new_epoch;
168 b.last_epoch_ = new_epoch;
169 }
170
174 void done() {
175 reset();
176 }
177
183 bool isTerminated() {
184 return theTerm()->testEpochTerminated(last_epoch_) == vt::term::TermStatusEnum::Terminated;
185 }

201 void checkInit() {
202 // Done here rather than the constructor to make static
203 // initialization of an instance safe
204 if (last_epoch_ == no_epoch) {
205 reset();
206 }
207 }
208
212 void reset() {
213 // Set up a 'closed' epoch so that we keep an invariant of always
214 // having an epoch to call addAction on, rather than edge cases of
215 // whether we've made one yet or not
216
217 // The parameter `true` here tells VT to use an efficient rooted DS-epoch
218 // by default. This can still be overridden by command-line flags
219 last_epoch_ = theTerm()->makeEpochRooted("DependentSendChain::reset", term::UseDS{true});
220 theTerm()->finishedEpoch(last_epoch_);
221 }
222
223 EpochType last_epoch_ = no_epoch;
224};
