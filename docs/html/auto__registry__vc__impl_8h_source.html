59inline HandlerType makeAutoHandlerVC() {
60 using FunctorT = FunctorAdapter<ActiveVrtTypedFnType<MsgT, VrtT>, f, MsgT, VrtT>;
61 using ContainerType = AutoActiveVCContainerType;
62 using RegInfoType = AutoRegInfoType<AutoActiveVCType>;
63 using FuncType = ActiveVirtualFnPtrType;
64
65 auto const id =
66 RunnableGen<FunctorT, ContainerType, RegInfoType, FuncType>::idx;
67 constexpr auto reg_type = RegistryTypeEnum::RegVrt;

69}
