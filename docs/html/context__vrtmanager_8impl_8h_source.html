84template <typename SysMsgT>
85/*static*/ void VirtualContextManager::remoteConstructVrt(SysMsgT* msg) {
86 using VrtContextT = typename SysMsgT::VirtualContextType;
87 using Args = typename SysMsgT::ArgsTupleType;
88
89 static constexpr auto size = std::tuple_size<Args>::value;
90 auto new_vc = VirtualContextManager::runConstructor<VrtContextT>(
91 &msg->tup, std::make_index_sequence<size>{}
92 );
93
94 auto const& info = msg->info;
95 VirtualProxyType new_proxy = info.proxy;
96
97 if (info.isImmediate) {
98 // nothing to do here?
99 } else {
100 auto const& cons_node = theContext()->getNode();
101 auto const& req_node = info.from_node;
102 auto const& request_id = info.req_id;
103
104 new_proxy = theVirtualManager()->generateNewProxy();
105 auto send_msg = makeMessage<VirtualProxyRequestMsg>(
106 cons_node, req_node, request_id, new_proxy
107 );
108 theMsg()->sendMsg<sendBackVirtualProxyHan>(
109 req_node, send_msg
110 );
111 }
112
113 theVirtualManager()->insertVirtualContext(std::move(new_vc), new_proxy);
114}

135template <typename VcT, typename MsgT, ActiveVrtTypedFnType<MsgT, VcT> *f>
136messaging::PendingSend VirtualContextManager::sendSerialMsg(
137 VirtualProxyType const& toProxy, MsgT *const msg
138) {
139 auto base_msg = promoteMsg(msg).template to<BaseMsgType>();
140
141 NodeType const& home_node = VirtualProxyBuilder::getVirtualNode(toProxy);
142 // register the user's handler
143 HandlerType const han = auto_registry::makeAutoHandlerVC<VcT,MsgT,f>();
144 // save the user's handler in the message
145 msg->setVrtHandler(han);
146 msg->setProxy(toProxy);
147
148 vt_debug_print(
149 normal, vrt,
150 "sending serialized msg to VC: msg={}, han={}, home_node={}, toProxy={}\n",
151 print_ptr(msg), han, home_node, toProxy
152 );
153
154 using SerialMsgT = SerializedEagerMsg<MsgT, VirtualMessage>;
155
156 // route the message to the destination using the location manager
157 messaging::PendingSend pending(
158 base_msg, [=](MsgPtr<BaseMsgType> mymsg){
159 // Uses special implementation overload not exposed in theMsg..
160 MsgT* typed_msg = reinterpret_cast<MsgT*>(mymsg.get());
161 auto sendSerialHan = auto_registry::makeAutoHandler<MsgT,virtualTypedMsgHandler<MsgT>>();
162 SerializedMessenger::sendSerialMsgSendImpl<MsgT, VirtualMessage>(
163 typed_msg,
164 sendSerialHan,
165 // custom send lambda to route the message
166 [=](MsgSharedPtr<SerialMsgT> innermsg) -> messaging::PendingSend {
167 innermsg->setProxy(toProxy);
168 theLocMan()->vrtContextLoc->routeMsgHandler<
169 SerialMsgT, SerializedMessenger::payloadMsgHandler
170 >(toProxy, home_node, innermsg);
171 return messaging::PendingSend(nullptr);
172 },
173 // custom data transfer lambda if above the eager threshold
174 [=](ActionNodeSendType action) -> messaging::PendingSend {
175 auto captured_action = [=](NodeType node){ action(node); };
176 theLocMan()->vrtContextLoc->routeNonEagerAction(
177 toProxy, home_node, captured_action
178 );
179 return messaging::PendingSend(nullptr);
180 }
181 );
182 }
183 );
184 return pending;
185}

187template <typename VrtContextT, typename... Args>
188VirtualProxyType VirtualContextManager::makeVirtualRemote(
189 NodeType const& dest, bool isImmediate, ActionProxyType action, Args&&... args
190) {
191 using ArgsTupleType = std::tuple<typename std::decay<Args>::type...>;
192 using MsgType = VrtConstructMsg<RemoteVrtInfo, ArgsTupleType, VrtContextT>;
193
194 auto sys_msg = makeMessage<MsgType>(ArgsTupleType{std::forward<Args>(args)...});
195
196 auto const& this_node = theContext()->getNode();
197 std::unique_ptr<RemoteVrtInfo> info = nullptr;
198 VirtualProxyType return_proxy = no_vrt_proxy;
199
200 if (isImmediate) {
201 auto const& remote_id = generateNewRemoteID(dest);
202 return_proxy = VirtualProxyBuilder::createRemoteProxy(
203 remote_id, myNode_, dest, false, false
204 );
205 info = std::make_unique<RemoteVrtInfo>(this_node, return_proxy);
206 } else {
207 auto next_req = cur_request_id++;
208
209 // insert a pending request to trigger the action
210 pending_request_.emplace(
211 std::piecewise_construct,
212 std::forward_as_tuple(next_req),
213 std::forward_as_tuple(PendingRequestType{next_req, action})
214 );
215
216 info = std::make_unique<RemoteVrtInfo>(this_node, next_req);
217 }
218
219 sys_msg->info = *info.get();
220
221 theMsg()->sendMsg<remoteConstructVrt<MsgType>>(dest, sys_msg);
222
223 return return_proxy;
224}

258template <typename VcT, typename MsgT, ActiveVrtTypedFnType<MsgT, VcT> *f>
259messaging::PendingSend VirtualContextManager::sendMsg(
260 VirtualProxyType const& toProxy, MsgT *const raw_msg
261) {
262 auto msg = promoteMsg(raw_msg);
263
264 auto const& home_node = VirtualProxyBuilder::getVirtualNode(toProxy);
265 // register the user's handler
266 auto const& han = auto_registry::makeAutoHandlerVC<VcT,MsgT,f>();
267 // save the user's handler in the message
268 msg->setVrtHandler(han);
269 msg->setProxy(toProxy);
270
271 vt_debug_print(
272 normal, vrt,
273 "sending msg to VC: msg={}, han={}, home_node={}\n",
274 print_ptr(msg.get()), han, home_node
275 );
276
277 auto base_msg = msg.template to<BaseMsgType>();
278 return messaging::PendingSend(base_msg,
279 [=](MsgPtr<BaseMsgType> mymsg){
280 // route the message to the destination using the location manager
281 auto msg_shared = promoteMsg(reinterpret_cast<MsgT*>(mymsg.get()));
282 theLocMan()->vrtContextLoc->routeMsg(toProxy, home_node, msg_shared);
283 }
284 );
285}
