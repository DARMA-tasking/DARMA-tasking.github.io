145 bool testEnabledFor(SpecIndex in_idx) const {
146 if (is_mod_) {
147 // Check if mod idx_ is within range, by dividing out idx_ we get the
148 // pos/neg offset from that mod value
149 return in_idx % idx_ >= idx_ + neg_ or in_idx % idx_ <= pos_;
150 } else {
151 // Check intersection with range: [idx_+neg_, idx_+pos_]
152 return in_idx >= idx_ + neg_ and in_idx <= idx_ + pos_;
153 }
154 }

161 template <typename SerializerT>
162 void serialize(SerializerT& s) {
163 s | idx_ | neg_ | pos_ | is_mod_;
164 }
