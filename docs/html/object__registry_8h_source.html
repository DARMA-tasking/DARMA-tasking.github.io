99template <typename ObjT>
100Registrar<ObjT>::Registrar() {
101 using BaseType = ::checkpoint::dispatch::vrt::checkpoint_base_type_t<ObjT>;
102
103 auto& reg = getRegistry<BaseType>();
104 index = static_cast<TypeIdx>(reg.size());
105
106 debug_checkpoint("object registrar: %zu, %s\n", reg.size(), typeid(ObjT).name());
107
108 reg.emplace_back(
109 ObjectEntry<BaseType>{
110 index,
111 sizeof(ObjT),
112 []() -> void* { return std::allocator<ObjT>{}.allocate(1); },
113 [](void* buf) -> BaseType* {
114 using ReconsObjT = dispatch::Reconstructor<ObjT>;
115 return ReconsObjT::constructAllowFail(buf).transferOwnership();
116 }
117 }
118 );
119}
