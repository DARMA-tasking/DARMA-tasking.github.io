144 bool selfEdge() const { return cat_ == CommCategory::SendRecv and from_ == to_; }
145 bool offNode() const {
146 if (cat_ == CommCategory::SendRecv) {
147 return objGetNode(from_) != objGetNode(to_);
148 } else if (cat_ == CommCategory::CollectionToNode) {
149 return objGetNode(from_) != nto_;
150 } else if (cat_ == CommCategory::NodeToCollection) {
151 return objGetNode(to_) != nfrom_;
152 } else {
153 return true;
154 }

163 }

175struct CommVolume {
176 CommBytesType bytes = 0.0;
177 uint64_t messages = 0;
178
179 void receiveMsg(double b) {
180 messages++;
181 bytes += b;
182 }
183
184 void sendMsg(double b) {
185 messages++;
186 bytes += b;
187 }
188
189 void operator+=(const CommVolume &rhs) {
190 bytes += rhs.bytes;
191 messages += rhs.messages;
192 }
193
194 template <typename SerializerT>
195 void serialize(SerializerT &s) {
196 s | bytes | messages;
197 }
