75 friend LoadData operator+(LoadData a1, LoadData const& a2) {
76 vt_debug_print(
77 verbose, lb,
78 "operator+: a1.N_={}, a2.N_={}\n", a1.N_, a2.N_
79 );
80
81 vt_debug_print(
82 verbose, lb,
83 "operator+: a1.avg_={}, a2.avg_={}\n", a1.avg_, a2.avg_
84 );
85
86 int32_t N = a1.N_ + a2.N_;
87 auto delta = a2.avg_ - a1.avg_;
88 auto delta_sur_N = delta / static_cast<double>(N);
89 auto delta2_sur_N2 = delta_sur_N * delta_sur_N;
90 int32_t n2 = a1.N_ * a1.N_;
91 int32_t n_c2 = a2.N_ * a2.N_;
92 int32_t prod_n = a1.N_ * a2.N_;
93 int32_t n_c = a2.N_;
94 int32_t n = a1.N_;
95 auto M2 = a1.M2_;
96 auto M2_c = a2.M2_;
97 auto M3 = a1.M3_;
98 auto M3_c = a2.M3_;
99
100 a1.M4_ += a2.M4_
101 + prod_n * ( n2 - prod_n + n_c2 ) * delta * delta_sur_N * delta2_sur_N2
102 + 6. * ( n2 * M2_c + n_c2 * M2 ) * delta2_sur_N2
103 + 4. * ( n * M3_c - n_c * M3 ) * delta_sur_N;
104
105 a1.M3_ += a2.M3_
106 + prod_n * ( n - n_c ) * delta * delta2_sur_N2
107 + 3. * ( n * M2_c - n_c * M2 ) * delta_sur_N;
108
109 a1.M2_ += a2.M2_ + prod_n * delta * delta_sur_N;
110 a1.avg_ += n_c * delta_sur_N;
111 a1.N_ = N;
112 a1.min_ = std::min(a1.min_, a2.min_);
113 a1.max_ = std::max(a1.max_, a2.max_);
114 a1.sum_ += a2.sum_;
115 a1.P_ += a2.P_;
116 a1.stat_ = a2.stat_;
117
118 return a1;
119 }

121 LoadType max() const { return max_; }
122 LoadType sum() const { return sum_; }
123 LoadType min() const { return min_; }
124 LoadType avg() const { return avg_; }
125 LoadType var() const { return N_ > 0 ? M2_ * (1.0f / N_) : 0.0; }
126 LoadType skew() const {
127 static const double min_sqrt = std::sqrt(std::numeric_limits<double>::min());
128 if (N_ == 1 or M2_ < min_sqrt) { // 1.e-150
129 return 0.0;
130 } else {
131 double nm1 = N_ - 1;
132 double inv_n = 1. / N_;
133 auto var_inv = nm1 / M2_;
134 auto nvar_inv = var_inv * inv_n;
135 return nvar_inv * std::sqrt( var_inv ) * M3_;
136 }
137 }
138 LoadType krte() const {
139 if (N_ == 1 or M2_ < 1.e-150) {
140 return 0.0;
141 } else {
142 double nm1 = N_ - 1;
143 double inv_n = 1. / N_;
144 auto var_inv = nm1 / M2_;
145 auto nvar_inv = var_inv * inv_n;
146 return nvar_inv * var_inv * M4_ - 3.;
147 }
148 }
149 LoadType I() const { return avg() > 0.0 ? (max() / avg()) - 1.0f : 0.0; }
150 LoadType stdv() const { return std::sqrt(var()); }
151 int32_t npr() const { return P_; }
152
153 static_assert(
154 std::is_same<LoadType, double>::value == true,
155 "LoadType must be a double"
156 );
157
158 LoadType max_ = 0.0;
159 LoadType sum_ = 0.0;
160 LoadType min_ = 0.0;
161 LoadType avg_ = 0.0;
162 LoadType M2_ = 0.0;
163 LoadType M3_ = 0.0;
164 LoadType M4_ = 0.0;
165 int32_t N_ = 0;
166 int32_t P_ = 0;
167 lb::Statistic stat_ = lb::Statistic::Rank_load_modeled;
168};
