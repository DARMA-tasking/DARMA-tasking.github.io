289template <CatEnum cat, ModeEnum mod, typename... Args>
290static inline void debugPrintImpl(
291 NodeType node, fmt::format_string<Args...> arg, Args&&... args
292) {
293 constexpr auto mask = ModeEnum::terse | ModeEnum::normal | ModeEnum::verbose;
294 constexpr auto level = mod & mask;
295 if (level <= vt::debug::preConfig()->vt_debug_level_val) {
296 auto user = fmt::format(arg, std::forward<Args>(args)...);
297 std::string debug_level = "";
298 if (level == ModeEnum::terse) {
299 debug_level = "(t)";
300 } else if (level == ModeEnum::normal) {
301 debug_level = "(n)";
302 } else if (level == ModeEnum::verbose) {
303 debug_level = "(v)";
304 }
305 fmt::print(
306 "{} {} {} {} {}",
307 vt_print_colorize,
308 vt_proc_print_colorize(node),
309 vt_print_colorize_impl(::vt::debug::yellow(), debug_level, ""),
310 vt_print_colorize_impl(
311 ::vt::debug::green(), PrettyPrintCat<cat>::print(), ":"
312 ),
313 user
314 );
315 bool const flush_enabled = vt_option_check_enabled(mod, ModeEnum::flush);
316 if (flush_enabled or vt::debug::preConfig()->alwaysFlush()) {
317 fflush(stdout);
318 }
319 }
320}

322template <CatEnum cat, ModeEnum mod>
323struct DebugPrintOp<cat, CtxEnum::node, mod> {
324 template <typename... Args>
325 void operator()(
326 bool const rt_option, fmt::format_string<Args...> arg, Args&&... args
327 ) {
328 if (rt_option or vt::debug::preConfig()->vt_debug_all) {
329 debugPrintImpl<cat, mod>(
330 vt::debug::preNode(), arg, std::forward<Args>(args)...
331 );
332 }
333 }
334};

337struct DebugPrintOp<cat, CtxEnum::unknown, mod> {

341 ) {
342 if (rt_option or vt::debug::preConfig()->vt_debug_all) {
343 debugPrintImpl<cat,mod>(-1, arg, std::forward<Args>(args)...);
344 }
