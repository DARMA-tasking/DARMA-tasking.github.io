218 ActionType in_dealloc_user_buf,

220 PriorityType in_priority
221 ) : InProgressBase{in_buf, in_probe_bytes, in_sender},

227 int flag = 0;
228 MPI_Status stat;
229 for ( ; cur < reqs.size(); cur++) {
230 VT_ALLOW_MPI_CALLS; // MPI_Test
231
232 MPI_Test(&reqs[cur], &flag, &stat);
233 num_mpi_tests++;
234
235 if (flag == 0) {
236 return false;
237 }
238 }
239
240 return true;
241 }
