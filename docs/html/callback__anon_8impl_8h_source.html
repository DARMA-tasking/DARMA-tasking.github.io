67template <typename MsgT>
68template <typename T>
69typename CallbackAnon<MsgT>::template IsVoidType<T>
70CallbackAnon<MsgT>::triggerDispatch(
71 [[maybe_unused]] SignalDataType* data, PipeType const& pid
72) {
73 // Overload when the signal is void
74 auto const& this_node = theContext()->getNode();
75 auto const& pipe_node = PipeIDBuilder::getNode(pid);
76 vt_debug_print(
77 terse, pipe,
78 "CallbackAnon: (void signal) trigger_: pipe_={:x}, pipe_node={}\n",
79 pid, pipe_node
80 );
81 if (this_node == pipe_node) {
82 triggerPipe(pid);
83 } else {
84 auto msg = makeMessage<CallbackMsg>(pid);
85 theMsg()->sendMsg<triggerCallbackHan>(pipe_node, msg);
86 }
87}
88
89template <typename MsgT>
90template <typename T>
91typename CallbackAnon<MsgT>::template IsNotVoidType<T>
92CallbackAnon<MsgT>::triggerDispatch(SignalDataType* data, PipeType const& pid) {
93 // Overload when the signal is non-void
94 auto const& this_node = theContext()->getNode();
95 auto const& pipe_node = PipeIDBuilder::getNode(pid);
96 vt_debug_print(
97 terse, pipe,
98 "CallbackAnon: (T signal) trigger_: pipe_={:x}, pipe_node={}\n",
99 pid, pipe_node
100 );
101 if (this_node == pipe_node) {
102 triggerPipeTyped<T>(pid,data);
103 } else {
104 /*
105 * Set pipe type on the message envelope; use the group in the envelope in
106 * indicate the pipe
107 */
108 setPipeType(data->env);
109 envelopeSetGroup(data->env,pid);
110 theMsg()->sendMsg<triggerCallbackMsgHan>(pipe_node, data);
111 }
112}
