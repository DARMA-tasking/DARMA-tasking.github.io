59template <
60 typename Serializer, typename Key, typename Value, typename Device,
61 typename Hasher, typename EqualTo
62>
63void serializeKokkosUnorderedMapElems(
64 Serializer& s,
65 Kokkos::UnorderedMap<Key, Value, Device, Hasher, EqualTo>& map
66) {
67 using mapSizeType =
68 typename Kokkos::UnorderedMap<Key, Value, Device, Hasher, EqualTo>::size_type;
69
70 auto host_map = Kokkos::create_mirror(map);
71 Kokkos::deep_copy(host_map,map);
72
73 for (mapSizeType i = 0; i < map.capacity(); i++) {
74 Key keyAtI = host_map.key_at(i);
75 if (host_map.exists(keyAtI)) {
76 Value val = host_map.value_at(host_map.find(keyAtI));
77
78 s | keyAtI;
79 s | val;
80 }
81 }
82}

84template <
85 typename Serializer, typename Key, typename Value, typename Device,
86 typename Hasher, typename EqualTo
87>
88void deserializeInsertElems(
89 Serializer& s, Kokkos::UnorderedMap<Key, Value, Device, Hasher, EqualTo>& map,
90 typename Kokkos::UnorderedMap<Key, Value, Device, Hasher, EqualTo>::size_type
91 map_size,
92 typename Kokkos::UnorderedMap<Key, Value, Device, Hasher, EqualTo>::size_type
93 map_capacity
94) {
95 using KeyAlloc = dispatch::Allocator<Key>;
96 using ValueAlloc = dispatch::Allocator<Value>;
97 using KeyReconstructor =
98 dispatch::Reconstructor<typename dispatch::CleanType<Key>::CleanT>;
99 using ValReconstructor =
100 dispatch::Reconstructor<typename dispatch::CleanType<Value>::CleanT>;
101
102 // resize unordered map
103 map.rehash(map_capacity);
104 auto host_map = Kokkos::create_mirror(map);
105 host_map.rehash(map_capacity);
106
107 KeyAlloc keyAllocated;
108 ValueAlloc valAllocated;
109 for (SerialSizeType i = 0; i < map_size; i++) {
110 auto key = KeyReconstructor::construct(keyAllocated.buf);
111 auto val = ValReconstructor::construct(valAllocated.buf);
112
113 s | *key;
114 s | *val;
115
116 host_map.insert(std::move(*key), std::move(*val));
117 }
118
119 Kokkos::deep_copy(map,host_map);
120}

122template <
123 typename SerializerT, typename Key, typename Value, typename Device,
124 typename Hasher, typename EqualTo
125>
126typename std::enable_if_t<
127 not checkpoint::is_footprinter_v<SerializerT>, void
128> serialize(
129 SerializerT& s,
130 Kokkos::UnorderedMap<Key, Value, Device, Hasher, EqualTo>& map
131) {
132 using UnorderedMapType =
133 Kokkos::UnorderedMap<Key, Value, Device, Hasher, EqualTo>;
134
135 typename UnorderedMapType::size_type size = serializeContainerSize(s, map);
136 typename UnorderedMapType::size_type capacity = serializeContainerCapacity(s, map);
137
138 if (s.isUnpacking()) {
139 deserializeInsertElems(s, map, size, capacity);
140 } else {
141 serializeKokkosUnorderedMapElems(s, map);
142 }
143}

145template <
146 typename SerializerT, typename Key, typename Value, typename Device,
147 typename Hasher, typename EqualTo
148>
149typename std::enable_if_t<
150 checkpoint::is_footprinter_v<SerializerT>, void
151> serialize(
152 SerializerT& s,
153 Kokkos::UnorderedMap<Key, Value, Device, Hasher, EqualTo>& map
154) {
155 using UnorderedMapType =
156 Kokkos::UnorderedMap<Key, Value, Device, Hasher, EqualTo>;
157
158 typename UnorderedMapType::size_type size = serializeContainerSize(s, map);
159 typename UnorderedMapType::size_type capacity = serializeContainerCapacity(s, map);
160
161 s.addBytes(capacity * sizeof(Key) + size * sizeof(Value));
162}
