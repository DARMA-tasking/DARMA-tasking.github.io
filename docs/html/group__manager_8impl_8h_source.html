69template <typename T>
70void GroupManagerT<T>::pushCleanupAction() {
71 // Push the typeless cleanup actions
72 if (continuation_actions_t_.size() == 0) {
73 theGroup()->addCleanupAction([]{ continuation_actions_t_.clear(); });
74 }
75 if (waiting_cont_.size() == 0) {
76 theGroup()->addCleanupAction([]{ waiting_cont_.clear(); });
77 }
78}

80template <typename T>
81RemoteOperationIDType GroupManagerT<T>::registerContinuationT(ActionTType act) {
82 pushCleanupAction();

85 continuation_actions_t_.emplace(

87 std::forward_as_tuple(next_id),

107template <typename T>
108/*static*/ void GroupManagerT<T>::triggerWaitingContinuations(
109 RemoteOperationIDType const op
110) {
111 auto action_iter = continuation_actions_t_.find(op);
112 auto const found = action_iter != continuation_actions_t_.end();
113
114 if (found) {
115 auto iter = waiting_cont_.find(op);
116 if (iter != waiting_cont_.end()) {
117 for (auto&& elm : iter->second) {
118 for (auto&& action : action_iter->second) {
119 action(elm);
120 }
121 }
122 waiting_cont_.clear();
123 }
124 }
125}

127template <typename T>
128void GroupManagerT<T>::triggerContinuationT(
129 RemoteOperationIDType const op, T t
130) {
131 auto iter = continuation_actions_t_.find(op);
132 auto const& found = iter != continuation_actions_t_.end();
133
134 vt_debug_print(
135 verbose, group,
136 "GroupManager::triggerContinuationT: op={:x}, found={}, size={}\n",
137 op, found, continuation_actions_t_.size()
138 );
139
140 if (found) {
141 for (auto&& elm : iter->second) {
142 elm(t);
143 }
144 } else {
145 auto iter_wait = waiting_cont_.find(op);
146 if (iter_wait == waiting_cont_.end()) {
147 waiting_cont_.emplace(
148 std::piecewise_construct,
149 std::forward_as_tuple(op),
150 std::forward_as_tuple(std::vector<T>{t})
151 );
152 } else {
153 waiting_cont_[op].push_back(t);
154 }
155 }
156}
