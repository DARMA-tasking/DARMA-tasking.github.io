78template <typename... Args, typename... Args2>
79inline
80std::enable_if_t<std::tuple_size<std::tuple<Args...>>::value != 0>
81assertOutInfo(
82 bool fail, std::string const cond, std::string const& str,
83 std::string const& file, int const line, std::string const& func,
84 ErrorCodeType error, std::tuple<Args2...>&& t1, std::tuple<Args...>&& t2
85) {
86 using KeyType = std::tuple<Args2...>;
87 using ValueType = std::tuple<Args...>;
88 static constexpr auto size = std::tuple_size<KeyType>::value;
89 using PrinterType = util::error::PrinterNameValue<size-1,KeyType,ValueType>;
90
91 // Output the standard assert message
92 assertOut(false,cond,str,file,line,func,error,std::make_tuple());
93
94 // Output each expression computed passed to the function along with the
95 // computed value of that passed expression
96 auto varlist = PrinterType::make(t1,t2);
97
98 auto node = debug::preNode();
99 auto vt_pre = debug::vtPre();
100 auto bred = debug::bred();
101 auto node_str = debug::proc(node);
102 auto prefix = vt_pre + node_str + " ";
103 auto byellow = debug::byellow();
104 auto yellow = debug::yellow();
105 auto reset = debug::reset();
106 auto green = debug::green();
107 auto separator = fmt::format("{}{}{:-^120}{}\n", prefix, yellow, "", reset);
108 auto space = fmt::format("{}\n", prefix);
109 auto title = fmt::format(
110 "{}{}{:-^120}{}\n", prefix, yellow, " Debug State Assert Info ", reset
111 );
112 auto pre = fmt::format("{}{}{}{}", separator, title, separator, space);
113
114 std::ostringstream str_buf;
115
116 auto buf = fmt::format("{}", pre);
117 str_buf << buf;
118
119 //str_buf << separator << separator;
120 for (auto&& var : varlist) {
121 auto cur = fmt::format("{}{}{:>25}{}\n", prefix, green, var, reset);
122 str_buf << cur;
123 }
124 str_buf << space << separator << separator;
125 ::vt::output(str_buf.str(),error,false,false,true,fail);
126
127 if (fail) {
128 vtAbort("Assertion failed");
129 }
130}
