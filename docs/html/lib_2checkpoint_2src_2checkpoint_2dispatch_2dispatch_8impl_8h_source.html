73template <typename T, typename TraverserT>
74TraverserT& withTypeIdx(TraverserT& t) {
75 using CleanT = typename CleanType<typeregistry::DecodedIndex>::CleanT;
76 using DispatchType =
77 typename TraverserT::template DispatcherType<TraverserT, CleanT>;
78
79 auto const thisTypeIdx = typeregistry::getTypeIdx<T>();
80 SerializerDispatch<TraverserT, CleanT, DispatchType> ap;
81 constexpr SerialSizeType len = 1;
82 if (t.isPacking() || t.isSizing()) {
83 auto val = cleanType(&thisTypeIdx);
84 ap(t, val, len);
85 } else if (t.isUnpacking()) {
86 typeregistry::DecodedIndex serTypeIdx = 0;
87 auto val = cleanType(&serTypeIdx);
88 ap(t, val, len);
89
90 if (
91 typeregistry::validateIndex(serTypeIdx) == false ||
92 thisTypeIdx != serTypeIdx
93 ) {
94 auto const err = std::string("Unpacking wrong type, got=") +
95 typeregistry::getTypeNameForIdx(thisTypeIdx) +
96 " (idx=" + std::to_string(thisTypeIdx) +
97 "), expected=" + typeregistry::getTypeNameForIdx(serTypeIdx) +
98 " (idx=" + std::to_string(serTypeIdx) + ")\n#0 " +
99 typeregistry::getTypeName<T>();
100 throw serialization_error(err);
101 }
102 }
103
104 return t;
105}

107template <typename T, typename TraverserT>
108TraverserT& withMemUsed(TraverserT& t, SerialSizeType len) {
109 using DispatchType =
110 typename TraverserT::template DispatcherType<TraverserT, SerialSizeType>;
111 SerializerDispatch<TraverserT, SerialSizeType, DispatchType> ap;
112
113 SerialSizeType const memUsed = sizeof(T) * len;
114 constexpr SerialSizeType memUsedLen = 1;
115 if (t.isPacking() || t.isSizing()) {
116 auto val = cleanType(&memUsed);
117 ap(t, val, memUsedLen);
118 } else if (t.isUnpacking()) {
119 SerialSizeType serMemUsed = 0;
120 auto val = cleanType(&serMemUsed);
121 ap(t, val, memUsedLen);
122
123 if (memUsed != serMemUsed) {
124 using CleanT = typename CleanType<T>::CleanT;
125 std::string msg = "For type '" + typeregistry::getTypeName<CleanT>() +
126 "' serialization used " + std::to_string(serMemUsed) +
127 "B, but deserialization used " + std::to_string(memUsed) + "B";
128
129 throw serialization_error(msg);
130 }
131 }
132
133 return t;
134}

136template <typename T, typename TraverserT>
137TraverserT& Traverse::with(T& target, TraverserT& t, SerialSizeType len) {
138 using CleanT = typename CleanType<T>::CleanT;
139 using DispatchType =
140 typename TraverserT::template DispatcherType<TraverserT, CleanT>;
141
142 #if defined(SERIALIZATION_ERROR_CHECKING)
143 withTypeIdx<CleanT>(t);
144 #endif
145
146 auto val = cleanType(&target);
147 SerializerDispatch<TraverserT, CleanT, DispatchType> ap;
148
149 #if defined(SERIALIZATION_ERROR_CHECKING)
150 try {
151 ap(t, val, len);
152 } catch (serialization_error const& err) {
153 auto const depth = err.depth_ + 1;
154 auto const what = std::string(err.what()) + "\n#" + std::to_string(depth) +
155 " " + typeregistry::getTypeName<T>();
156 throw serialization_error(what, depth);
157 }
158 #else
159 ap(t, val, len);
160 #endif
161
162 #if defined(SERIALIZATION_ERROR_CHECKING)
163 withMemUsed<CleanT>(t, len);
164 #endif
165
166 return t;
167}

169template <typename T, typename TraverserT, typename Traits, typename... Args>
170TraverserT Traverse::with(T& target, Args&&... args) {
171 #if !defined(SERIALIZATION_ERROR_CHECKING)
172 using CleanT = typename CleanType<T>::CleanT;
173 #endif
174
175 TraverserT t_base(std::forward<Args>(args)...);
176 auto t = SerializerRef(&t_base, Traits{});
177
178 #if !defined(SERIALIZATION_ERROR_CHECKING)
179 withTypeIdx<CleanT>(t);
180 #endif
181
182 with(target, t);
183
184 #if !defined(SERIALIZATION_ERROR_CHECKING)
185 withMemUsed<CleanT>(t, 1);
186 #endif
187
188 return t_base;
189}

237template <typename TargetT, typename PackerT>
238inline void
239validatePackerBufferSize(PackerT const& p, SerialSizeType bufferSize) {
240 auto const usedBufferSize = p.usedBufferSize();
241 if (usedBufferSize != bufferSize) {
242 using CleanT = typename CleanType<TargetT>::CleanT;
243
244 std::string msg = "For type '" + typeregistry::getTypeName<CleanT>() +
245 "' Sizer reported " + std::to_string(bufferSize) + "B, but Packer used " +
246 std::to_string(usedBufferSize) + "B";
247 throw serialization_error(msg);
248 }

251template <typename T, typename UserTraits>
252buffer::ImplReturnType
253packBuffer(T& target, SerialSizeType size, BufferObtainFnType fn) {
254 SerialByteType* user_buf = fn ? fn(size) : nullptr;
255 if (user_buf == nullptr) {
256 auto p =
257 Standard::pack<T, PackerBuffer<buffer::ManagedBuffer>, UserTraits>(target, size);
258 validatePackerBufferSize<T>(p, size);
259 return std::make_tuple(std::move(p.extractPackedBuffer()), size);
260 } else {
261 auto p = Standard::pack<T, PackerBuffer<buffer::UserBuffer>, UserTraits>(
262 target, size, std::make_unique<buffer::UserBuffer>(user_buf, size)
263 );
264 validatePackerBufferSize<T>(p, size);
265 return std::make_tuple(std::move(p.extractPackedBuffer()), size);
266 }
267}

276template <typename T, typename UserTraits>
277T* deserializeType(SerialByteType* data, SerialByteType* allocBuf) {
281
282 auto mem = allocBuf ? allocBuf : Standard::allocate<T>();
283
284 // Transfer ownership of T* to the unique_ptr, which is now responsible for
285 // destructing it
286 auto t_buf = std::unique_ptr<T>(Standard::construct<T>(mem).transferOwnership());
287 T* traverser =
288 Standard::unpack<T, UnpackerBuffer<buffer::UserBuffer>, UserTraits>(t_buf.get(), data);
289 t_buf.release();
290 return traverser;
291}
