75// Covering case work-about for NVCC (CUDA 10.1). See usage site.
76template <class T>
77struct has_any_serialize_member_t
78{
79 template <typename U>
80 static auto test(U* u) -> decltype(
81 u->template serialize<SerializeSizerType>(std::declval<SerializeSizerType&>())
82 , char(0)) { return char(0); }
83 static double test(...) { return 0; }
84 static const bool value = (sizeof(test((T*)0)) == 1);
85};

87template <typename U, typename = void>
88struct has_own_serialize_member_t : std::false_type {};
89
90template <typename U>
91struct has_own_serialize_member_t<U,
92 std::enable_if_t<
93 std::is_same<
94 void (U::*)(::checkpoint::Sizer&),
95 decltype(&U::template serialize<SerializeSizerType>)
96 >::value
97 >>
98 : std::true_type
99{};
100
101// Either declared out-of-line (no inheritance in play)
102// or a serialize member is declared on the precise type.
103template <typename T>
104static constexpr auto const has_own_serialize =
105 ::checkpoint::SerializableTraits<T>::has_serialize_noninstrusive
106 // NVCC (CUDA 10.1 and 11) has SFINAE issues with decltype above,
107 // and possible fallout with lazy evaluation of 'and'.
108 // While this compiles, NVCC incorrectly detects some types
109 // as requiring serialization. See #949.
110 or cxx14_conjunction<
111 has_any_serialize_member_t<T>,
112 has_own_serialize_member_t<T>
113 >::value;

242// Trivial test to ensure that the type is eligible to be byte-copied,
243// regardless of if the message supports explicit serialization.
244// N.B. This will return TRUE on certain compilers even though the guarantee
245// cannot be validated due to implementation quirks of trivially copyable.
246#ifndef vt_quirked_trivially_copyable_on_msg
247template <typename T>
248struct is_byte_copyable_t {
249 constexpr static bool value = std::is_trivially_copyable<T>::value and not std::is_pointer<T>::value;
250};
251#else
252template <typename T>
253struct is_byte_copyable_t {
254 // Although this can detect some cases it is not strictly true.
255 constexpr static bool value = std::is_trivially_destructible<T>::value and not std::is_pointer<T>::value;
256};
