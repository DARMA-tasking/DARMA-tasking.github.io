184 template <typename SerializerT>
185 void serialize(SerializerT& s) {
186 envelopeSetHasBeenSerialized(env, true);
187 // There is some (pipe) code that currently requires the envelope to
188 // be serialized for correct operation; this may be fixed, in which
189 // case it might be possible to omit envelope serialization here.
190 s | env;
