77template <typename MsgT, typename TupleT>
78template <typename MsgU, typename CallbackT>
79typename RemoteContainer<MsgT,TupleT>::template IsVoidType<MsgU>
80RemoteContainer<MsgT,TupleT>::triggerDirect(CallbackT cb, MsgU*) {
81 auto const& pid = BaseContainer<MsgT>::getPipe();
82 constexpr auto multi_callback = std::tuple_size<decltype(trigger_list_)>();
83 vt_debug_print(
84 terse, pipe,
85 "RemoteContainer: (void) invoke trigger: pipe={:x}, multi={}\n",
86 pid, multi_callback
87 );
88 cb.trigger(nullptr,pid);
89}

91template <typename MsgT, typename TupleT>
92template <typename MsgU, typename CallbackT>
93typename RemoteContainer<MsgT,TupleT>::template IsNotVoidType<MsgU>
94RemoteContainer<MsgT,TupleT>::triggerDirect(CallbackT cb, MsgU* data) {
95 auto const& pid = BaseContainer<MsgT>::getPipe();
96 auto const& multi_callback = std::tuple_size<decltype(trigger_list_)>() > 0;
97 vt_debug_print(
98 terse, pipe,
99 "RemoteContainer: (typed) invoke trigger: pipe={:x}, multi={}, ptr={}\n",
100 pid, multi_callback, print_ptr(data)
101 );
102
103 if (multi_callback) {
104 /*
105 * Make a copy of the message for each callback trigger because the callback
106 * trigger may send/bcast messages, thus a copy must be made
107 */
108 auto cur_msg = makeMessage<MsgT>(*data);
109 cb.trigger(cur_msg.get(),pid);
110 } else {
111 cb.trigger(data,pid);
112 }
113}

115template <typename MsgT, typename TupleT>
116template <typename MsgU>
117void RemoteContainer<MsgT,TupleT>::trigger(MsgU* data) {
118 auto const& pipe = BaseContainer<MsgT>::getPipe();
119 bool const& is_send_back = isSendBack();
120 vt_debug_print(
121 terse, pipe,
122 "RemoteContainer: pipe={:x}, send_back={}, size={}\n",
123 pipe, is_send_back, std::tuple_size<decltype(trigger_list_)>()
124 );
125 if (is_send_back) {
126 auto const& retpipe = this->getPipe();
127 triggerSendBack<MsgT>(retpipe,data);
128 } else {
129 triggerDirect(data);
130 }
131}
