112template <typename BufferT>
113void PackerBuffer<BufferT>::contiguousBytes(
114 void* ptr, SerialSizeType size, SerialSizeType num_elms
115) {
116 debug_checkpoint(
117 "PackerBuffer: offset=%ld, size=%ld, num_elms=%ld, ptr=%p, cur_=%p\n",
118 cur_ - start_, size_, num_elms, ptr, static_cast<void*>(cur_)
119 );
120
121 SerialSizeType const len = size * num_elms;
122 SerialByteType* spot = this->getSpotIncrement(len);
123 #pragma GCC diagnostic push
124#if !defined(__has_warning)
125 #pragma GCC diagnostic ignored "-Wmaybe-uninitialized"
126#elif __has_warning("-Wmaybe-uninitialized")
127 #pragma GCC diagnostic ignored "-Wmaybe-uninitialized"
128#endif
129 std::memcpy(spot, ptr, len);
130 #pragma GCC diagnostic pop
131
132 usedSize_ += len;
133}
