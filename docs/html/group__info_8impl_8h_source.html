67template <typename MsgT>
68/*static*/ void Info::groupSetupHandler(MsgT* msg) {
69 using RangeType = typename MsgT::RangeType;
70
71 auto const& this_node = theContext()->getNode();
72 auto const& group_size = msg->getCount();
73 auto const& group_total_size = msg->getTotalCount();
74 auto const& group_root = msg->getRoot();
75 auto const& is_static = msg->isStatic();
76 auto group = msg->getGroup();
77 auto op_id = msg->getOpID();
78 auto parent = msg->getParent();
79 auto const& range = msg->getRange();
80
81 vtAssert(
82 this_node == group_root,
83 "This msg should only be sent to the root of the new local group"
84 );
85
86 auto new_size = group_size - 1;
87
88 vt_debug_print(
89 terse, group,
90 "Info::groupSetupHandler: group size={}, group_total_size={}, "
91 "new_size={}, min_spanning_tree_size={}\n",
92 group_size, group_total_size, new_size, min_spanning_tree_size
93 );
94
95 if (new_size < min_spanning_tree_size) {
96 vt_debug_print(
97 normal, group,
98 "Info::groupSetupHandler: create leaf node: size={}\n",
99 new_size
100 );
101
102 auto region = range.tail();
103 auto owning_region = region->copy();
104
105 theGroup()->initializeRemoteGroup(
106 group, std::move(owning_region), is_static, group_total_size
107 );
108
109 vt_debug_print(
110 verbose, group,
111 "Info::groupSetupHandler: op={}, parent={}\n",
112 op_id, parent
113 );
114
115 if (op_id != no_op_id) {
116 // Send back message
117 auto retmsg = makeMessage<GroupOnlyMsg>(group, op_id);
118 theMsg()->sendMsg<Info::groupTriggerHandler>(
119 parent, retmsg
120 );
121 }
122 } else {
123 vt_debug_print(
124 normal, group,
125 "Info::groupSetupHandler: split and send to children: size={}\n",
126 new_size
127 );
128
129 auto const& range_tail = range.tail();
130
131 std::vector<RegionType::BoundType> local_nodes;
132
133 auto parent_cont = [parent,group,op_id,this_node]{
134 auto iter = theGroup()->remote_group_info_.find(group);
135 vtAssertExpr(iter != theGroup()->remote_group_info_.end());
136 auto info = iter->second.get();
137
138 vt_debug_print(
139 verbose, group,
140 "Info::parent continuation: wait_count_={}, parent={}\n",
141 info->wait_count_, parent
142 );
143
144 info->wait_count_--;
145 if (info->wait_count_ == 0 && parent != uninitialized_destination) {
146 vt_debug_print(
147 verbose, group,
148 "Info::parent continuation: sending to parent={}, op_id={}\n",
149 parent, op_id
150 );
151
152 auto contmsg = makeMessage<GroupOnlyMsg>(group, op_id);
153 if (parent != this_node) {
154 theMsg()->sendMsg<Info::groupTriggerHandler>(
155 parent, contmsg
156 );
157 } else {
158 Info::groupTriggerHandler(contmsg.get());
159 }
160 }
161 };
162
163 range_tail->splitN(default_num_children, [&](RegionPtrType region){
164 auto const& c = region->head();
165 auto const& c_size = static_cast<NodeType>(region->getSize());
166
167 local_nodes.push_back(c);
168
169 vt_debug_print(
170 verbose, group,
171 "Info::groupSetupHandler: child c={}, c_size={}\n", c, c_size
172 );
173
174 auto op1 = theGroup()->registerContinuation(parent_cont);
175 auto l1 = static_cast<RangeType*>(region.get());
176 auto c_msg = makeMessage<MsgT>(
177 c, c_size, group, op1, group_total_size, this_node, l1
178 );
179 if (c != this_node) {
180 theMsg()->sendMsg<MsgT, groupSetupHandler>(c, c_msg);
181 } else {
182 groupSetupHandler(c_msg.get());
183 }
184 });
185
186 auto const& num_children = local_nodes.size();
187
188 // Create parent region
189 auto region = std::make_unique<region::List>(
190 local_nodes.data(), num_children, true
191 );
192
193 theGroup()->initializeRemoteGroup(
194 group, std::move(region), is_static, group_size
195 );
196 // end parent region
197
198 auto iter = theGroup()->remote_group_info_.find(group);
199 vtAssertExpr(iter != theGroup()->remote_group_info_.end());
200 auto info = iter->second.get();
201 info->wait_count_ += num_children;
202 }
203}
204
205}} /* end namespace vt::group */
