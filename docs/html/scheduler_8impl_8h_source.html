53template <typename Callable>
54void runInEpoch([[maybe_unused]] std::string const& label, EpochType ep, Callable&& fn) {
55 theSched()->triggerEvent(sched::SchedulerEvent::PendingSchedulerLoop);
56
57 theMsg()->pushEpoch(ep);
58 {
59# if vt_check_enabled(trace_enabled)
60 trace::TraceScopedEventHash event_{label};
61# endif
62 fn();
63 }
64 theMsg()->popEpoch(ep);
65 theTerm()->finishedEpoch(ep);
66 runSchedulerThrough(ep);
67}

100 work_queue_.emplace(UnitType(is_term, r));

103template <typename MsgT, typename RunT>
104void Scheduler::enqueue(MsgT* msg, RunT r) {
105 bool const is_term = envelopeIsTerm(msg->env);
106
107 if (is_term) {
108 num_term_msgs_++;
109 }
110
111# if vt_check_enabled(priorities)
112 auto priority = envelopeGetPriority(msg->env);
113 work_queue_.emplace(UnitType(is_term, std::move(r), priority));
114# else
115 work_queue_.emplace(UnitType(is_term, std::move(r)));
116# endif
117}

119template <typename MsgT, typename RunT>
120void Scheduler::enqueue(MsgSharedPtr<MsgT> const& msg, RunT r) {
121# if vt_check_enabled(priorities)
122 //
123 // Assume that MsgSharedPtr<MsgT> is already captured in the action.
124 //
125 // To speed this up, in the future, we could have a pure message queue that
126 // could be dispatched directly based on type/state-bits
127 //
128 enqueue<MsgT>(msg.get(), std::move(r));
129#else
130 bool const is_term = envelopeIsTerm(msg->env);
131 enqueue<RunT>(is_term, r);
132#endif
133}
