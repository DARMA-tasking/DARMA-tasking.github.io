51template <typename ColT>
52EpochType ConstructParams<ColT>::deferWithEpoch(ProxyFnType cb) {
53 // validate all the inputs for the configuration
54 validateInputs();
55 // Set constructed to true
56 constructed_ = true;
57
58 auto tup = theCollection()->makeCollection<ColT>(*this);
59 auto epoch = std::get<0>(tup);
60 auto proxy = std::get<1>(tup);
61
62 theTerm()->addAction(epoch, [=]{ cb(CollectionProxy<ColT>{proxy}); });
63
64 return epoch;
65}

72template <typename ColT>
73typename ConstructParams<ColT>::ProxyType ConstructParams<ColT>::wait() {
74 // wait for the collection to be constructed
75 ProxyType out_proxy;
76 bool action_run = false;
77 auto ep = deferWithEpoch([&](ProxyType proxy) {
78 out_proxy = proxy;
79 action_run = true;
80 });
81 runSchedulerThrough(ep);
82 theSched()->runSchedulerWhile([&]{ return not action_run; });
83 return out_proxy;
84}
