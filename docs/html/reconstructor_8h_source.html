69template <typename T>
70struct InPlaceWrapper {
71 // Constructor: Takes ownership of an already constructed object
72 explicit InPlaceWrapper(T* inPlaceObject) : object(inPlaceObject) {}
73
74 // Destructor: Calls the destructor of T
75 ~InPlaceWrapper() {
76 if (object) {
77 object->~T(); // Explicitly call the destructor
78 }
79 }
80
81 // Delete copy constructor and copy assignment to avoid accidental copying
82 InPlaceWrapper(const InPlaceWrapper&) = delete;
83 InPlaceWrapper& operator=(const InPlaceWrapper&) = delete;
84
85 // Allow move semantics
86 InPlaceWrapper(InPlaceWrapper&& other) noexcept : object(nullptr) {
87 std::swap(object, other.object);
88 }
89
90 InPlaceWrapper& operator=(InPlaceWrapper&& other) noexcept {
91 if (this != &other) {
92 if (object) {
93 object->~T();
94 }
95 object = nullptr;
96 std::swap(object, other.object);
97 }
98 return *this;
99 }
100
101 // Access the wrapped object
102 T* operator->() {
103 return object;
104 }
105
106 const T* operator->() const {
107 return object;
108 }
109
110 T& operator*() {
111 return *object;
112 }
113
114 const T& operator*() const {
115 return *object;
116 }
117
122 T* transferOwnership() {
123 auto ptr = object;
124 object = nullptr;
125 return ptr;
126 }
127
128private:
129 T* object;
130};

134 // Default-construct as lowest priority in reconstruction preference
135 template <typename U = T>
136 static T* constructDefault(void* buf, isDefaultConsType<U>* = nullptr) {
137 debug_checkpoint(
138 "DeserializerDispatch: default constructor: buf=%p\n", buf
139 );
140 T* t_ptr = new (buf) T{};
141 return t_ptr;
142 }
143
144 // Fail, no valid option to constructing T
145 template <typename U = T>
146 static T* constructDefault(void*, isNotDefaultConsType<U>* = nullptr) {
147 static_assert(
148 SerializableTraits<U, void>::is_tagged_constructible or
149 SerializableTraits<U, void>::is_reconstructible or
150 std::is_default_constructible<U>::value,
151 "Either a default constructor, reconstruct() function, or tagged "
152 "constructor are required for de-serialization"
153 );
154 return nullptr;
155 }
156
157 /*
158 * Try to reconstruct with the following precedence:
159 *
160 * - Tagged constructor: T(SERIALIZE_CONSTRUCT_TAG{})
161 * - Reconstruct: T::reconstruct(buf) or reconstruct(t, buf)
162 * - Default constructor: T()
163 *
164 * If none of these options work, static assert failure
165 */
166
167 // Intrusive reconstruct
168 template <typename U = T>
169 static T* constructReconstruct(void* buf, isReconstructibleType<U>* = nullptr) {
170 debug_checkpoint("DeserializerDispatch: T::reconstruct(): buf=%p\n", buf);
171 auto& t = T::reconstruct(buf);
172 return &t;
173 }
174
175 template <typename U = T>
176 static T* constructReconstruct(void* buf, isSpecializedReconstructibleType<U>* = nullptr) {
177 debug_checkpoint(
178 "DeserializerDispatch: CheckpointReconstructor<T>::reconstruct(): buf=%p\n",
179 buf
180 );
181 T* t = nullptr;
182 checkpoint::CheckpointReconstructor<T>::reconstruct(t, buf);
183 return t;
184 }
185
186 // Non-intrusive reconstruct
187 template <typename U = T>

196 }

232 constexpr int max_buffer_length = 32768;
233 std::unique_ptr<char[]> msg = std::make_unique<char[]>(max_buffer_length);
234 snprintf(
235 &msg[0],
236 max_buffer_length,
237 "Checkpoint is failing to reconstruct a class %s, due to it being "
238 "abstract or the absence of a suitable constructor (default or tagged) "
239 "or reconstruct()",

241 );
242 checkpointAssert(false, msg.get());
243 return InPlaceWrapper<T>{nullptr};
244 }
