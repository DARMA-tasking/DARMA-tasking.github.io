59template <typename VrtContextT>
60inline void vrLaunchMainContext() {
61 if (theContext()->getNode() == main_node) {
62 vt_debug_print(verbose, gen, "vrLaunchMainContext: launching main context\n");

67inline void vtMainScheduler() {
68 vt_debug_print(verbose, gen, "vtMainScheduler: running main scheduler\n");
69
70 vt::theSched()->runSchedulerWhile([]{ return !rt->isTerminated();});
71}

73template <typename VrtContextT>
74inline void vrCommThreadWork() {
75 vrLaunchMainContext<VrtContextT>();
76 vtMainScheduler();
77}

79template <typename VrtContextT>
80int vt_main(int argc, char** argv) {
81 auto rt = CollectiveOps::initialize(argc, argv);
82 vt_debug_print(verbose, gen, "vt_main: initialized\n");
83
84 auto comm_fn = vrCommThreadWork<VrtContextT>;
85 comm_fn();
86
87 vt_debug_print(verbose, gen, "vt_main: auto finalize\n");
88 return 0;
