80template <typename MsgT>
81template <typename T>
82typename CallbackSend<MsgT>::template IsVoidType<T>
83CallbackSend<MsgT>::triggerDispatch(
84 [[maybe_unused]] SignalDataType* data, PipeType const& pid
85) {
86 auto const& this_node = theContext()->getNode();
87 vt_debug_print(
88 terse, pipe,
89 "CallbackSend: (void) trigger_: this_node={}, send_node_={}\n",
90 this_node, send_node_
91 );
92 if (this_node == send_node_) {
93 runnable::makeRunnableVoid(true, handler_, this_node)
94 .withTDEpoch(theMsg()->getEpoch())
95 .enqueue();
96 } else {
97 auto msg = makeMessage<CallbackMsg>(pid);
98 theMsg()->sendMsg<CallbackMsg>(send_node_, handler_, msg);
99 }
100}

102template <typename MsgT>
103template <typename T>
104typename CallbackSend<MsgT>::template IsNotVoidType<T>
105CallbackSend<MsgT>::triggerDispatch(
106 SignalDataType* data, [[maybe_unused]] PipeType const& pid
107) {
108 auto const& this_node = theContext()->getNode();
109 vt_debug_print(
110 terse, pipe,
111 "CallbackSend: trigger_: this_node={}, send_node_={}\n",
112 this_node, send_node_
113 );
114 if (this_node == send_node_) {
115 auto msg = reinterpret_cast<ShortMessage*>(data);
116 auto m = promoteMsg(msg);
117 runnable::makeRunnable(m, true, handler_, this_node)
118 .withTDEpochFromMsg()
119 .enqueue();
120 } else {
121 theMsg()->sendMsg<SignalDataType>(send_node_, handler_, data);
122 }
123}
