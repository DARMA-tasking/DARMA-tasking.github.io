56/*
57 * Discrete interval tree for encoding a set of partially tangent,
58 * non-overlapping integral values into interval ranges seamlessly in a
59 * tree. Maintains a "DIET" data structure: Discrete Interval Encoding
60 * Tree. When inserting contiguous elements in order, the storage (non-bit) and
61 * time is constant O(1).
62 */
63
64template <
65 typename DomainT,
66 DomainT sentinel,
67 template <class> class AllocatorT,
68 template <class U, U> class IntervalT,
69 template <class, class, class> class OrderedSetT,
70 template <class> class IntervalCompareT
71>
72struct IntegralSetBase {
73 using DomainType = DomainT;
74 using IntervalType = IntervalT<DomainT, sentinel>;
75 using CompareType = IntervalCompareT<DomainT>;
76 using AllocType = AllocatorT<IntervalType>;
77 using OrderedSetType = OrderedSetT<IntervalType, CompareType, AllocType>;
78 using IteratorType = typename OrderedSetType::iterator;
79 using PositionType = typename IntervalType::PositionEnum;
80
81 IntegralSetBase() : hint_(set_.end()) { }
82 IntegralSetBase(IntegralSetBase const&) = default;
83 IntegralSetBase(IntegralSetBase&&) = default;
84 IntegralSetBase& operator=(IntegralSetBase const&) = default;

128 template <
129 typename IntervalU,
130 typename = std::enable_if_t<
131 std::is_same<
132 std::remove_const_t<std::remove_reference_t<IntervalU>>, IntervalType
133 >::value
134 >
135 >
136 IteratorType insertInterval(IntervalU&& i) {
137 vt_debug_print(
138 normal, gen,
139 "OrderedSet: insertInterval: bounds=[{:x},{:x}] elms_={}, i={}, is_end={}\n",
140 lb_, ub_, elms_, i, hint_ == set_.end()
141 );
142
143 auto iter = elms_ == 0 ? set_.end() : hint_;
144
145 // Expand the global bounds in this interval set
146 insertGlobal(i);
147
148 // Insert interval into the compressed tree
149 return insertSet(iter,std::move(i));

152 template <
153 typename IntervalU,
154 typename = std::enable_if_t<
155 std::is_same<
156 std::remove_const_t<std::remove_reference_t<IntervalU>>, IntervalType
157 >::value
158 >
159 >
160 IteratorType insertInterval(IteratorType it, IntervalU&& i) {
161 // Expand the global bounds in this interval set
162 insertGlobal(i);
163
164 // Must be valid iterator to call this overload
165 vtAssert(it != set_.end(), "Must be valid iterator");
166
167 // If the val is tangent to the hint iterator, directly update
168 auto pos = it->tangent(i);
169 if (pos not_eq PositionType::NotTangent) {
170 it->join(i, pos);
171 return it;
172 }
173
174 // Insert interval into the compressed tree
175 return insertSet(it,std::move(i));
176 }

178 void erase(DomainT const& val) {
179 vtAssert(existsGlobal(val), "This element must exist in the set");
180 IntervalType j(val);
181 auto iter = set_.find(j);
182 bool in_set = iter != set_.end();
183 vt_debug_print(
184 normal, gen,
185 "OrderedSet: erase: bounds=[{:x},{:x}] size={}, comp={}, val={}, in={}\n",
186 lb_, ub_, size(), compressedSize(), j, in_set
187 );
188 vtAssert(in_set, "The element must exist in a interval bucket");
189 if (in_set) {
190 if (iter->width() == 1) {
191 bool invalid_hint = iter == hint_;
192 auto ret = set_.erase(iter);
193 if (invalid_hint) {
194 hint_ = ret;
195 }
196 } else if (iter->upper() == val) {
197 auto& to_split = const_cast<IntervalType&>(*iter);
198 to_split.setUpper(iter->upper() - 1);
199 } else if (iter->lower() == val) {
200 auto& to_split = const_cast<IntervalType&>(*iter);
201 to_split.setLower(iter->lower() + 1);
202 } else {
203 // Slice the interval into two interval nodes
204 vtAssert(iter->width() > 2, "Interval width must be greater than 2");
205 auto ub = iter->upper();
206 auto& to_split = const_cast<IntervalType&>(*iter);
207 to_split.setUpper(val - 1);
208 IntervalType i(val+1, ub);
209 insertSet(iter,std::move(i));
210 }
211 eraseGlobal(val);
212 }
213 }

270 IteratorType insertSet(IteratorType it, IntervalType&& i) {
271 vt_debug_print(
272 normal, gen,
273 "OrderedSet: insertSet: bounds=[{:x},{:x}] elms_={}, i={}, is_end={}\n",
274 lb_, ub_, elms_, i, it == set_.end()
275 );
276
277 // Insert into the set
278 auto ret = set_.insert(it,std::move(i));
279 vtAssert(ret not_eq it, "Should be a valid insert");
280 vtAssert(ret not_eq set_.end(), "Must be valid insert---live iterator");
281
282 // Fuse interval set elements that are now tangent after this insertion
283 return hint_ = join(ret);
284 }

286 bool existsGlobal(DomainT const& val) const {
287 return val >= lb_ and val <= ub_;
288 }
289
290 void eraseGlobal(DomainT const& val) {
291 bool is_lower = lb_ == val;
292 bool is_upper = ub_ == val;
293 if (is_lower) {
294 if (set_.size() > 0) {
295 lb_ = set_.begin()->lower();
296 } else {
297 lb_ = 0;
298 }
299 }
300 if (is_upper) {
301 if (set_.size() > 0) {
302 ub_ = std::prev(set_.end())->upper();
303 } else {
304 ub_ = 0;
305 }
306 }
307 // Decrement count of non-compressed elements
308 elms_--;
309 }
310
311 void insertGlobal(IntervalType const& i) {
312 vt_debug_print(
313 verbose, gen,
314 "OrderedSet: insertGlobal: bounds=[{:x},{:x}] elms_={}, i={}\n",
315 lb_, ub_, elms_, i
316 );
317
318 if (elms_ == 0) {
319 lb_ = i.lower();
320 ub_ = i.upper();
321 } else {
322 // Expand the global bounds in this interval set
323 lb_ = std::min<DomainT>(lb_, i.lower());
324 ub_ = std::max<DomainT>(ub_, i.upper());
325 }
326 // Increment count of non-compressed elements
327 elms_ += i.width();
328 }

330 IteratorType join(IteratorType it) {
331 vt_debug_print(
332 verbose, gen,
333 "OrderedSet: join: bounds=[{:x},{:x}] elms_={}\n",
334 lb_, ub_, elms_
335 );
336
337 auto it2 = joinLeft(it);
338 return joinRight(it2);
339 }
340
341 IteratorType joinLeft(IteratorType it) {
342 vtAssertExpr(it not_eq set_.end());
343 if (it not_eq set_.begin()) {
344 auto prev = std::prev(it);
345 if (it->tangent(*prev) == PositionType::TangentLeft) {
346 auto tmp = *prev;
347 set_.erase(prev);
348 auto& to_fuse = const_cast<IntervalType&>(*it);
349 to_fuse.join(tmp, PositionType::TangentLeft);
350 }
351 }
352 return it;
353 }
354
355 IteratorType joinRight(IteratorType it) {
356 vtAssertExpr(it not_eq set_.end());
357 auto next = std::next(it);
358 if (next not_eq set_.end()) {
359 if (it->tangent(*next) == PositionType::TangentRight) {
360 auto tmp = *next;
361 set_.erase(next);
362 auto& to_fuse = const_cast<IntervalType&>(*it);
363 to_fuse.join(tmp, PositionType::TangentRight);
364 }
365 }
366 return it;
367 }

380 template <typename Impl>
381 struct IntervalSetIter {
382 using Iter = IntervalSetIter;
383 using iterator_category = std::bidirectional_iterator_tag;
384 using difference_type = DomainT;
385 using value_type = DomainT;
386 using pointer = DomainT;
387 using reference = DomainT;
388
389 IntervalSetIter(Impl in_impl, std::size_t in_cur)
390 : impl_(in_impl),
391 cur_(in_cur)
392 { }
393
394 Iter& operator++() {
395 if (impl_->lower() + static_cast<DomainT>(cur_) < impl_->upper()) {
396 cur_++;
397 } else {
398 cur_ = 0;
399 impl_ = std::next(impl_);
400 }
401 return *this;
402 }
403
404 Iter& operator--() {
405 if (cur_ == 0) {
406 impl_ = std::prev(impl_);
407 cur_ = impl_->width() - 1;
408 } else {
409 cur_--;
410 }
411 return *this;
412 }
413
414 Iter operator++(int) { Iter ret = *this; ++(*this); return ret; }
415 Iter operator--(int) { Iter ret = *this; --(*this); return ret; }
416 bool operator==(Iter i) const { return i.impl_ == impl_ and i.cur_ == cur_; }
417 bool operator!=(Iter i) const { return !(*this == i); }
418 pointer operator*() const { return impl_->get(cur_); }
419 reference operator->() const { return impl_->get(cur_); }
420
421 private:
422 Impl impl_ = {};
423 std::size_t cur_ = 0;
424 };
