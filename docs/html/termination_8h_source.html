74struct EpochStack {
75 using DataType = epoch::detail::EpochImplType;
76
77 void push(DataType in) { stack_[cur_++] = in; }
78 DataType top() const { return stack_[cur_-1]; }
79 void pop() { cur_--; }
80 unsigned int size() const { return cur_; }
81
82 int cur_ = 0;
83 std::array<DataType, 64> stack_;
84};

124 virtual ~TerminationDetector() {
125 //Pop all extraneous epochs off the stack greater than 1
126 while (epoch_stack_.size() > 1) {
127 epoch_stack_.pop();
128 }
129 // Pop off the last epoch: term::any_epoch_sentinel
130 auto const ret_epoch = popEpoch(term::any_epoch_sentinel);
131 vtAssertInfo(
132 ret_epoch == term::any_epoch_sentinel, "Last pop must be any epoch",
133 ret_epoch, term::any_epoch_sentinel, epoch_stack_.size()
134 );
135 vtAssertExpr(epoch_stack_.size() == 0);
136 }
