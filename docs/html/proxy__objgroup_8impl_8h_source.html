89template <typename ObjT>
90template <auto f, typename... Params>
91typename Proxy<ObjT>::PendingSendType
92Proxy<ObjT>::broadcast(Params&&... params) const {
93 using MsgT = typename ObjFuncTraits<decltype(f)>::MsgT;
94 if constexpr (std::is_same_v<MsgT, NoMsg>) {
95 using Tuple = typename ObjFuncTraits<decltype(f)>::TupleType;
96 using SendMsgT = messaging::ParamMsg<Tuple>;
97 auto msg = vt::makeMessage<SendMsgT>();
98 msg->setParams(std::forward<Params>(params)...);
99 auto const ctrl = proxy::ObjGroupProxy::getID(proxy_);
100 auto const han = auto_registry::makeAutoHandlerObjGroupParam<
101 ObjT, decltype(f), f, SendMsgT
102 >(ctrl);
103 return theObjGroup()->broadcast(msg, han);
104 } else {
105 auto msg = makeMessage<MsgT>(std::forward<Params>(params)...);
106 return broadcastMsg<MsgT, f>(msg);
107 }
108
109 // Silence nvcc warning (no longer needed for CUDA 11.7 and up)
110 return typename Proxy<ObjT>::PendingSendType{std::nullptr_t{}};
111}

113template <typename ObjT>
114template <auto f, typename... Params>
115typename Proxy<ObjT>::PendingSendType
116Proxy<ObjT>::multicast(GroupType type, Params&&... params) const{
117 using MsgT = typename ObjFuncTraits<decltype(f)>::MsgT;
118 if constexpr (std::is_same_v<MsgT, NoMsg>) {
119 using Tuple = typename ObjFuncTraits<decltype(f)>::TupleType;
120 using SendMsgT = messaging::ParamMsg<Tuple>;
121 auto msg = vt::makeMessage<SendMsgT>();
122 msg->setParams(std::forward<Params>(params)...);
123 vt::envelopeSetGroup(msg->env, type);
124 auto const ctrl = proxy::ObjGroupProxy::getID(proxy_);
125 auto const han = auto_registry::makeAutoHandlerObjGroupParam<
126 ObjT, decltype(f), f, SendMsgT
127 >(ctrl);
128 return theObjGroup()->broadcast(msg, han);
129 } else {
130 auto msg = makeMessage<MsgT>(std::forward<Params>(params)...);
131 vt::envelopeSetGroup(msg->env, type);
132 return broadcastMsg<MsgT, f>(msg);
133 }
134
135 // Silence nvcc warning (no longer needed for CUDA 11.7 and up)
136 return typename Proxy<ObjT>::PendingSendType{std::nullptr_t{}};
137}

139template <typename ObjT>
140template <auto f, typename... Params>
141typename Proxy<ObjT>::PendingSendType Proxy<ObjT>::multicast(
142 group::region::Region::RegionUPtrType&& nodes, Params&&... params
143) const {
144 vtAssert(
145 not dynamic_cast<group::region::ShallowList*>(nodes.get()),
146 "multicast: range of nodes is not supported for ShallowList!"
147 );
148
149 nodes->sort();
150 auto& range = nodes->makeList();
151
152 auto groupID = theGroup()->GetTempGroupForRange(range);
153 if (!groupID.has_value()) {
154 return typename Proxy<ObjT>::PendingSendType{
155 theTerm()->getCurrentEpoch(),
156 [&, this,
157 args = std::make_tuple(std::forward<Params>(params)...)] {
158 std::apply(
159 [&, this](auto&&... unpackedArgs) {
160 // Explicit copy, we move 'nodes' in group creation
161 auto range_copy = nodes->makeList();
162 auto id =
163 theGroup()->newGroup(std::move(nodes), [&, this](GroupType type) {
164 multicast<f>(
165 type, std::forward<decltype(unpackedArgs)>(unpackedArgs)...);
166 });
167 theGroup()->AddNewTempGroup(range_copy, id);
168 },
169 std::move(args));
170 }};
171 } else {
172 return multicast<f>(groupID.value(), std::forward<Params>(params)...);
173 }
174
175 // Silence nvcc warning (no longer needed for CUDA 11.7 and up)
176 return typename Proxy<ObjT>::PendingSendType{std::nullptr_t{}};
177}

265 typename OpT, typename MsgPtrT, typename MsgT, ActiveTypedFnType<MsgT> *f
266>
267typename Proxy<ObjT>::PendingSendType Proxy<ObjT>::reduce(
268 MsgPtrT inmsg, Callback<MsgT> cb, ReduceStamp stamp
269) const {
270 auto proxy = Proxy<ObjT>(*this);
271 // Force promoteMsg(Msg*) to avoid special-case overload behavior
272 // which is now deprecated. The issue is caused because inmsg can be either
273 // Msg* or MsgPtr<Msg> based on call sites.
274 MsgPtr<MsgT> msg = promoteMsg(static_cast<MsgT*>(inmsg));
275 msg->setCallback(cb);
276 return theObjGroup()->reduce<ObjT, MsgT, f>(proxy,msg,stamp);
277}
