55/*
56 * Discrete interval for encoding compressed sets (DIETs). Inclusive in
57 * representation.
58 */

60template <typename DomainT, DomainT sentinel = DomainT()>
61struct Interval {
62 using IntervalType = Interval<DomainT, sentinel>;
63 using DomainType = DomainT;
64
65 Interval() = default;
66 Interval(Interval const&) = default;
67 Interval(Interval&&) = default;
68 Interval& operator=(Interval const&) = default;

70 explicit Interval(DomainT const& val) : lb_(val), ub_(val) { }
71 Interval(DomainT const& lb, DomainT const& ub) : lb_(lb), ub_(ub) { }

73public:
74 DomainT lower() const { return lb_; }
75 DomainT upper() const { return ub_; }
76 DomainT width() const { return (ub_-lb_)+1; }
77 bool valid() const { return lb_ <= ub_; }
78
79 DomainT get(std::size_t i) const {
80 auto const val = lb_ + static_cast<DomainT>(i);
81 vtAssert(lb_ not_eq sentinel, "Lower bound must be valid");
82 vtAssert(ub_ not_eq sentinel, "Upper bound must be valid");
83 vtAssert(val <= ub_, "Must be in interval range");
84 return val;
85 }

87 void setUpper(DomainT const& val) {
88 ub_ = val;
89 vtAssert(val >= lb_, "Upper must be greater than lower bound");
90 vtAssert(valid(), "Interval must remain valid");
91 }

93 void setLower(DomainT const& val) {
94 lb_ = val;
95 vtAssert(ub_ >= val, "Lower must be less than upper bound");
96 vtAssert(valid(), "Interval must remain valid");
97 }

99 void set(DomainT const& lo, DomainT const& hi) {
100 setLower(lo);
101 setUpper(hi);
102 }

104 bool contains(DomainT const& val) const {
105 return val >= lb_ and val <= ub_;

108 enum struct PositionEnum : int {
109 TangentLeft = -1,
110 NotTangent = 0,
111 TangentRight = 1
112 };
113
114 /*
115 * Determine if a input interval is tangent to this interval within a
116 * specified gap.
117 *
118 * TangentLeft TangentRight
119 * |------i-----| lb_ ub_ |------i-----|
120 * |----this----|
121 * ^ ^
122 * gap gap
123 */
124
125 PositionEnum tangent(IntervalType const& i, DomainT gap = 1) const {
126 if (lb_ == i.ub_ + gap) {
127 return PositionEnum::TangentLeft;
128 } else if (ub_ == i.lb_ - gap) {
129 return PositionEnum::TangentRight;
130 } else {
131 return PositionEnum::NotTangent;
132 }
133 }
134
135 void join(IntervalType const& i, PositionEnum pos) {
136 switch (pos) {
137 case PositionEnum::TangentLeft:
138 vtAssertExpr(lb_ == i.upper() + 1);
139 lb_ = i.lower();
140 break;
141 case PositionEnum::TangentRight:
142 vtAssertExpr(ub_ == i.lower() - 1);
143 ub_ = i.upper();
144 break;
145 default:
146 vtAssert(false, "Must have tangent position to join");
147 }
148 }

150 template <typename SerializerT>
151 void serialize(SerializerT& s) {
152 s | lb_ | ub_;
153 }
154
155 bool operator==(IntervalType const& i) const { return i.lb_ == lb_ and i.ub_ == ub_; }
156 bool operator!=(IntervalType const& i) const { return !(*this == i); }
157
158public:
159 /*
160 * Determine if two intervals intersect:
161 *
162 *
163 *. [i1.lower(), ..., i1.upper()]
164 * ^ ^
165 * | |
166 * [i2.lower(), ..., i2.upper()]
167 *
168 * Two intervals intersect iff:
169 * (i2.lower() <= i1.upper()) && (i1.lower() <= i2.upper())
170 */
171 template <typename IntT, typename IntU>
172 static bool intersects(IntT&& i1, IntU&& i2) {
173 bool ret = i1.lower() <= i2.upper() and i2.lower() <= i1.upper();
174 vt_debug_print(
175 verbose, gen,
176 "Interval intersects: i1={}, i2={}, op:{}<={}={} and {}<={}={} => {}\n",
177 i1, i2,
178 i1.lower(), i2.upper(), i1.lower() <= i2.upper(),
179 i2.lower(), i1.upper(), i2.lower() <= i1.upper(),
180 ret
181 );
182 return ret;
183 }
184
185 template <typename IntT, typename IntU>
186 static bool less(IntT&& i1, IntU&& i2) {
187 return i1.lower() < i2.lower() and i1.upper() < i2.lower();
188 }
