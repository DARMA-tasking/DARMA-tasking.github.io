120 #define vtAssertImpl(fail,cond,str) \
121 do { \
122 if (!(cond)) { \
123 ::vt::debug::assert::assertOut( \
124 fail,#cond,str,DEBUG_LOCATION,1,std::make_tuple() \
125 ); \
126 } \
127 } while (false)
128 #define vtAssertExprImpl(fail,cond) \
129 do { \
130 if (!(cond)) { \
131 ::vt::debug::assert::assertOutExpr( \
132 fail,#cond,DEBUG_LOCATION,1 \
133 ); \
134 } \
135 } while (false)
136 #define vtAssertArgImpl(fail,cond,str,...) \
137 do { \
138 if (!(cond)) { \
139 ::vt::debug::assert::assertOutInfo( \
140 fail,#cond,str,DEBUG_LOCATION,1, \
141 std::make_tuple(argsToString(__VA_ARGS__)""), \
142 std::make_tuple(__VA_ARGS__,"") \
143 ); \
144 } \
145 } while (false)
146 #define vtAssertExprArgImpl(fail,cond,...) \
147 vtAssertArgImpl(fail,cond,#cond,__VA_ARGS__)
148
149 #define vt_assert_should_fail (!vt::debug::preConfig()->vt_no_assert_fail)
150
151 #define vtAssert(cond,str) \
152 vtAssertImpl(vt_assert_should_fail,cond,str)
153 #define vtAssertInfo(cond,str,...) \
154 vtAssertArgImpl(vt_assert_should_fail,cond,str,__VA_ARGS__)
155 #define vtAssertExpr(cond) \
156 vtAssertExprImpl(vt_assert_should_fail,cond)
157 #define vtAssertExprInfo(cond,...) \
158 vtAssertExprArgImpl(vt_assert_should_fail,cond,__VA_ARGS__)
159 #define vtWarnInfo(cond,str,...) \
160 vtAssertArgImpl(false,cond,str,__VA_ARGS__)
