96template <typename>
97void CollectionManager::cleanupAll() {
98 /*
99 * Run the cleanup functions for type-specific cleanup that can not be
100 * performed without capturing the type of each collection
101 */
102 int num_cleanup = 0;
103 int tot_cleanup = static_cast<int>(cleanup_fns_.size());
104 auto iter = cleanup_fns_.begin();
105 while (iter != cleanup_fns_.end()) {
106 auto lst = std::move(iter->second);
107 iter = cleanup_fns_.erase(iter);
108 num_cleanup++;
109 for (auto fn : lst) {
110 fn();
111 }
112 }
113 vtAssertExpr(cleanup_fns_.size() == 0);
114 vtAssertExpr(num_cleanup == tot_cleanup);
115}

150template <typename ColT>
151GroupType CollectionManager::createGroupCollection(
152 VirtualProxyType const& proxy, bool const in_group
153) {
154 using IndexType = typename ColT::IndexType;
155
156 vt_debug_print(
157 terse, vrt_coll,
158 "createGroupCollection: proxy={:x}, in_group={}\n",
159 proxy, in_group
160 );
161
162 {
163 // Delete the old group that is no longer needed since we are replacing it
164 auto elm_holder = theCollection()->findElmHolder<IndexType>(proxy);
165 auto old_group_id = elm_holder->group();
166 if (old_group_id != no_group and old_group_id != default_group) {
167 auto old_group_default = theGroup()->isGroupDefault(old_group_id);
168 if (not old_group_default) {
169 theGroup()->deleteGroupCollective(old_group_id);
170 }
171 }
172 }
173
174 auto const group_id = theGroup()->newGroupCollective(
175 in_group, [proxy](GroupType new_group){
176 auto const& group_root = theGroup()->groupRoot(new_group);
177 auto const& is_group_default = theGroup()->isGroupDefault(new_group);
178 auto elm_holder = theCollection()->findElmHolder<IndexType>(proxy);
179 elm_holder->setGroup(new_group);
180 elm_holder->setUseGroup(!is_group_default);
181 elm_holder->setGroupReady(true);
182 if (!is_group_default) {
183 elm_holder->setGroupRoot(group_root);
184 }
185
186 vt_debug_print(
187 normal, vrt_coll,
188 "group finished construction: proxy={:x}, new_group={:x}, use_group={}, "
189 "ready={}, root={}, is_group_default={}\n",
190 proxy, new_group, elm_holder->useGroup(), elm_holder->groupReady(),
191 group_root, is_group_default
192 );
193 }
194 );
195
196 vt_debug_print(
197 verbose, vrt_coll,
198 "createGroupCollection (after): proxy={:x}, in_group={}, group_id={:x}\n",
199 proxy, in_group, group_id
200 );
201
202 return group_id;
203}

205template <typename ColT, typename IndexT, typename MsgT>
206/*static*/ void CollectionManager::collectionAutoMsgDeliver(
207 MsgT* msg, Indexable<IndexT>* base, HandlerType han, NodeType from,
208 [[maybe_unused]] trace::TraceEventIDType event, bool immediate
209) {
210 // Expand out the index for tracing purposes; Projections takes up to
211 // 4-dimensions
212#if vt_check_enabled(trace_enabled)
213 auto idx = base->getIndex();
214 uint64_t const idx1 = idx.ndims() > 0 ? idx[0] : 0;
215 uint64_t const idx2 = idx.ndims() > 1 ? idx[1] : 0;
216 uint64_t const idx3 = idx.ndims() > 2 ? idx[2] : 0;
217 uint64_t const idx4 = idx.ndims() > 3 ? idx[3] : 0;
218#endif
219
220 auto m = promoteMsg(msg);
221
222 runnable::makeRunnable(m, true, han, from)
223 .withTDEpoch(theMsg()->getEpochContextMsg(msg))
224 .withCollection(base)
225#if vt_check_enabled(trace_enabled)
226 .withTraceIndex(event, idx1, idx2, idx3, idx4)
227#endif
228 .withLBData(base)
229 .runOrEnqueue(immediate);
230}

232template <typename ColT, typename IndexT, typename MsgT>
233/*static*/ void CollectionManager::collectionBcastHandler(MsgT* msg) {
234 auto const col_msg = static_cast<CollectionMessage<ColT>*>(msg);
235 auto const bcast_proxy = col_msg->getBcastProxy();
236 auto const& group = envelopeGetGroup(msg->env);
237 auto const& cur_epoch = theMsg()->getEpoch();
238 auto const& msg_epoch = envelopeGetEpoch(msg->env);
239 theMsg()->pushEpoch(cur_epoch);
240 vt_debug_print(
241 terse, vrt_coll,
242 "collectionBcastHandler: bcast_proxy={:x}, han={}, "
243 "epoch={:x}, msg epoch={:x}, group={}, default group={}\n",
244 bcast_proxy, col_msg->getVrtHandler(),
245 cur_epoch, msg_epoch, group, default_group
246 );
247 auto elm_holder = theCollection()->findElmHolder<IndexT>(bcast_proxy);
248 if (elm_holder) {
249 auto const handler = col_msg->getVrtHandler();
250 vt_debug_print(
251 normal, vrt_coll,
252 "broadcast apply: size={}\n", elm_holder->numElements()
253 );
254 elm_holder->foreach([col_msg, msg, handler](
255 [[maybe_unused]] IndexT const& idx, Indexable<IndexT>* base
256 ) {
257 vtAssert(base != nullptr, "Must be valid pointer");
258
259 // be very careful here, do not touch `base' after running the active
260 // message because it might have migrated out and be invalid
261 auto const from = col_msg->getFromNode();
262 trace::TraceEventIDType trace_event = trace::no_trace_event;
263 #if vt_check_enabled(trace_enabled)
264 trace_event = col_msg->getFromTraceEvent();
265 #endif
266 collectionAutoMsgDeliver<ColT,IndexT,MsgT>(
267 msg, base, handler, from, trace_event, false
268 );
269 });
270 }
271 /*
272 * Termination: consume for default epoch for correct termination: on the
273 * other end the sender produces p units for each broadcast to the default
274 * group
275 */
276 vt_debug_print(
277 verbose, vrt_coll,
278 "collectionBcastHandler: (consume) bcast_proxy={:x}, han={}, "
279 "epoch={:x}, msg epoch={:x}, group={}, default group={}\n",
280 bcast_proxy, col_msg->getVrtHandler(),
281 cur_epoch, msg_epoch, group, default_group
282 );
283 theMsg()->popEpoch(cur_epoch);
284}

335template <typename ColT, typename MsgT>
336/*static*/ void CollectionManager::recordLBData(ColT* col_ptr, MsgT* msg) {
337 auto const pfrom = msg->getSenderElm();
338
339 if (pfrom.id == elm::no_element_id) {
340 return;
341 }
342
343 auto const pto = col_ptr->getElmID();
344 auto& lb_data = col_ptr->getLBData();
345 auto const msg_size = serialization::MsgSizer<MsgT>::get(msg);
346 auto const cat = msg->getCat();
347 vt_debug_print(
348 normal, vrt_coll,
349 "recordLBData: receive msg: elm(to={}, from={}),"
350 " no={}, size={}, category={}\n",
351 pto, pfrom, elm::no_element_id, msg_size,
352 static_cast<typename std::underlying_type<elm::CommCategory>::type>(cat)
353 );
354 if (
355 cat == elm::CommCategory::SendRecv or
356 cat == elm::CommCategory::Broadcast
357 ) {
358 bool bcast = cat == elm::CommCategory::SendRecv ? false : true;
359 lb_data.recvObjData(pto, pfrom, msg_size, bcast);
360 } else if (
361 cat == elm::CommCategory::NodeToCollection or
362 cat == elm::CommCategory::NodeToCollectionBcast
363 ) {
364 bool bcast = cat == elm::CommCategory::NodeToCollection ? false : true;
365 auto nfrom = msg->getFromNode();
366 lb_data.recvFromNode(pto, nfrom, msg_size, bcast);
367 }

571#endif

574 auto& inner_holder = elm_holder->lookup(idx);
575 auto const col_ptr = inner_holder.getRawPtr();
576 auto const from = theContext()->getNode();

595 msg.get(), col_ptr, han, from, trace_event, true

614template <typename ColT, typename IndexT, typename MsgT>
615messaging::PendingSend CollectionManager::broadcastFromRoot(MsgT* raw_msg) {
616 auto msg = promoteMsg(raw_msg);
617
618 // broadcast to all nodes
619 auto const& this_node = theContext()->getNode();
620 auto const& proxy = msg->getBcastProxy();
621 auto elm_holder = theCollection()->findElmHolder<IndexT>(proxy);
622 auto const bcast_node = VirtualProxyBuilder::getVirtualNode(proxy);
623
624 vtAssert(elm_holder != nullptr, "Must have elm holder");
625 vtAssert(this_node == bcast_node, "Must be the bcast node");
626
627 auto const cur_epoch = theMsg()->getEpochContextMsg(msg);
628 theMsg()->pushEpoch(cur_epoch);
629
630 vt_debug_print(
631 normal, vrt_coll,
632 "broadcastFromRoot: proxy={:x}, han={}\n",
633 proxy, msg->getVrtHandler()
634 );
635
636 auto const& group_ready = elm_holder->groupReady();
637 auto const& use_group = elm_holder->useGroup();
638 bool const send_group = group_ready && use_group;
639
640 vt_debug_print(
641 verbose, vrt_coll,
642 "broadcastFromRoot: proxy={:x}, han={}, group_ready={}, "
643 "group_active={}, use_group={}, send_group={}, group={:x}, cur_epoch={:x}\n",
644 proxy, msg->getVrtHandler(),
645 group_ready, send_group, use_group, send_group,
646 use_group ? elm_holder->group() : default_group, cur_epoch
647 );

896template <typename ColT, typename MsgT, ActiveTypedFnType<MsgT> *f>
897messaging::PendingSend CollectionManager::reduceMsgExpr(
898 CollectionProxyWrapType<ColT> const& proxy,
899 MsgT *const raw_msg, ReduceIdxFuncType<typename ColT::IndexType> expr_fn,
900 ReduceStamp stamp, NodeType root
901) {
902 using IndexT = typename ColT::IndexType;
903
904 vtAssert(
905 hasContext<IndexT>(), "Must have collection element context"
906 );
907 vtAssert(
908 queryProxyContext<IndexT>() == proxy.getProxy(),
909 "Must have matching proxy context"
910 );
911
912 // Get the current running index context
913 IndexT idx = *queryIndexContext<IndexT>();
914
915 auto msg = promoteMsg(raw_msg);
916
917 vt_debug_print(
918 terse, vrt_coll,
919 "reduceMsg: msg={}\n", print_ptr(raw_msg)
920 );
921
922 auto const col_proxy = proxy.getProxy();
923 auto const cur_epoch = theMsg()->getEpochContextMsg(msg);
924
925 theMsg()->pushEpoch(cur_epoch);
926 auto elm_holder = findElmHolder<IndexT>(col_proxy);
927
928 std::size_t num_elms = 0;
929 if (expr_fn == nullptr) {
930 num_elms = elm_holder->numElements();
931 } else {
932 num_elms = elm_holder->numElementsExpr(expr_fn);
933 }
934
935 auto const root_node =
936 root == uninitialized_destination ? default_collection_reduce_root_node :
937 root;
938
939 auto const group_ready = elm_holder->groupReady();
940 auto const send_group = elm_holder->useGroup();
941 auto const group = elm_holder->group();
942 bool const use_group = group_ready && send_group;
943
944 vtAssert(group_ready, "Must be ready");
945
946 auto cur_stamp = stamp;
947 auto default_stamp = ReduceStamp{};
948 if (cur_stamp == default_stamp) {
949 cur_stamp = proxy(idx).tryGetLocalPtr()->getNextStamp();
950 }
951
952 collective::reduce::Reduce* r = nullptr;
953 if (use_group) {
954 r = theGroup()->groupReducer(group);
955 } else {
956 r = theCollective()->getReducerVrtProxy(col_proxy);
957 }
958
959 r->reduceImmediate<f>(root_node, msg.get(), cur_stamp, num_elms);
960
961 vt_debug_print(
962 normal, vrt_coll,
963 "reduceMsg: col_proxy={:x}, num_elms={}\n",
964 col_proxy, num_elms
965 );
966
967 theMsg()->popEpoch(cur_epoch);
968 return messaging::PendingSend{nullptr};
969}

1545template <typename ColT>
1546void CollectionManager::finishModification(
1547 CollectionProxyWrapType<ColT> const& proxy, ModifierToken&& token
1548) {
1549 using IndexType = typename ColT::IndexType;
1550
1551 auto untyped_proxy = proxy.getProxy();
1552
1553 auto const epoch = token.modifyEpoch();
1554
1555 vt_debug_print(
1556 normal, vrt_coll,
1557 "finishModification: epoch={:x}\n", epoch
1558 );
1559
1560 theTerm()->finishedEpoch(epoch);
1561 runSchedulerThrough(epoch);
1562
1563 // Compute the proper reduce stamp for the insertions that took place
1564 using StrongSeq = collective::reduce::detail::StrongSeq;
1565 using SeqType = typename StrongSeq::Type;
1566
1567 auto elm_holder = findElmHolder<IndexType>(untyped_proxy);
1568 SeqType min_seq = std::numeric_limits<SeqType>::max();
1569 elm_holder->foreach([&](IndexType const&, Indexable<IndexType>* c) {
1570 // skip zeros since they are insertions that just happened
1571 if (*c->reduce_stamp_ != 0) {
1572 min_seq = std::min(*c->reduce_stamp_, min_seq);
1573 }
1574 });
1575
1576 // Compute the global min stamp with a reduction
1577 auto r = theCollection()->reducer();
1578
1579 using collective::reduce::makeStamp;
1580 using collective::reduce::StrongUserID;
1581
1582 NodeType collective_root = 0;
1583 auto stamp = makeStamp<StrongUserID>(untyped_proxy);
1584 auto msg = makeMessage<CollectionStampMsg>(untyped_proxy, min_seq);
1585 auto cb = theCB()->makeBcast<&CollectionManager::computeReduceStamp>();
1586 r->reduce<collective::MinOp<SeqType>>(collective_root, msg.get(), cb, stamp);
1587
1588 theSched()->runSchedulerWhile([untyped_proxy]{
1589 return theCollection()->reduce_stamp_.find(untyped_proxy) ==
1590 theCollection()->reduce_stamp_.end();
1591 });
1592
1593 auto iter = reduce_stamp_.find(untyped_proxy);
1594 vtAssert(iter != reduce_stamp_.end(), "Must have value");
1595
1596 SeqType const global_min_stamp = iter->second;
1597 reduce_stamp_.erase(iter);
1598
1599 // set all new insertions to proper value
1600 elm_holder->foreach([&](IndexType const&, Indexable<IndexType>* c) {
1601 if (*c->reduce_stamp_ == 0) {
1602 *c->reduce_stamp_ = global_min_stamp;
1603 }
1604 });
1605
1606 vt_debug_print(
1607 verbose, vrt_coll,
1608 "finishedInserting: creating new group\n"
1609 );
1610
1611 runInEpochCollective([&]{ constructGroup<ColT>(untyped_proxy); });
1612}

1645template <typename ColT, typename MsgT>
1646void CollectionManager::insert(
1647 CollectionProxyWrapType<ColT> const& proxy, typename ColT::IndexType idx,
1648 NodeType const node, ModifierToken& token, MsgSharedPtr<MsgT> insert_msg,
1649 bool pinged_home_already
1650) {
1651 using IndexType = typename ColT::IndexType;
1652
1653 auto const modify_epoch = token.modifyEpoch();
1654 auto const untyped_proxy = proxy.getProxy();
1655 vtAssert(modify_epoch != no_epoch, "Modification epoch should be valid");
1656
1657 vt_debug_print(
1658 normal, vrt_coll,
1659 "insert: proxy={:x}\n",
1660 untyped_proxy
1661 );
1662
1663 theMsg()->pushEpoch(modify_epoch);
1664
1665 auto const mapped_node = getMappedNode<ColT>(proxy, idx);
1666 auto const has_explicit_node = node != uninitialized_destination;
1667 auto const insert_node = has_explicit_node ? node : mapped_node;
1668 auto const this_node = theContext()->getNode();
1669
1670 bool proceed_with_insertion = true;
1671
1672 vt_debug_print(
1673 normal, vrt_coll,
1674 "insert: insert_node={}, mapped_node={}\n",
1675 insert_node, mapped_node
1676 );
1677
1678 if (insert_node == this_node and not pinged_home_already) {
1679 // Case 0--insertion from home node onto home node (or message sent from
1680 // another node to insert on home node)
1681 if (mapped_node == this_node) {
1682 auto elm_holder = findElmHolder<IndexType>(untyped_proxy);
1683 auto const elm_exists = elm_holder->exists(idx);
1684 if (elm_exists) {
1685 // element exists here and is live--return
1686 proceed_with_insertion = false;
1687 } else {
1688 auto lm = theLocMan()->getCollectionLM<IndexType>(untyped_proxy);
1689 auto elm_lives_somewhere = lm->isCached(idx);
1690 if (elm_lives_somewhere) {
1691 // element exists somewhere in the system and since we are the home
1692 // we check the cache to determine if it has been inserted
1693 proceed_with_insertion = false;
1694 }
1695 }
1696 } else {
1697 // Case 1: insertion from the non-home node---we must check if the home
1698 // has a reserved entry from another insertion. If so, we cancel the
1699 // insertion---otherwise, we reserve for this insertion
1700 auto msg = makeMessage<InsertMsg<ColT, MsgT>>(
1701 proxy, idx, insert_node, mapped_node, modify_epoch, insert_msg
1702 );
1703 theMsg()->markAsCollectionMessage(msg);
1704 theMsg()->sendMsg<InsertMsg<ColT, MsgT>, pingHomeHandler<ColT, MsgT>>(
1705 mapped_node, msg
1706 );
1707 proceed_with_insertion = false;
1708 }
1709 }
1710
1711 if (insert_node == this_node and proceed_with_insertion) {
1712 auto cons_fn = detail::InsertMsgDispatcher<MsgT, ColT>::makeCons(insert_msg);
1713 makeCollectionElement<ColT>(untyped_proxy, idx, mapped_node, cons_fn);
1714 } else if (insert_node != this_node) {
1715 auto msg = makeMessage<InsertMsg<ColT, MsgT>>(
1716 proxy, idx, insert_node, mapped_node, modify_epoch, insert_msg
1717 );
1718 theMsg()->markAsCollectionMessage(msg);
1719 theMsg()->sendMsg<InsertMsg<ColT, MsgT>,insertHandler<ColT, MsgT>>(
1720 insert_node, msg
1721 );
1722 }
1723
1724 theMsg()->popEpoch(modify_epoch);
1725}

1728void CollectionManager::destroyElm(
1729 CollectionProxyWrapType<ColT> const& proxy, typename ColT::IndexType idx,
1730 ModifierToken& token
1731) {

1733
1734 auto const modify_epoch = token.modifyEpoch();

1815 );
1816
1817 vt_debug_print(
1818 terse, vrt_coll, "migrating from {} to {}\n", this_node, dest
1819 );

1822 auto const& proxy = CollectionProxy<ColT, IndexT>(col_proxy).operator()(
1823 idx
1824 );

1826 vtAssert(
1827 elm_holder != nullptr, "Element must be registered here"
1828 );

1830#if vt_check_enabled(runtime_checks)
1831 {
1832 bool const exists = elm_holder->exists(idx);
1833 vtAssert(
1834 exists, "Local element must exist here for migration to occur"

1840 if (!exists) {
1841 return MigrateStatus::ElementNotLocal;
1842 }

1844 vt_debug_print(
1845 verbose, vrt_coll,
1846 "migrateOut: (before remove) holder numElements={}\n",
1847 elm_holder->numElements()
1848 );

1851 vt_debug_print(
1852 normal, vrt_coll,
1853 "migrateOut: do not migrate last element\n"
1854 );
1855 return MigrateStatus::ElementNotLocal;
1856 }
1857
1858 auto col_unique_ptr = elm_holder->remove(idx);
1859 auto& typed_col_ref = *static_cast<ColT*>(col_unique_ptr.get());
1860
1861 vt_debug_print(
1862 verbose, vrt_coll,
1863 "migrateOut: (after remove) holder numElements={}\n",
1864 elm_holder->numElements()
1865 );

1867 /*
1868 * Invoke the virtual prelude migrate out function
1869 */
1870 col_unique_ptr->preMigrateOut();
1871

1873 verbose, vrt_coll,
1874 "migrateOut: col_proxy={:x}, idx={}, dest={}: serializing collection elm\n",
1875 col_proxy, print_index(idx), dest
1876 );

1889 * Invoke the virtual epilog migrate out function
1890 */

1892

1897 );

1905
1906 auto const home_node = getMappedNode<IndexT>(col_proxy, idx);
1907 elm_holder->applyListeners(
1908 listener::ElementEventEnum::ElementMigratedOut, idx, home_node
1909 );
1910
1911 return MigrateStatus::MigratedToRemote;
1912 }
1913
1914#if vt_check_enabled(runtime_checks)
1915 vtAssert(false, "Migration should only be called when to_node is != this_node");
1916#endif

1918 return MigrateStatus::NoMigrationNecessary;
1919}
