60template <typename T, typename SerializerT>
61void virtualSerialize(T*& base, SerializerT& s) {
62 //We can't support traited serializing with virtual types.
63 static_assert(std::is_same_v<SerializerT, typename SerializerT::TraitlessT>, "User Traits are incompatible with virtual serialization");
64
65 // Get the real base in case this is called on a derived type
66 using BaseT = ::checkpoint::dispatch::vrt::checkpoint_base_type_t<T>;
67 auto serializer_idx = serializer_registry::makeObjIdx<BaseT, SerializerT>();
68
69 base->_checkpointDynamicSerialize(&s, serializer_idx, no_type_idx);
70}

86template <typename T, typename SerializerT, typename _enabled = void>
87struct SerializeVirtualTypeIfNeeded;

89template <typename T, typename SerializerT>
90struct SerializeVirtualTypeIfNeeded<
91 T,
92 SerializerT,
93 typename std::enable_if_t<
94 dispatch::vrt::VirtualSerializeTraits<T>::has_not_virtual_serialize
95 >
96>
97{
98 static dispatch::vrt::TypeIdx apply(SerializerT&, T*) {
99 // no type idx needed in this case
100 return dispatch::vrt::no_type_idx;
101 }
102};
103
104template <typename T, typename SerializerT>
105struct SerializeVirtualTypeIfNeeded<
106 T,
107 SerializerT,
108 typename std::enable_if_t<
109 dispatch::vrt::VirtualSerializeTraits<T>::has_virtual_serialize
110 >
111>
112{
113 static typename dispatch::vrt::TypeIdx apply(SerializerT& s, T* target) {
114 dispatch::vrt::TypeIdx entry = dispatch::vrt::no_type_idx;
115
116 if (not s.isUnpacking()) {
117 entry = target->_checkpointDynamicTypeIndex();
118 }
119
120 // entry doesn't count as part of a footprint
121 if (not s.isFootprinting()) {
122 s | entry;
123 }
124
125 if (target != nullptr) {
126 // Support deserialization in place, and make sure it's safe
127 checkpointAssert(entry == target->_checkpointDynamicTypeIndex(),
128 "Trying to deserialize in place over a mismatched type");
129 }
130
131 return entry;
132 }
133};

223template <typename SerializerT, typename T>
224void reconstructPointedToObjectIfNeeded(SerializerT& s, T*& target) {
225 auto entry = SerializeVirtualTypeIfNeeded<T, SerializerT>::apply(s, target);
226
227 if (target != nullptr) {
228 // Support deserialization in place; assumes matching of virtual
229 // types was checked in serializeDynamicTypeIndex()
230 return;
231 }
232
233 if (s.isUnpacking()) {
234 target = ReconstructAsVirtualIfNeeded<T, SerializerT>::apply(s, entry);
235 }
236}
