98template <typename KeyT, typename ValueT>
99void LocationCache<KeyT, ValueT>::insert(KeyT const& key, ValueT const& value) {
100 auto iter = lookup_.find(key);
101
102 vt_debug_print(
103 verbose, location,
104 "location cache: insert: found={}, size={}\n",
105 print_bool(iter != lookup_.end()), lookup_.size()
106 );
107
108 if (iter == lookup_.end()) {
109 if (lookup_.size() + 1 > max_size_) {
110 auto last_iter = cache_.crbegin();
111 lookup_.erase(std::get<0>(*last_iter));
112 cache_.pop_back();
113 }
114
115 cache_.push_front(std::make_tuple(key, value));
116
117 lookup_.emplace(
118 std::piecewise_construct,
119 std::forward_as_tuple(key),
120 std::forward_as_tuple(cache_.begin())
121 );
122 } else {
123 std::get<1>(*iter->second) = value;
124 cache_.splice(cache_.cbegin(), cache_, iter->second);
125 }
126}
