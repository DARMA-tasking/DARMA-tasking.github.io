63struct CallbackBase {
64 using SignalType = SignalT;

88 { }

90 virtual ~CallbackBase() = default;
91
92public:
93 bool expired() const { return reference_counted_ && refs_ < 1; }
94
95protected:

97
98 virtual void trigger_(
99 SignalDataType* data, [[maybe_unused]] PipeType const& pipe_id
100 ) {
101 return trigger_(data);
102 }
103
104public:
105 void trigger(SignalDataType* data, PipeType const& pipe_id) {
106 if (reference_counted_) {
107 refs_--;
108 triggered_++;
109 }
110 vt_debug_print(
111 verbose, pipe,
112 "CallbackBase: (before) invoke trigger_: pipe={:x}\n",
113 pipe_id
114 );
115 trigger_(data,pipe_id);
116 vt_debug_print(
117 verbose, pipe,
118 "CallbackBase: (after) invoke trigger_: pipe={:x}\n",
119 pipe_id
120 );
121 }
122
123 bool finished() const {
124 if (reference_counted_) {
125 vtAssert(refs_ != -1, "refs_ must be a valid positive integer");
126 return refs_ == 0;
127 } else {
128 return false;
129 }
130 }
131
132 template <typename SerializerT>
133 void serialize(SerializerT& s) {
134 s | reference_counted_ | triggered_ | refs_;
135 }
136
137private:
138 bool reference_counted_ = true;
139 PipeRefType triggered_ = 0;
140 PipeRefType refs_ = 1;
141};
