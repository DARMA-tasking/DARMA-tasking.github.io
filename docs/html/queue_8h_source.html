54template <typename T>
55struct Queue {
56 Queue() = default;
57 Queue(Queue const&) = default;
58 Queue(Queue&&) = default;
59
60 void push(T elm) {
61 if (buf_.full()) {
62 impl_.push(elm);
63 } else {
64 buf_.push(elm);
65 }
66 }
67
68 void emplace(T&& elm) {
69 if (buf_.full()) {
70 impl_.emplace(std::move(elm));
71 } else {
72 buf_.push(std::move(elm));
73 }
74 }
75
76 T pop() {
77 vtAssert(not buf_.empty(), "Must have at least one element");
78 bool const full_buf = buf_.full();
79 auto t = buf_.pop();
80 if (full_buf and not impl_.empty()) {
81 buf_.push(std::move(impl_.front()));
82 impl_.pop();
83 }
84 return t;
85 }
86
87 std::size_t size() const {
88 return buf_.size() + impl_.size();
89 }
90
91 bool empty() const {
92 vtAssert(not buf_.empty() or impl_.empty(), "Buf empty implies queue empty");
93 return buf_.empty();
94 }
95
96 template <typename Serializer>
97 void serialize(Serializer& s) {
98 s | impl_;
99 s | buf_;
100 }
101
102private:
103 util::container::CircularBuffer<T, 64> buf_;
104 std::queue<T, std::deque<T>> impl_;
105};
