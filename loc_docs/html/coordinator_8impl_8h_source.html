61template <typename EntityID, typename Comm>
62void Coordinator<EntityID, Comm>::getLocation(
63 EntityID const& id, NodeType home, NodeAction action
64) {
65 if (local_registered_.count(id) != 0) {
66 action(this_node_);
67 return;
68 }
69 if (recs_.exists(id)) {
70 action(recs_.get(id).getRemoteNode());
71 return;
72 }
73 auto& list = pending_[id];
74 list.push_back(std::move(action));
75 // Only the first waiter issues the request; the home is not this rank here
76 // (otherwise it would be locally known once registered).
77 if (list.size() == 1 && home != this_node_) {
78 comm_.template send<&ThisType::locationRequest>(
79 home, handle_, id, this_node_
80 );
81 }
82}

84template <typename EntityID, typename Comm>
85void Coordinator<EntityID, Comm>::entityExists(
86 EntityID const& id, NodeType home, ExistsAction action
87) {
88 if (local_registered_.count(id) != 0) {
89 action(true, this_node_);
90 return;
91 }
92 if (recs_.exists(id)) {
93 action(true, recs_.get(id).getRemoteNode());
94 return;
95 }
96 if (home == this_node_) {
97 // The home is authoritative: unknown here means it does not exist.
98 action(false, no_node);
99 return;
100 }
101 auto& list = pending_exists_[id];
102 list.push_back(std::move(action));
103 if (list.size() == 1) {
104 comm_.template send<&ThisType::existsRequest>(home, handle_, id, this_node_);
105 }
106}

128template <typename EntityID, typename Comm>
129void Coordinator<EntityID, Comm>::locationRequest(
130 EntityID id, NodeType requester
131) {
132 if (recs_.exists(id)) {
133 loc_asks_[id].insert(requester);
134 comm_.template send<&ThisType::resolveResponse>(
135 requester, handle_, id, recs_.get(id).getRemoteNode()
136 );
137 } else if (local_registered_.count(id) != 0) {
138 loc_asks_[id].insert(requester);
139 comm_.template send<&ThisType::resolveResponse>(
140 requester, handle_, id, this_node_
141 );
142 } else {
143 // Not known yet: buffer until the entity registers / is updated here.
144 pending_home_[id].push_back(requester);
145 }
146}

175template <typename EntityID, typename Comm>
176void Coordinator<EntityID, Comm>::existsResponse(
177 EntityID id, bool exists, NodeType node
178) {
179 if (exists) {
180 recs_.update(id, makeRec(id, node));
181 }
182 auto it = pending_exists_.find(id);
183 if (it != pending_exists_.end()) {
184 for (auto& action : it->second) {
185 action(exists, node);
186 }
187 pending_exists_.erase(it);
188 }

214template <typename EntityID, typename Comm>
215void Coordinator<EntityID, Comm>::announceLocation(
216 EntityID const& id, NodeType node
217) {
218 recs_.insert(id, /*home=*/this_node_, makeRec(id, node));
219
220 // Eagerly refresh caches of ranks we have already answered.
221 auto ask_it = loc_asks_.find(id);
222 if (ask_it != loc_asks_.end()) {
223 for (auto asker : ask_it->second) {
224 comm_.template send<&ThisType::resolveResponse>(asker, handle_, id, node);
225 }
226 }
227
228 // Answer ranks that asked before the entity was known, and remember them
229 // for future eager refreshes.
230 auto pend_it = pending_home_.find(id);
231 if (pend_it != pending_home_.end()) {
232 auto& askers = loc_asks_[id];
233 for (auto requester : pend_it->second) {
234 comm_.template send<&ThisType::resolveResponse>(
235 requester, handle_, id, node
236 );
237 askers.insert(requester);
238 }
239 pending_home_.erase(pend_it);
240 }
241
242 // Satisfy local waiters on this (home) rank.
243 flushPending(pending_, id, node);
244}
