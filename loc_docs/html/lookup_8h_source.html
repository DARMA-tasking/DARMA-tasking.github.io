26template <typename KeyT, typename ValueT>
27struct LocLookup {
28 LocLookup(LocationSizeType const& in_max_cache_size, NodeType in_this_node)
29 : max_cache_size_(in_max_cache_size),
30 cache_(in_max_cache_size),
31 this_node_(in_this_node)
32 { }
33
34 bool exists(KeyT const& key) const {
35 return directory_.exists(key) or cache_.exists(key);
36 }
37
38 LocationSizeType getCacheSize() const { return cache_.getSize(); }
39
40 ValueT const& get(KeyT const& key) {
41 auto dir_iter = directory_.getIter(key);
42 if (dir_iter != directory_.getIterEnd()) {
43 return dir_iter->second;
44 }
45 return cache_.get(key);
46 }
47
48 void remove(KeyT const& key) {
49 directory_.remove(key);
50 cache_.remove(key);
51 }
52
53 void insert(KeyT const& key, NodeType const home, ValueT const& value) {
54 // If this node is the home, maintain location in the permanent directory,
55 // otherwise insert/update in the local cache of locations.
56 if (this_node_ == home) {
57 directory_.insert(key, value);
58 } else {
59 cache_.insert(key, value);
60 }
61 }
62
63 void update(KeyT const& key, ValueT const& value) {
64 auto dir_iter = directory_.getIter(key);
65 if (dir_iter != directory_.getIterEnd()) {
66 dir_iter->second = value;
67 } else {
68 cache_.insert(key, value);
69 }
70 }
71
72 void clearCache() {
73 cache_ = LocationCache<KeyT, ValueT>{max_cache_size_};
74 }
75
77 auto const& getDirectory() const { return directory_; }
78
79 template <typename Serializer>
80 void serialize(Serializer& s) {
81 s | max_cache_size_ | directory_ | cache_ | this_node_;
82 }
83
84private:
85 LocationSizeType max_cache_size_ = 0;
86 Directory<KeyT, ValueT> directory_;
87 LocationCache<KeyT, ValueT> cache_;
88 NodeType this_node_ = no_node;
89};
