211 class BufferIntInterpreter {
212 char* const ptr_;
213 public:
214 explicit BufferIntInterpreter(char* buf) : ptr_(buf) {
215 assert((reinterpret_cast<std::uintptr_t>(buf) % alignof(int)) == 0);
216 }
217 int& handlerIndex() { return *reinterpret_cast<int*>(ptr_); }
218 int& classIndex() { return *(reinterpret_cast<int*>(ptr_) + 1); }
219 int& isTermination() { return *(reinterpret_cast<int*>(ptr_) + 2); }
220 };

222 template <auto fn, typename... Args>
223 void sendImpl(int dest, int idx, bool is_termination_msg, Args&&... args) {
224 // Layout of buffer:
225 // <handler_index : int> -> used to get the correct registered handler
226 // <class_index : int> -> used to dispatch to the correct class instance
227 // <is_termination_msg : int> -> used to determine if it should count as a normal message or termination message
228 // <serialized buffer> -> the actual serialized buffer from magistrate
229
230 // Validate destination rank
231 if (dest < 0 || dest >= numRanks()) {
232 VT_LB_LOG(Communicator, terse, "Invalid destination rank {}\n", dest);
233 throw std::runtime_error("Invalid destination rank");
234 }
235
236 auto const handler_index = detail::registerMember<fn>();
237 std::tuple<std::decay_t<Args>...> tup{std::forward<Args>(args)...};
238 char* send_ptr = nullptr;
239 std::size_t send_ptr_size = 0;
240 std::tuple<MPI_Request, std::unique_ptr<char[]>> pending_operation;
241 auto ret = checkpoint::serialize(
242 tup, [&send_ptr, &send_ptr_size, &pending_operation](std::size_t size) -> char* {
243 auto constexpr extra_size = 3*sizeof(int);
244 send_ptr_size = size + extra_size;
245 pending_operation = std::make_tuple(MPI_Request{}, std::make_unique<char[]>(send_ptr_size));
246 send_ptr = std::get<1>(pending_operation).get();
247 return send_ptr + extra_size;
248 }
249 );
250 BufferIntInterpreter buf_interpreter(send_ptr);
251 buf_interpreter.handlerIndex() = handler_index;
252 buf_interpreter.classIndex() = idx;
253 buf_interpreter.isTermination() = is_termination_msg ? 1 : 0;
254
255 VT_LB_LOG(
256 Communicator, normal,
257 "MPI_Isend to {} handler_index={} class_index={} is_termination={}\n",
258 dest,
259 buf_interpreter.handlerIndex(),
260 buf_interpreter.classIndex(),
261 is_termination_msg ? 1 : 0
262 );
263
264 MPI_Request req;
265 MPI_Isend(
266 send_ptr, send_ptr_size, MPI_BYTE, dest, 0, comm_,
267 &req
268 );
269 std::get<0>(pending_operation) = req;
270 pending_.emplace_back(std::move(pending_operation));
271 if (!is_termination_msg) {
272 termination_detector_->notifyMessageSend();
273 }
274 }

296 template <typename T>
297 std::unordered_map<int, std::vector<T>> allgather(T const* sendbuf, int sendcount) {
298 static_assert(std::is_trivially_copyable_v<T>, "CommMPI::allgather requires trivially copyable T");
299 MPI_Datatype datatype = deduceDatatype<T>();
300
301 if (comm_ == MPI_COMM_NULL) {
302 throw std::runtime_error("CommMPI not initialized (MPI_Comm is MPI_COMM_NULL)");
303 }
304
305 const int n = numRanks();
306 std::vector<int> counts(n, 0);
307 // Gather local counts from each rank (block for small metadata)
308 MPI_Allgather(&sendcount, 1, MPI_INT, counts.data(), 1, MPI_INT, comm_);
309
310 std::vector<int> displs(n, 0);
311 int total = 0;
312 for (int i = 0; i < n; ++i) {
313 displs[i] = total;
314 total += counts[i];
315 }
316
317 std::vector<T> recvbuf;
318 recvbuf.resize(static_cast<std::size_t>(total));
319
320 VT_LB_LOG(Communicator, normal, "MPI_Iallgatherv sendcount={} total={} ranks={}\n", sendcount, total, n);
321
322 MPI_Request req;
323 MPI_Iallgatherv(
324 sendbuf, sendcount, datatype,
325 recvbuf.data(), counts.data(), displs.data(), datatype,
326 comm_, &req
327 );
328
329 int flag = 0;
330 while (!flag) {
331 MPI_Status status;
332 MPI_Test(&req, &flag, &status);
333 poll();
334 }
335
336 std::unordered_map<int, std::vector<T>> by_rank;
337 by_rank.reserve(static_cast<std::size_t>(n));
338 for (int r = 0; r < n; ++r) {
339 const int cnt = counts[r];
340 const int off = displs[r];
341 if (cnt > 0) {
342 by_rank.emplace(
343 r,
344 std::vector<T>(recvbuf.begin() + off, recvbuf.begin() + off + cnt)
345 );
346 } else {
347 by_rank.emplace(r, std::vector<T>{});
348 }
349 }
350
351 return by_rank;
352 }

357 // Helper to trigger static_assert for unsupported types.
358 template <typename> struct AlwaysFalse : std::false_type {};
359
360 // Map a limited set of basic C++ types to MPI_Datatype.
361 template <typename T>
362 static MPI_Datatype deduceDatatype() {
363 if constexpr (std::is_same_v<T, char>) return MPI_CHAR;
364 else if constexpr (std::is_same_v<T, signed char>) return MPI_SIGNED_CHAR;
365 else if constexpr (std::is_same_v<T, unsigned char>) return MPI_UNSIGNED_CHAR;
366 else if constexpr (std::is_same_v<T, short>) return MPI_SHORT;
367 else if constexpr (std::is_same_v<T, unsigned short>) return MPI_UNSIGNED_SHORT;
368 else if constexpr (std::is_same_v<T, int>) return MPI_INT;
369 else if constexpr (std::is_same_v<T, unsigned int>) return MPI_UNSIGNED;
370 else if constexpr (std::is_same_v<T, long>) return MPI_LONG;
371 else if constexpr (std::is_same_v<T, unsigned long>) return MPI_UNSIGNED_LONG;
372 else if constexpr (std::is_same_v<T, long long>) return MPI_LONG_LONG;
373 else if constexpr (std::is_same_v<T, unsigned long long>) return MPI_UNSIGNED_LONG_LONG;
374 else if constexpr (std::is_same_v<T, float>) return MPI_FLOAT;
375 else if constexpr (std::is_same_v<T, double>) return MPI_DOUBLE;
376 else if constexpr (std::is_same_v<T, long double>) return MPI_LONG_DOUBLE;
377 else {
378 static_assert(AlwaysFalse<T>::value, "Unsupported T for CommMPI::allgather");
379 return MPI_DATATYPE_NULL; // unreachable
380 }
381 }
