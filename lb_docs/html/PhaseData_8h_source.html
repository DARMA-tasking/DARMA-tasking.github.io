61 void addTask(Task const& t) { tasks_.emplace(t.getId(), t); }
62 void addCommunication(Edge const& e) {
63 for (int i = 0; i < static_cast<int>(communications_.size()); ++i) {
64 auto& comm = communications_[i];
65 if (comm.getFrom() == e.getFrom() && comm.getTo() == e.getTo()) {
66 comm = e;
67 return;
68 }
69 }
70 communications_.push_back(e);
71 }
72 void aggregateCommunication(Edge const& e) {
73 for (int i = 0; i < static_cast<int>(communications_.size()); ++i) {
74 auto& comm = communications_[i];
75 if (comm.getFrom() == e.getFrom() && comm.getTo() == e.getTo()) {
76 comm.setVolume(comm.getVolume() + e.getVolume());
77 comm.setNumMessages(comm.getNumMessages() + e.getNumMessages());
78 task_messages_out_[e.getFrom()] += e.getNumMessages();
79 task_messages_in_[e.getTo()] += e.getNumMessages();
80 return;
81 }
82 }
83 communications_.push_back(e);
84 task_messages_out_[e.getFrom()] += e.getNumMessages();
85 task_messages_in_[e.getTo()] += e.getNumMessages();
86 }

176 void purgeDanglingCommunications() {
177 std::vector<Edge> filtered;
178 filtered.reserve(communications_.size());
179 for (auto const& e : communications_) {
180 bool from_present = hasTask(e.getFrom());
181 bool to_present = hasTask(e.getTo());
182 // Keep edge unless both sides are missing
183 if (from_present || to_present) {
184 filtered.push_back(e);
185 }
186 }
187 communications_.swap(filtered);
188 }
