19inline std::string buildTaskGraphDOT(
20 vt_lb::model::PhaseData const& pd,
21 Clusterer const* clusterer,
22 bool show_clusters,
23 bool show_loads
24) {
25 using TaskType = vt_lb::model::TaskType;
26 std::ostringstream out;
27 out << "digraph TaskGraph {\n";
28 out << " rankdir=LR;\n";
29 out << " node [shape=ellipse, fontname=\"Helvetica\"];\n";
30
31 std::unordered_map<TaskType,int> cluster_map;
32 std::vector<Cluster> clusters;
33 if (clusterer && show_clusters) {
34 cluster_map = clusterer->taskToCluster();
35 clusters = clusterer->clusters();
36 }
37
38 // Compute per-cluster boundary bytes (sum of edges crossing cluster boundary)
39 // Use integer accumulation to avoid FP off-by-one when printing.
40 std::unordered_map<int, unsigned long long> cluster_boundary_bytes;
41 // New: intra-cluster total bytes (sum of edges where both endpoints in same cluster)
42 std::unordered_map<int, unsigned long long> cluster_intra_bytes;
43 if (show_clusters && !clusters.empty()) {
44 // Build fast membership sets per cluster
45 std::unordered_map<int, std::unordered_set<TaskType>> cluster_members;
46 for (auto const& cl : clusters) {
47 cluster_members[cl.id] = std::unordered_set<TaskType>(cl.members.begin(), cl.members.end());
48 cluster_boundary_bytes[cl.id] = 0ULL;
49 cluster_intra_bytes[cl.id] = 0ULL; // initialize intra
50 }
51 // Iterate communications and add volume to intra or boundary
52 for (auto const& e : pd.getCommunications()) {
53 auto f = e.getFrom();
54 auto t = e.getTo();
55 // Skip edges without tasks
56 if (!pd.hasTask(f) || !pd.hasTask(t)) continue;
57
58 // Round each edge volume to integer bytes before summing
59 unsigned long long vol = static_cast<unsigned long long>(std::llround(e.getVolume()));
60
61 auto cf_it = cluster_map.find(f);
62 auto ct_it = cluster_map.find(t);
63 if (cf_it != cluster_map.end() || ct_it != cluster_map.end()) {
64 if (cf_it != cluster_map.end() && ct_it != cluster_map.end()) {
65 int cf = cf_it->second;
66 int ct = ct_it->second;
67 if (cf == ct) {
68 // Intra-cluster communication
69 cluster_intra_bytes[cf] += vol;
70 } else {
71 // Crossing cluster boundary
72 cluster_boundary_bytes[cf] += vol;
73 cluster_boundary_bytes[ct] += vol;
74 }
75 } else {
76 // One endpoint is in some cluster, the other is unclustered
77 if (cf_it != cluster_map.end()) {
78 cluster_boundary_bytes[cf_it->second] += vol;
79 }
80 if (ct_it != cluster_map.end()) {
81 cluster_boundary_bytes[ct_it->second] += vol;
82 }
83 }
84 }
85 }
86 }
87
88 // Color palette
89 std::array<const char*, 12> palette = {
90 "#1f77b4","#ff7f0e","#2ca02c","#d62728",
91 "#9467bd","#8c564b","#e377c2","#7f7f7f",
92 "#bcbd22","#17becf","#393b79","#637939"
93 };
94
95 // Emit clusters or standalone nodes
96 if (show_clusters && !clusters.empty()) {
97 for (auto const& cl : clusters) {
98 auto color = palette[cl.id % palette.size()];
99 unsigned long long bytes_boundary = cluster_boundary_bytes.count(cl.id) ? cluster_boundary_bytes[cl.id] : 0ULL;
100 unsigned long long bytes_intra = cluster_intra_bytes.count(cl.id) ? cluster_intra_bytes[cl.id] : 0ULL;
101 out << " subgraph cluster_" << cl.id << " {\n";
102 out << " label=\"cluster " << cl.id
103 << "\\nload=" << std::fixed << std::setprecision(2) << cl.load
104 << "\\nbytes_intra=" << bytes_intra
105 << "\\nbytes_boundary=" << bytes_boundary << "\";\n";
106 out << " color=\"" << color << "\";\n";
107 for (auto t : cl.members) {
108 auto const* task = pd.getTask(t);
109 if (!task) continue;
110 out << " \"" << t << "\" [style=filled, fillcolor=\"" << color << "\", label=\""
111 << t;
112 if (show_loads) {
113 out << "\\nL=" << std::fixed << std::setprecision(1) << task->getLoad();
114 }
115 out << "\"];\n";
116 }
117 out << " }\n";
118 }
119 } else {
120 // Just tasks
121 for (auto const& kv : pd.getTasksMap()) {
122 auto const& task = kv.second;
123 out << " \"" << task.getId() << "\" [label=\"" << task.getId();
124 if (show_loads) {
125 out << "\\nL=" << std::fixed << std::setprecision(1) << task.getLoad();
126 }
127 out << "\"];\n";
128 }
129 }
130
131 // Edges
132 for (auto const& e : pd.getCommunications()) {
133 auto f = e.getFrom();
134 auto t = e.getTo();
135 if (!pd.hasTask(f) || !pd.hasTask(t)) continue;
136 out << " \"" << f << "\" -> \"" << t << "\" [label=\"" << std::fixed << std::setprecision(0)
137 << e.getVolume() << "\"";
138 // If clusters shown, style inter-cluster edges bold
139 if (show_clusters && !cluster_map.empty()) {
140 auto cf_it = cluster_map.find(f);
141 auto ct_it = cluster_map.find(t);
142 if (cf_it != cluster_map.end() && ct_it != cluster_map.end() && cf_it->second != ct_it->second) {
143 out << ", penwidth=2, color=red";
144 }
145 }
146 out << "];\n";
147 }
148
149 out << "}\n";
150 return out.str();
151}
