61template <typename CommT>
62struct GraphEdgeResolver {
63 using ThisType = GraphEdgeResolver<CommT>;
64 using HandleType = typename CommT::template HandleType<ThisType>;
65 using PhaseData = vt_lb::model::PhaseData;
66 using Edge = vt_lb::model::Edge;
67 using TaskType = vt_lb::model::TaskType;
68 using RankType = vt_lb::model::RankType;
69
70 explicit GraphEdgeResolver(CommT& comm, PhaseData& pd)
71 : comm_(comm.clone()), pd_(&pd)
72 {
73 handle_ = comm_.template registerInstanceCollective<ThisType>(this);
74 }
75
76 void run() {
77 // first remove any communication where neither endpoint task exists
78 pd_->purgeDanglingCommunications();
79
80 std::list<Edge*> to_resolve;
81 // resolve rank ID -> list of task IDs
82 std::unordered_map<int, std::vector<model::TaskType>> rank_to_tasks;
83 std::unordered_map<int, std::vector<model::TaskType>> resolve_tasks_requests;
84
85 auto const num_ranks = comm_.numRanks();
86
87 // Proactively inform brokers of all local tasks to handle asymmetric graphs
88 // Assumes PhaseData exposes tasks as a map from TaskType -> Task
89 for (auto const& id : pd_->getTaskIds()) {
90 rank_to_tasks[id % num_ranks].push_back(id);
91 }
92
93 for (auto& e : pd_->getCommunicationsRef()) {
94 auto from_task = pd_->getTask(e.getFrom());
95 auto to_task = pd_->getTask(e.getTo());
96 vt_lb_assert(
97 from_task != nullptr || to_task != nullptr,
98 "At least one endpoint task must exist after purge"
99 );
100 if (from_task == nullptr || to_task == nullptr) {
101 if (from_task == nullptr) {
102 resolve_tasks_requests[e.getFrom() % num_ranks].push_back(e.getFrom());
103 } else {
104 rank_to_tasks[e.getFrom() % num_ranks].push_back(e.getFrom());
105 }
106 if (to_task == nullptr) {
107 resolve_tasks_requests[e.getTo() % num_ranks].push_back(e.getTo());
108 } else {
109 rank_to_tasks[e.getTo() % num_ranks].push_back(e.getTo());
110 }
111
112 to_resolve.push_back(&e);
113 }
114 }
115
116 // Deduplicate per-rank task IDs
117 for (auto& kv : rank_to_tasks) {
118 auto& v = kv.second;
119 std::sort(v.begin(), v.end());
120 v.erase(std::unique(v.begin(), v.end()), v.end());
121 }
122 for (auto& kv : resolve_tasks_requests) {
123 auto& v = kv.second;
124 std::sort(v.begin(), v.end());
125 v.erase(std::unique(v.begin(), v.end()), v.end());
126 }
127
128 for (auto& kv : rank_to_tasks) {
129 int target_rank = kv.first;
130 auto& task_ids = kv.second;
131 if (!task_ids.empty()) {
132 handle_[target_rank].template send<&ThisType::informTaskLocation>(comm_.getRank(), task_ids);
133 }
134 }
135
136 for (auto& kv : resolve_tasks_requests) {
137 int target_rank = kv.first;
138 auto& task_ids = kv.second;
139 if (!task_ids.empty()) {
140 handle_[target_rank].template send<&ThisType::resolveTasks>(comm_.getRank(), task_ids);
141 }
142 }
143
144 while (comm_.poll()) {
145 // do nothing
146 }
147
148 for (auto e : to_resolve) {
149 auto from_task = pd_->getTask(e->getFrom());
150 auto to_task = pd_->getTask(e->getTo());
151 vt_lb_assert(
152 from_task != nullptr || to_task != nullptr,
153 "At least one endpoint task must exist after purge"
154 );
155 if (from_task == nullptr) {
156 auto it = resolved_tasks_.find(e->getFrom());
157 vt_lb_assert(
158 it != resolved_tasks_.end(),
159 "From task must have been resolved"
160 );
161 e->setToRank(comm_.getRank());
162 e->setFromRank(it->second);
163 }
164 if (to_task == nullptr) {
165 auto it = resolved_tasks_.find(e->getTo());
166 vt_lb_assert(
167 it != resolved_tasks_.end(),
168 "To task must have been resolved"
169 );
170 e->setToRank(it->second);
171 e->setFromRank(comm_.getRank());
172 }
173 }
174
175#if VALIDATE_RESOLVED_GRAPH_EDGES
176 auto comm2 = comm_.clone();
177 auto handle2 = comm2.template registerInstanceCollective<ThisType>(this);
178
179 // Validate end points
180 for (auto e : to_resolve) {
181 auto from_task = pd_->getTask(e->getFrom());
182 auto to_task = pd_->getTask(e->getTo());
183 if (from_task == nullptr) {
184 handle2[e->getFromRank()].template send<&ThisType::validateEndpoint>(
185 comm_.getRank(),
186 e->getFrom()
187 );
188 } else if (to_task == nullptr) {
189 handle2[e->getToRank()].template send<&ThisType::validateEndpoint>(
190 comm_.getRank(),
191 e->getTo()
192 );
193 }
194 }
195
196 while (comm2.poll()) {
197 // do nothing
198 }
199#endif
200 }
201
202 void returnTaskLocation(
203 RankType rank,
204 model::TaskType task_id
205 ) {
206 resolved_tasks_[task_id] = rank;
207 }
208
209 void informTaskLocation(
210 int rank,
211 std::vector<model::TaskType> const& task_ids
212 ) {
213 VT_LB_LOG(
214 LoadBalancer, verbose,
215 "GraphEdgeResolver::informTaskLocation: received {} task locations from rank {}\n",
216 task_ids.size(), rank
217 );
218
219 for (auto const& tid : task_ids) {
220 task_location_map_[tid] = rank;
221
222 if (auto it = resolve_tasks_requests_.find(tid); it != resolve_tasks_requests_.end()) {
223 for (auto const& req_rank : it->second) {
224 handle_[req_rank].template send<&ThisType::returnTaskLocation>(rank, tid);
225 }
226 resolve_tasks_requests_.erase(it);
227 }
228 }
229 }
230
231 void resolveTasks(
232 int rank,
233 std::vector<model::TaskType> const& task_ids
234 ) {
235 VT_LB_LOG(
236 LoadBalancer, verbose,
237 "GraphEdgeResolver::resolveTasks: received {} task resolution requests from rank {}\n",
238 task_ids.size(), rank
239 );
240
241 for (auto const& tid : task_ids) {
242 if (auto it = task_location_map_.find(tid); it != task_location_map_.end()) {
243 handle_[rank].template send<&ThisType::returnTaskLocation>(it->second, tid);
244 } else {
245 resolve_tasks_requests_[tid].push_back(rank);
246 }
247 }
248 }
249
250 void validateEndpoint(
251 int rank,
252 model::TaskType task_id
253 ) {
254 VT_LB_LOG(
255 LoadBalancer, verbose,
256 "GraphEdgeResolver::validateEndpoint: received validation request for task {} from rank {}\n",
257 task_id, rank
258 );
259
260 auto task = pd_->getTask(task_id);
261 vt_lb_assert(
262 task != nullptr,
263 "Validated task must exist locally"
264 );
265 }
266
267private:
268 CommT comm_;
269 PhaseData* pd_ = nullptr;
270 HandleType handle_;
271 std::unordered_map<model::TaskType, int> task_location_map_;
272 std::unordered_map<model::TaskType, std::vector<int>> resolve_tasks_requests_;
273 std::unordered_map<model::TaskType, int> resolved_tasks_;
274};
