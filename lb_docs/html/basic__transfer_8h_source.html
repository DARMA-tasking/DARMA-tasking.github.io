58template <comm::Communicator CommT>
59struct BasicTransfer final : Transferer<CommT> {
60 BasicTransfer(
61 CommT& comm,
62 model::PhaseData& pd,
63 std::unordered_map<int, RankInfo> const& load_info,
64 Statistics stats
65 ) : Transferer<CommT>(comm, pd),
66 load_info_(load_info),
67 stats_(stats)
68 {}
69
70 bool isOverloaded(model::LoadType load) const {
71 return load > stats_.avg;
72 }
73
74 void run(
75 TransferUtil::CMFType cmf_type,
76 TransferUtil::ObjectOrder obj_ordering,
77 CriterionEnum criterion,
78 bool deterministic,
79 model::LoadType target_max_load,
80 std::mt19937 gen_sample_,
81 std::random_device& seed_
82 ) {
83 RankInfo& cur_load = load_info_.at(this->pd_.getRank());
84
85 // Initialize transfer and rejection counters
86 int n_transfers = 0, n_rejected = 0;
87
88 // Try to migrate objects only from overloaded ranks
89 if (isOverloaded(cur_load.getScaledLoad())) {
90 std::vector<int> under = TransferUtil::makeUnderloaded(deterministic, load_info_, stats_.avg);
91
92 auto cur_tasks = this->pd_.getTasksMap();
93
94 if (under.size() > 0) {
95 std::vector<model::TaskType> ordered_obj_ids = TransferUtil::orderObjects(
96 obj_ordering, cur_tasks, cur_load, target_max_load
97 );
98
99 // Iterate through all the objects
100 for (auto iter = ordered_obj_ids.begin(); iter != ordered_obj_ids.end(); ) {
101 auto obj_id = *iter;
102 auto obj_load = cur_tasks[obj_id].getLoad();
103
104 if (cmf_type == TransferUtil::CMFType::Original) {
105 // Rebuild the relaxed underloaded set based on updated load of this node
106 under = TransferUtil::makeUnderloaded(deterministic, load_info_, stats_.avg);
107 if (under.size() == 0) {
108 break;
109 }
110 } else if (cmf_type == TransferUtil::CMFType::NormByMaxExcludeIneligible) {
111 // Rebuild the underloaded set and eliminate processors that will
112 // fail the Criterion for this object
113 under = TransferUtil::makeSufficientlyUnderloaded(
114 deterministic, load_info_, criterion, cur_load, target_max_load, obj_load
115 );
116 if (under.size() == 0) {
117 ++n_rejected;
118 iter++;
119 continue;
120 }
121 }
122 // Rebuild the CMF with the new loads taken into account
123 auto cmf = TransferUtil::createCMF(cmf_type, load_info_, target_max_load, under);
124
125 // Select a node using the CMF
126 auto const selected_rank = TransferUtil::sampleFromCMF(deterministic, under, cmf, gen_sample_, seed_);
127
128 VT_LB_LOG(
129 LoadBalancer, verbose,
130 "BasicTransfer: selected_rank={}, load_info_.size()={}\n",
131 selected_rank, load_info_.size()
132 );
133
134 // Find load of selected node
135 auto load_iter = load_info_.find(selected_rank);
136 vt_lb_assert(load_iter != load_info_.end(), "Selected node not found");
137 auto& selected_load = load_iter->second;
138
139 // Check if object is migratable and evaluate criterion for proposed transfer
140 bool is_migratable = cur_tasks[obj_id].isMigratable();
141 bool eval = Criterion(criterion)(
142 cur_load, selected_load, obj_load, target_max_load
143 );
144
145 VT_LB_LOG(
146 LoadBalancer, verbose,
147 "BasicTransfer: under.size()={}, "
148 "selected_rank={}, selected_load={}, obj_id={}, "
149 "is_migratable()={}, obj_load={}, target_max_load={}, "
150 "cur_load={}, criterion={}\n",
151 under.size(),
152 selected_rank,
153 selected_load,
154 obj_id,
155 is_migratable,
156 obj_load,
157 target_max_load,
158 cur_load,
159 eval
160 );
161
162 // Decide about proposed migration based on criterion evaluation
163 if (is_migratable && eval) {
164 ++n_transfers;
165 // Transfer the object load in seconds
166 // to match the object load units on the receiving end
167 this->migrateTask(selected_rank, cur_tasks[obj_id]);
168
169 VT_LB_LOG(
170 LoadBalancer, verbose,
171 "BasicTransfer: migrating obj_id={:x} of load={} to rank={}\n",
172 obj_id, model::LoadType(obj_load), selected_rank
173 );
174
175 // Update loads
176 cur_load.load -= obj_load;
177 selected_load.load += obj_load;
178
179 iter = ordered_obj_ids.erase(iter);
180 cur_tasks.erase(obj_id);
181 } else {
182 ++n_rejected;
183 iter++;
184 }
185
186 if (!(cur_load.getScaledLoad() > target_max_load)) {
187 break;
188 }
189 }
190 }
191 } else {
192 // do nothing (underloaded-based algorithm), waits to get work from
193 // overloaded nodes
194 }
195
196 this->doMigrations();
197 }
198
199 /*virtual*/ bool acceptIncomingTask(model::Task const& task) override final {
200 RankInfo& cur_load = load_info_.at(this->pd_.getRank());
201 VT_LB_LOG(
202 LoadBalancer, verbose,
203 "BasicTransfer::acceptIncomingTask: current load={} task load={} total load={} max={}\n",
204 cur_load.load, task.getLoad(), cur_load.load + task.getLoad(), stats_.max
205 );
206 if ((cur_load.load + task.getLoad()) * cur_load.rank_alpha > stats_.max) {
207 return false;
208 } else {
209 cur_load.load += task.getLoad();
210 return true;
211 }
212 }
213
214private:
215 std::unordered_map<int, RankInfo> load_info_;
216 Statistics stats_;
217};
