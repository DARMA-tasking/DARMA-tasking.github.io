95enum struct CriterionEnum : uint8_t {
96 Grapevine = 0,
97 ModifiedGrapevine = 1
98};
99
100struct GrapevineCriterion {
101 bool operator()(RankInfo, RankInfo under, model::LoadType obj, model::LoadType avg) const {
102 return !(under.getScaledLoad() + obj * under.rank_alpha > avg);
103 }
104};
105
106struct ModifiedGrapevineCriterion {
107 bool operator()(RankInfo over, RankInfo under, model::LoadType obj, model::LoadType) const {
108 return obj * under.rank_alpha <= over.getScaledLoad() - under.getScaledLoad();
109 }
110};
111
112struct Criterion {
113 explicit Criterion(CriterionEnum const criterion)
114 : criterion_(criterion)
115 { }
116
117 bool operator()(RankInfo over, RankInfo under, model::LoadType obj, model::LoadType avg) const {
118 switch (criterion_) {
119 case CriterionEnum::Grapevine:
120 return GrapevineCriterion()(over, under, obj, avg);
121 break;
122 case CriterionEnum::ModifiedGrapevine:
123 return ModifiedGrapevineCriterion()(over, under, obj, avg);
124 break;
125 default:
126 vt_lb_assert(false, "Incorrect criterion value");
127 return false;
128 break;
129 };
130 }
131
132protected:
133 CriterionEnum const criterion_;
134};

136inline auto format_as(CriterionEnum c) {
137 std::string_view name = "Unknown";
138 switch (c) {
139 case CriterionEnum::Grapevine:
140 name = "Grapevine";
141 break;
142 case CriterionEnum::ModifiedGrapevine:
143 name = "ModifiedGrapevine";
144 break;
145 default:
146 name = "Unknown";
147 break;
148 }
149 return name;
150}
