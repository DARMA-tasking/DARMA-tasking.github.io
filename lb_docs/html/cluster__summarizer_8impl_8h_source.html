58template <typename CommT>
59void ClusterSummarizer<CommT>::resolveClusterIDForTask(
60 int from_rank,
61 model::TaskType task_id,
62 model::TaskType source_task_id,
63 int source_global_cluster_id
64) {
65 VT_LB_LOG(
66 LoadBalancer, verbose,
67 "ClusterSummarizer::resolveClusterIDForTask: trying to resolve task_id={}, source_task_id={}, "
68 "source_global_cluster_id={}, from_rank={}, found={}\n",
69 task_id, source_task_id, source_global_cluster_id, from_rank,
70 clusterer_->taskToCluster().find(task_id) != clusterer_->taskToCluster().end()
71 );
72
73 auto iter = clusterer_->taskToCluster().find(task_id);
74 vt_lb_assert(iter != clusterer_->taskToCluster().end(), "Task must be clustered here");
75 int local_cluster_id = iter->second;
76 int global_cluster_id = localToGlobalClusterID(
77 local_cluster_id,
78 comm_.getRank(),
79 global_max_clusters_
80 );
81 handle_[from_rank].template send<&ClusterSummarizer::recvClusterIDForTask>(
82 task_id, global_cluster_id
83 );
84 recvClusterIDForTask(source_task_id, source_global_cluster_id);
85}

100template <typename CommT>
101std::unordered_map<int, TaskClusterSummaryInfo>
102ClusterSummarizer<CommT>::buildClusterSummaries(
103 model::PhaseData const& pd,
104 Configuration const& config
105) {
106 vt_lb_assert(clusterer_ != nullptr, "Clusterer must be initialized to build summaries");
107 int const rank = pd.getRank();
108
109 // Validate assumption: every task must be assigned to a cluster
110 vt_lb_assert(allTasksClustered(*clusterer_, pd), "All tasks must exist in at least one cluster");
111
112 // Task -> local cluster id
113 auto const& t2c = clusterer_->taskToCluster();
114
115 // Prepare summary per cluster id, saved as the global cluster ID
116 std::unordered_map<int, TaskClusterSummaryInfo> summary_by_global;
117 for (auto const& cl : clusterer_->clusters()) {
118 TaskClusterSummaryInfo info;
119 info.cluster_id = localToGlobalClusterID(cl.id, rank, global_max_clusters_);
120 info.num_tasks_ = static_cast<int>(cl.members.size());
121 info.cluster_load = cl.load;
122 summary_by_global.emplace(info.cluster_id, std::move(info));
123 }
124
125 std::vector<model::Edge> to_resolve_later;
126
127 // Walk communications: accumulate intra send/recv; collect broadened inter edges starting in a cluster
128 for (auto const& e : pd.getCommunications()) {
129 auto u = e.getFrom();
130 auto v = e.getTo();
131
132 auto itu = t2c.find(u);
133 auto itv = t2c.find(v);
134 int cu = (itu != t2c.end()) ? itu->second : -1; // local cluster id or -1 if cluster is elsewhere
135 int cv = (itv != t2c.end()) ? itv->second : -1;
136 int cug = (cu != -1) ? localToGlobalClusterID(cu, rank, global_max_clusters_) : -1;
137 int cvg = (cv != -1) ? localToGlobalClusterID(cv, rank, global_max_clusters_) : -1;
138 auto vol = e.getVolume();
139
140 if ((cu == -1 && cv == -1) || (pd.getTask(u) == nullptr && pd.getTask(v) == nullptr)) {
141 // Both endpoints are remote: nothing to do here; or one of the tasks does not exist on this rank
142 continue;
143 }
144
145 // Intra-cluster: both endpoints mapped and equal -> accumulate send/recv
146 if (cu != -1 && cv != -1 && cu == cv) {
147 assert(
148 e.getFromRank() == e.getToRank() && e.getFromRank() == rank &&
149 "Intra-cluster edge must be intra-rank"
150 );
151 auto& sum = summary_by_global.at(cug);
152 sum.cluster_intra_send_bytes += vol;
153 sum.cluster_intra_recv_bytes += vol;
154 continue;
155 }
156
157 if (cu != -1 && cv != -1 && cu != cv) {
158 // Inter-cluster on same rank: do not count as intra, but add to inter edges for both clusters
159 model::ClusterEdge cluster_edge(
160 cug, // from cu
161 cvg, // to cv
162 vol
163 );
164 summary_by_global.at(cug).inter_edges_.push_back(cluster_edge);
165 summary_by_global.at(cvg).inter_edges_.push_back(cluster_edge);
166 continue;
167 }
168
169 VT_LB_LOG(
170 LoadBalancer, verbose,
171 "buildClusterSummaries: rank={} edge from {}(cu={}) to {}(cv={}) vol={:.2f}\n",
172 rank, u, cu, v, cv, vol
173 );
174
175 // Now, discover the target clusters that are not local to this rank
176 if (cu != -1) {
177 VT_LB_LOG(
178 LoadBalancer, verbose,
179 "ClusterSummarizer: resolveClusterIDForTask: from_rank={}, to_rank={} task_id={} source_task_id={} "
180 "source_global_cluster_id={}\n",
181 rank, e.getToRank(), u, v, cug
182 );
183 // Source cluster is local; destination is remote
184 auto to_rank = e.getToRank();
185 // Request cluster ID for destination task, send cluster ID for source task
186 handle_[to_rank].template send<&ClusterSummarizer::resolveClusterIDForTask>(
187 comm_.getRank(), e.getTo(), e.getFrom(), cug
188 );
189 to_resolve_later.push_back(e);
190 } else if (cv != -1) {
191 VT_LB_LOG(
192 LoadBalancer, verbose,
193 "ClusterSummarizer: resolveClusterIDForTask: from_rank={}, to_rank={} task_id={} source_task_id={} "
194 "source_global_cluster_id={}\n",
195 rank, e.getToRank(), u, v, cvg
196 );
197 to_resolve_later.push_back(e);
198 }
199 }
200
201 // Wait for termination
202 while (comm_.poll()) {
203 // Process incoming requests for cluster IDs
204 }
205
206 // After all cluster IDs are resolved, process deferred edges
207 for (auto const& e : to_resolve_later) {
208 auto u = e.getFrom();
209 auto v = e.getTo();
210 auto itu = t2c.find(u);
211 auto itv = t2c.find(v);
212 int cu = (itu != t2c.end()) ? itu->second : -1;
213 int cv = (itv != t2c.end()) ? itv->second : -1;
214 auto vol = e.getVolume();
215
216 // Only one endpoint is local, the other is remote
217 int local_cluster = (cu != -1) ? cu : cv;
218 int remote_task = (cu != -1) ? v : u;
219
220 // Find the global cluster ID for the remote task
221 auto it_remote_gid = task_to_global_cluster_id_.find(remote_task);
222 if (it_remote_gid == task_to_global_cluster_id_.end()) {
223 VT_LB_LOG(
224 LoadBalancer, normal,
225 "BUG: Missing global cluster ID for remote_task={} (local_cluster={}, cu={}, cv={}, rank={})\n",
226 remote_task, local_cluster, cu, cv, rank
227 );
228 }
229 assert(
230 it_remote_gid != task_to_global_cluster_id_.end() &&
231 "Should not happen if all resolutions are complete"
232 );
233
234 int remote_global_cluster_id = it_remote_gid->second;
235 int local_global_cluster_id = localToGlobalClusterID(local_cluster, rank, global_max_clusters_);
236
237 // Add the inter-cluster edge to the local cluster summary
238 model::ClusterEdge cluster_edge(
239 local_global_cluster_id,
240 remote_global_cluster_id,
241 vol
242 );
243
244 summary_by_global.at(local_global_cluster_id).inter_edges_.push_back(cluster_edge);
245 }
246
247 // Memory summaries per cluster (only if enabled)
248 if (config.hasMemoryInfo()) {
249 // Precompute a full set of task IDs for outside-cluster checks
250 std::unordered_set<model::TaskType> all_tasks;
251 for (auto const& kv : pd.getTasksMap()) {
252 all_tasks.insert(kv.first);
253 }
254
255 for (auto const& cl : clusterer_->clusters()) {
256 int global_cluster_id = localToGlobalClusterID(cl.id, rank, global_max_clusters_);
257 auto& sum = summary_by_global.at(global_cluster_id);
258
259 // Build set of tasks in this cluster
260 std::unordered_set<model::TaskType> cluster_tasks(cl.members.begin(), cl.members.end());
261
262 // Initialize maxima and footprint
263 model::BytesType max_working_inside = 0;
264 model::BytesType max_serialized_inside = 0;
265 model::BytesType max_working_outside = 0;
266 model::BytesType max_serialized_outside = 0;
267 model::BytesType total_footprint = 0;
268
269 // Collect shared blocks used by cluster tasks
270 std::unordered_set<model::SharedBlockType> shared_blocks_in_cluster;
271
272 // Iterate tasks to compute inside values and shared blocks
273 for (auto t : cl.members) {
274 auto it = pd.getTasksMap().find(t);
275 if (it == pd.getTasksMap().end()) continue;
276 auto const& task = it->second;
277
278 // Footprint sum
279 if (config.hasTaskFootprintMemoryInfo()) {
280 total_footprint += task.getMemory().getFootprint();
281 }
282
283 // Max inside working/serialized
284 if (config.hasTaskWorkingMemoryInfo()) {
285 max_working_inside = std::max(max_working_inside, task.getMemory().getWorking());
286 }
287 if (config.hasTaskSerializedMemoryInfo()) {
288 max_serialized_inside = std::max(max_serialized_inside, task.getMemory().getSerialized());
289 }
290
291 // Shared blocks union
292 if (config.hasSharedBlockMemoryInfo()) {
293 for (auto const& sb : task.getSharedBlocks()) {
294 shared_blocks_in_cluster.insert(sb);
295 }
296 }
297 }
298
299 // Compute max outside values by scanning tasks not in this cluster
300 if (config.hasTaskWorkingMemoryInfo() || config.hasTaskSerializedMemoryInfo()) {
301 for (auto const& kv : pd.getTasksMap()) {
302 auto const& task = kv.second;
303 if (cluster_tasks.find(task.getId()) != cluster_tasks.end()) continue; // skip inside
304 if (config.hasTaskWorkingMemoryInfo()) {
305 max_working_outside = std::max(max_working_outside, task.getMemory().getWorking());
306 }
307 if (config.hasTaskSerializedMemoryInfo()) {
308 max_serialized_outside = std::max(max_serialized_outside, task.getMemory().getSerialized());
309 }
310 }
311 }
312
313 // Fill shared_block_bytes_ with sizes
314 if (config.hasSharedBlockMemoryInfo()) {
315 for (auto const& sb : shared_blocks_in_cluster) {
316 if (!pd.hasSharedBlock(sb)) continue;
317 auto info = pd.getSharedBlock(sb);
318 sum.shared_block_bytes_[sb] = info->getSize();
319 }
320 }
321
322 // Store computed values on summary
323 sum.max_object_working_bytes = max_working_inside;
324 sum.max_object_serialized_bytes = max_serialized_inside;
325 sum.max_object_working_bytes_outside = max_working_outside;
326 sum.max_object_serialized_bytes_outside = max_serialized_outside;
327 sum.cluster_footprint = total_footprint;
328 }
329 }
330
331 // Emit summaries
332 for (auto const& cl : clusterer_->clusters()) {
333 auto global_cl_id = localToGlobalClusterID(cl.id, rank, global_max_clusters_);
334 auto const& sum = summary_by_global.at(global_cl_id);
335 VT_LB_LOG(
336 LoadBalancer,
337 verbose,
338 "buildClusterSummaries cluster {} size={} load={:.2f} intra_send={:.2f} intra_recv={:.2f} "
339 "inter_edges={} footprint={:.0f} max_work_in={:.0f} max_work_out={:.0f} "
340 "max_ser_in={:.0f} max_ser_out={:.0f} shared_block_count={}\n",
341 localToGlobalClusterID(cl.id, rank, global_max_clusters_),
342 cl.members.size(),
343 cl.load,
344 sum.cluster_intra_send_bytes,
345 sum.cluster_intra_recv_bytes,
346 sum.inter_edges_.size(),
347 sum.cluster_footprint,
348 sum.max_object_working_bytes,
349 sum.max_object_working_bytes_outside,
350 sum.max_object_serialized_bytes,
351 sum.max_object_serialized_bytes_outside,
352 sum.shared_block_bytes_.size()
353 );
354 }
355
356 return summary_by_global;
357}
