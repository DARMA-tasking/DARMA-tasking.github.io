65struct WorkModel {
67 double rank_alpha = 1.0;
69 double beta = 0.0;
71 double gamma = 0.0;
73 double delta = 0.0;
74
76 bool has_memory_info = true;
78 bool has_task_serialized_memory_info = true;
80 bool has_task_working_memory_info = true;
82 bool has_task_footprint_memory_info = true;
84 bool has_shared_block_memory_info = true;
85
96 inline double applyWorkFormula(
97 double compute, double inter_comm_bytes, double intra_comm_bytes,
98 double shared_comm_bytes
99 ) const {
100 return
101 rank_alpha * compute +
102 beta * inter_comm_bytes +
103 gamma * intra_comm_bytes +
104 delta * shared_comm_bytes;
105 }
106};
