66template <comm::Communicator CommT>
67struct Transferer {
68 using HandleType = typename CommT::template HandleType<Transferer<CommT>>;
69
76 Transferer(CommT& comm, model::PhaseData& pd)
77 : comm_(comm.clone()),
78 handle_(comm_.template registerInstanceCollective<Transferer<CommT>>(this)),
79 pd_(pd)
80 { }
81
82 virtual ~Transferer() = default;

84 struct TransferTask {
85 model::Task task;
86 std::vector<model::Edge> edges;
87 std::vector<model::SharedBlock> shared_blocks;
88
89 template <typename SerializerT>
90 void serialize(SerializerT& s) {
91 s | task;
92 s | edges;
93 s | shared_blocks;
94 }
95 };
96
97 void migrateTask(int const rank, model::Task const& task, bool include_comm = false) {
98 std::vector<model::Edge> edges;
99 std::vector<model::SharedBlock> shared_blocks;
100
101 if (include_comm) {
102 for (auto const& sb_id : task.getSharedBlocks()) {
103 auto const* sb = pd_.getSharedBlock(sb_id);
104 if (sb != nullptr) {
105 shared_blocks.push_back(*sb);
106 }
107 }
108 for (auto const& edge : pd_.getCommunications()) {
109 if (edge.getFrom() == task.getId() || edge.getTo() == task.getId()) {
110 edges.push_back(edge);
111 }
112 }
113 }
114
115 migrate_tasks_[rank].push_back(TransferTask{task, edges, shared_blocks});
116 pd_.eraseTask(task.getId());
117
118 // We can't clean up everything here, because the task might come back.
119 // Do this later once all migrations are done.
120 }
121
122 void doMigrations(bool do_cleanup = true) {
123 for (auto&& [rank, tasks] : migrate_tasks_) {
124 handle_[rank].template send<&Transferer::migrationHandler>(comm_.getRank(), tasks);
125 }
126
127 // Wait for termination on the migrations
128 while (comm_.poll()) {
129 // do nothing
130 }
131
132 migrate_tasks_.clear();
133
134 if (do_cleanup) {
135 cleanup();
136 }
137 }
138
139private:
140 void sendBackHandler(model::Task const& task) {
141 // Add the task back
142 pd_.addTask(task);
143 }
144
145 void migrationHandler(int from_rank, std::vector<TransferTask> const& tasks) {
146 for (auto& [task, edges, shared_blocks] : tasks) {
147 if (!acceptIncomingTask(task)) {
148 VT_LB_LOG(
149 LoadBalancer, normal,
150 "Transferer::migrationHandler: rejecting incoming task {}, load {} due to load constraints\n",
151 task.getId(),
152 task.getLoad()
153 );
154 handle_[from_rank].template send<&Transferer::sendBackHandler>(task);
155 } else {
156 pd_.addTask(task);
157 for (auto const& edge : edges) {
158 pd_.addCommunication(edge);
159 }
160 for (auto const& sb : shared_blocks) {
161 if (!pd_.hasSharedBlock(sb.getId())) {
162 pd_.addSharedBlock(sb);
163 }
164 }
165 }
166 }
167 }
168
169 void cleanup() {
170 // 1) Remove communications where neither endpoint exists locally
171 pd_.purgeDanglingCommunications();
172
173 // 2) Compute shared-block references from current tasks
174 std::unordered_map<model::SharedBlockType, int> refcounts;
175 for (auto const& [id, task] : pd_.getTasksMap()) {
176 for (auto const& sb : task.getSharedBlocks()) {
177 ++refcounts[sb];
178 }
179 }
180
181 // 3) Remove shared blocks with zero references on this rank
182 for (auto const& [sb_id, sb] : pd_.getSharedBlocksMap()) {
183 if (refcounts.find(sb_id) == refcounts.end()) {
184 pd_.eraseSharedBlock(sb_id);
185 }
186 }
187 }
188
189protected:
198 virtual bool acceptIncomingTask(
199 [[maybe_unused]] model::Task const& task
200 ) { return true; }
201
202protected:
203 CommT comm_;
204 HandleType handle_;
205 model::PhaseData& pd_;
206 std::unordered_map<int, std::vector<TransferTask>> migrate_tasks_;
207};
