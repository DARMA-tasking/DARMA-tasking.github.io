54// New concept helpers capturing common communicator API across CommMPI and CommVT
55namespace detail {
56
57 // Dummy class and member-function pointer used to instantiate templated send<fn>(...)
58 struct DummyClass {
59 void dummyMethod();
60 };
61 inline constexpr auto dummyMemFn = &DummyClass::dummyMethod;
62
63 // Concepts for required member functions
64
65 template <typename Comm>
66 concept HasInit = requires(Comm c, int& argc, char**& argv) {
67 { c.init(argc, argv) };
68 };
69
70 template <typename Comm>
71 concept HasFinalize = requires(Comm c) {
72 { c.finalize() };
73 };
74
75 template <typename Comm>
76 concept HasNumRanks = requires(Comm const& c) {
77 { c.numRanks() } -> std::integral;
78 };
79
80 template <typename Comm>
81 concept HasGetRank = requires(Comm const& c) {
82 { c.getRank() } -> std::integral;
83 };
84
85 template <typename Comm>
86 concept HasPoll = requires(Comm c) {
87 { c.poll() } -> std::convertible_to<bool>;
88 };
89
90 // registerInstanceCollective<T>(T*)
91 template <typename Comm, typename T>
92 concept HasRegisterInstanceCollective = requires(Comm c, T* t) {
93 { c.template registerInstanceCollective<T>(t) };
94 };
95
96 // Resolve instance handle type returned by registerInstanceCollective<T>(T*)
97 template <typename Comm, typename T>
98 using instance_handle_t =
99 decltype(std::declval<Comm&>().template registerInstanceCollective<T>(std::declval<T*>()));
100
101 // Unified send pattern: send<fn>(int dest, Handle proxy, ...)
102 template <typename Comm>
103 concept HasSendUnified =
104 HasRegisterInstanceCollective<Comm, DummyClass> &&
105 requires(Comm c) {
106 { c.template send<dummyMemFn>(
107 std::declval<int>(),
108 std::declval<instance_handle_t<Comm, DummyClass>>()
109 )
110 };
111 };
112
113} // namespace detail

115// Primary concept that a communicator must satisfy
116template <typename Comm>
117concept Communicator =
118 detail::HasInit<Comm> &&
119 detail::HasFinalize<Comm> &&
120 detail::HasNumRanks<Comm> &&
121 detail::HasGetRank<Comm> &&
122 detail::HasPoll<Comm> &&
123 detail::HasRegisterInstanceCollective<Comm, detail::DummyClass> &&
124 detail::HasSendUnified<Comm>;
125
126// Backward-compatible traits shim implemented in terms of concepts
127template <typename Comm>
128struct CommunicatorTraits {
129 static constexpr bool has_init = detail::HasInit<Comm>;
130 static constexpr bool has_finalize = detail::HasFinalize<Comm>;
131 static constexpr bool has_num_ranks = detail::HasNumRanks<Comm>;
132 static constexpr bool has_get_rank = detail::HasGetRank<Comm>;
133 static constexpr bool has_poll = detail::HasPoll<Comm>;
134
135 template <typename T>
136 using instance_handle_t = detail::instance_handle_t<Comm, T>;
137
138 template <typename T>
139 static constexpr bool has_register_instance =
140 detail::HasRegisterInstanceCollective<Comm, T>;
141
142 static constexpr bool has_send_unified =
143 detail::HasSendUnified<Comm>;
144
145 static constexpr bool is_valid = Communicator<Comm>;
146};
147
148// Convenience alias to check conformance at compile time
149template <typename Comm>
150using is_comm_conformant = std::bool_constant<Communicator<Comm>>;
