105 void clusterBasedOnCommunication() {
106 auto& pd = this->getPhaseData();
107 clusterer_ = std::make_unique<LeidenCPMStandaloneClusterer>(pd, 1000.0);
108 clusterer_->compute();
109 }

118 if (config_.cluster_based_on_communication_ || config_.cluster_based_on_shared_blocks_) {

122 clusterBasedOnSharedBlocks();

190 std::unordered_set<model::TaskType> run() {
191 // Make communications symmetric before running trials so we only have to do it once
192 makeCommunicationsSymmetric();
193
194 for (int trial = 0; trial < config_.num_trials_; ++trial) {
195 if (comm_.getRank() == 0) {
196 VT_LB_LOG(LoadBalancer, normal, "Starting trial {}/{}\n", trial + 1, config_.num_trials_);
197 }
198 runTrial(trial);
199 if (comm_.getRank() == 0) {
200 VT_LB_LOG(LoadBalancer, normal, "Finished trial {}/{}\n", trial + 1, config_.num_trials_);
201 }
202 }
203
204 // Sort trial work distribution by max work
205 std::sort(
206 trial_work_distribution_.begin(),
207 trial_work_distribution_.end(),
208 [](auto const& a, auto const& b) {
209 return std::get<0>(a) < std::get<0>(b);
210 }
211 );
212
213 if (comm_.getRank() == 0) {
214 VT_LB_LOG(
215 LoadBalancer, normal,
216 "Best trial: max work = {}\n", std::get<0>(trial_work_distribution_.front())
217 );
218 }
219
220 return std::get<1>(trial_work_distribution_.front());
221 }
222
223 void runTrial(int trial) {
224 // Save a clone of the phase data before load balancing
225 savePhaseData();
226
227 for (int iter = 0; iter < config_.num_iters_; ++iter) {
228 if (comm_.getRank() == 0) {
229 VT_LB_LOG(
230 LoadBalancer, normal,
231 " Starting iteration {}/{}\n",
232 iter + 1, config_.num_iters_
233 );
234 }
235
236 auto const& wm = config_.work_model_;
237 if (!(wm.beta == 0.0 && wm.gamma == 0.0 && wm.delta == 0.0)) {
238 // Edges might have the wrong rank after transfers, so fix them
239 resolveGraphEdges();
240
241 // Make communications symmetric
242 makeCommunicationsSymmetric();
243 }
244
245 runIteration(trial, iter);
246
247 if (comm_.getRank() == 0) {
248 VT_LB_LOG(
249 LoadBalancer, normal,
250 " Finished iteration {}/{}\n",
251 iter + 1, config_.num_iters_
252 );
253 }
254 }
255
256 // Before we restore phase data for the next trial, save the work and task distribution
257 // @todo: for now, we recompute work from scratch but we probably can use the breakdown
258 auto after_iters_work = WorkModelCalculator::computeWork(
259 config_.work_model_,
260 WorkModelCalculator::computeWorkBreakdown(this->getPhaseData(), config_)
261 );
262 auto final_stats = computeStatistics(after_iters_work, "Final Work After Iters");
263
264 // Save the max work and task distribution for this trial
265 trial_work_distribution_.emplace_back(
266 final_stats.max,
267 this->getPhaseData().getTaskIds()
268 );
269
270 // Restore phase data
271 restorePhaseData();
272 }

274 void runIteration(int trial, int iter) {
275 double total_load = computeLoad();
276 auto load_stats = computeStatistics(total_load, "Compute Load");
277
278 auto work_breakdown = WorkModelCalculator::computeWorkBreakdown(
279 this->getPhaseData(), config_
280 );
281 double const total_work = WorkModelCalculator::computeWork(
282 config_.work_model_, work_breakdown
283 );
284
285 VT_LB_LOG(LoadBalancer, normal, "Total work: {}\n", total_work);
286
287 auto work_stats = computeStatistics(total_work, "Work");
288
289 if (config_.hasMemoryInfo()) {
290 double const total_memory_usage = WorkModelCalculator::computeMemoryUsage(
291 config_,
292 this->getPhaseData()
293 ).current_memory_usage;
294 computeStatistics(total_memory_usage, "Memory Usage");
295 }
296
297 // Run the clustering algorithm if appropriate for the configuration
298 doClustering();
299
300 // Generate visualization after clustering
301 visualizeGraph(
302 "temperedlb_rank" + std::to_string(comm_.getRank()) +
303 "_trial" + std::to_string(trial) +
304 "_iter" + std::to_string(iter)
305 );
306
307 visualizeFullGraphIfNeeded(
308 comm_,
309 this->getPhaseData(),
310 getClusterer(),
311 global_max_clusters_,
312 config_,
313 "temperedlb_full_graph_trial" + std::to_string(trial) + "_iter" + std::to_string(iter)
314 );
315
316 auto& wm = config_.work_model_;
317 if (wm.beta == 0.0 && wm.gamma == 0.0 && wm.delta == 0.0) {
318 auto rank_info = RankInfo{total_work, config_.work_model_.rank_alpha};
319 auto info = runInformationPropagation(rank_info);
320 VT_LB_LOG(LoadBalancer, normal, "runTrial: gathered load info from {} ranks\n", info.size());
321 BasicTransfer<CommT> transfer(comm_, *phase_data_, info, work_stats);
322 std::mt19937 gen_select_;
323 std::random_device seed_;
324 transfer.run(
325 config_.cmf_type_,
326 config_.obj_ordering_,
327 config_.criterion_,
328 config_.deterministic_,
329 load_stats.avg,
330 gen_select_,
331 seed_
332 );
333 double const after_work = WorkModelCalculator::computeWork(
334 config_.work_model_,
335 WorkModelCalculator::computeWorkBreakdown(this->getPhaseData(), config_)
336 );
337 computeStatistics(after_work, "After Work");
338 } else {
339 // computeGlobalMaxClusters();
340 // Just assume max of 1000 clusters per rank for now, until we have bcast
341
342 // For now, we will assume that if beta/gamma/delta are non-zero, clustering must occur.
343 // Every task could be its own cluster, but clusters must exist
344 vt_lb_assert(clusterer_ != nullptr, "Clusterer must be valid");
345 auto local_summary = buildClusterSummaries();
346
347 double total_inter_bytes = std::max(
348 work_breakdown.inter_node_recv_comm, work_breakdown.inter_node_send_comm
349 );
350 computeStatistics(total_inter_bytes, "Work intercomm bytes");
351
352 auto rank_info = RankClusterInfo{
353 local_summary,
354 this->getPhaseData().getRankFootprintBytes(),
355 config_.work_model_.rank_alpha,
356 work_breakdown,
357 this->getPhaseData().getSharedBlockIdsHomed()
358 };
359 auto info = runInformationPropagation(rank_info);
360
361 VT_LB_LOG(
362 LoadBalancer, verbose,
363 "runTrial: gathered load info from {} ranks\n",
364 info.size()
365 );
366
367 RelaxedClusterTransfer<CommT> transfer(
368 comm_, *phase_data_, config_, clusterer_.get(), global_max_clusters_, info, work_stats
369 );
370 transfer.run();
371 }
372 }

395 void computeGlobalMaxClusters() {
396 // compute max number of clusters on any rank
397 int local_clusters = 0;
398 if (clusterer_) {
399 // assume Clusterer provides numClusters(); if not, set appropriately
400 local_clusters = static_cast<int>(clusterer_->clusters().size());
401 }
402
403 int const root = 0;
404 handle_.reduce(root, MPI_INT, MPI_MAX, &local_clusters, &global_max_clusters_, 1);
405
406 if (comm_.getRank() == root) {
407 VT_LB_LOG(
408 LoadBalancer, normal,
409 "global max clusters across ranks: {}\n",
410 global_max_clusters_
411 );
412 }
413 // @todo: once we have a bcast, broadcast global_max_clusters_ to all ranks
414 }

416 template <typename T>
417 Statistics computeStatistics(T quantity, std::string const& name) {
418 // Compute min, max, avg of quantity across all ranks
419 double local_value = static_cast<double>(quantity);
420 double global_min = 0.0;
421 double global_max = 0.0;
422 double global_sum = 0.0;
423
424 handle_.reduce(0, MPI_DOUBLE, MPI_MIN, &local_value, &global_min, 1);
425 handle_.reduce(0, MPI_DOUBLE, MPI_MAX, &local_value, &global_max, 1);
426 handle_.reduce(0, MPI_DOUBLE, MPI_SUM, &local_value, &global_sum, 1);
427
428 handle_.broadcast(0, MPI_DOUBLE, &global_min, 1);
429 handle_.broadcast(0, MPI_DOUBLE, &global_max, 1);
430 handle_.broadcast(0, MPI_DOUBLE, &global_sum, 1);
431
432 double global_avg = global_sum / static_cast<double>(comm_.numRanks());
433 double I = 0;
434 if (global_avg > 0.0) {
435 I = (global_max / global_avg) - 1.0;
436 }
437 if (comm_.getRank() == 0) {
438 VT_LB_LOG(
439 LoadBalancer, normal, "{} statistics -- min: {}, max: {}, avg: {}, I: {}\n",
440 name, global_min, global_max, global_avg, I
441 );
442 }
443 return Statistics{global_min, global_max, global_avg, I};
444 }
